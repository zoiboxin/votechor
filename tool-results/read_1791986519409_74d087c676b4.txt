    67→/** True when (x,z) sits inside any non-walkable collider inflated by `pad` (walkable platforms are step-ups, fine). */
    68→function releasePointBlocked(x: number, z: number, pad: number): boolean {
    69→  for (const c of WORLD.colliders) {
    70→    if (c.walkable) continue
    71→    if (x >= c.minX - pad && x <= c.maxX + pad && z >= c.minZ - pad && z <= c.maxZ + pad) return true
    72→  }
    73→  return false
    74→}

    76→/** Validated release points (colliding candidates nudged on a small deterministic spiral, then rejected). */
    77→export const RELEASE_POINTS: ReleasePoint[] = (() => {
    78→  const out: ReleasePoint[] = []
    79→  for (const c of RELEASE_CANDIDATES) {
    80→    if (!releasePointBlocked(c.x, c.z, RELEASE_PAD)) { out.push(c); continue }
    81→    let placed = false
    82→    for (const [dx, dz] of [[2, 0], [-2, 0], [0, 2], [0, -2], [3, 0], [-3, 0], [0, 3], [0, -3], [4, 0], [-4, 0], [0, 4], [0, -4]]) {
    83→      const nx = c.x + dx, nz = c.z + dz
    84→      if (nx < -46 || nx > 46 || nz < -568 || nz > 7) continue
    85→      if (!releasePointBlocked(nx, nz, RELEASE_PAD)) { out.push({ ...c, x: nx, z: nz }); placed = true; break }
    86→    }
    87→    if (!placed) console.log(`[room] release point REJECTED (collides with world): ${c.id}`)
    88→  }
    89→  return out
    90→})()

    92→/**
    93→ * Pick a random release destination (Phase 6).
    94→ * Filters: (a) never the player's lastReleaseId, (b) ≥30 m from any non-detained
    95→ * squadmate — but only while ≥2 alternatives remain, (c) outside the boss arena zone,
    96→ * (d) still clear of world colliders. Fallbacks: relax the squad filter → any point →
    97→ * the spawn plaza (legacy detention behavior).
    98→ */
    99→export function pickReleasePoint(
   100→  lastReleaseId: string | null,
   101→  squadPositions: ReadonlyArray<{ x: number; z: number }>,
   102→): ReleasePoint {
   103→  const arena = WORLD.bossArena
   104→  const inArena = (x: number, z: number) => Math.hypot(x - arena.x, z - arena.z) < arena.radius + 4
   105→  const base = RELEASE_POINTS.filter((pt) =>
   106→    pt.id !== lastReleaseId && !inArena(pt.x, pt.z) && !releasePointBlocked(pt.x, pt.z, RELEASE_PAD))
   107→  let pool = base.filter((pt) => !squadPositions.some((s) => Math.hypot(s.x - pt.x, s.z - pt.z) < 30))
   108→  if (pool.length < 2) pool = base // keep ≥2 alternatives: drop the squadmate filter first
   109→  if (pool.length === 0) pool = RELEASE_POINTS.filter((pt) => !inArena(pt.x, pt.z))
   110→  if (pool.length === 0) pool = RELEASE_POINTS.slice()
   111→  if (pool.length === 0) return { id: 'spawn', x: WORLD.spawn.x, z: WORLD.spawn.z, label: 'Protest Plaza' }
   112→  return randomOf(pool)
   113→}

   332→  /** Store a validated player state update. Detained players are ignored. */
   333→  handleState(socketId: string, s: { x?: unknown; y?: unknown; z?: unknown; ry?: unknown; anim?: unknown; carrying?: unknown }) {
   334→    const p = this.players.get(socketId)
   335→    if (!p || p.detained) return // detained → freeze, ignore updates
   336→    const now = Date.now()
   337→
   338→    let x = clamp(num(s.x, p.x), -BOUNDS_X, BOUNDS_X)
   339→    let z = clamp(num(s.z, p.z), BOUNDS_Z_MIN, BOUNDS_Z_MAX)
   340→    const y = Math.min(num(s.y, p.y), MAX_STATE_Y) // Phase 3/10: impossible altitude is clamped (walls are 14 m)
   341→    const ry = num(s.ry, p.ry)
   342→
   343→    // Boss gate: while closed, nobody slips through the barrier corridor
   344→    if (!this.gateOpen && z < WORLD.bossArena.gateZ - 1 && Math.abs(x) < 11) {
   345→      z = WORLD.bossArena.gateZ + 1 // snap back (-477)
   346→    }
   347→
   348→    // Speed check: teleport jumps snap back to last valid position
   349→    if (dist2d(x, z, p.lastX, p.lastZ) > MAX_STATE_DIST) {
   350→      x = p.lastX
   351→      z = p.lastZ
   352→    }
   353→
   354→    p.x = x
   355→    p.y = y
   356→    p.z = z
   357→    p.ry = ry
   358→    const a = typeof s.anim === 'string' ? (s.anim as AnimName) : 'idle'
   359→    p.anim = ANIMS.includes(a) ? a : 'idle'
   360→    p.carrying = Boolean(s.carrying) // cosmetic flag; boss ballots are validated by proximity
   361→    p.lastX = x
   362→    p.lastZ = z
   363→    p.lastStateTime = now
   364→  }

   366→  /** Validate + apply pickup. Returns true if the pickup happened. */
   367→  pickup(socketId: string, itemId: unknown): boolean {
   368→    const p = this.players.get(socketId)
   369→    if (!p || p.detained) return false
   370→    if (typeof itemId !== 'string') return false
   371→    const def = WORLD.items.find(i => i.id === itemId)
   372→    const st = this.itemStates.get(itemId)
   373→    if (!def || !st || st.taken) return false
   374→    if (dist2d(p.x, p.z, def.x, def.z) > PICKUP_RADIUS) return false
   375→
   376→    st.taken = true
   377→    st.takenBy = p.id
   378→    if (def.type === 'token') {
   379→      this.tokens++
   380→      p.tokens++
   381→      st.respawnAt = undefined // tokens are single-use
   382→      this.emitRoom('game:tokens', this.tokens)
   383→      this.checkGate()
   384→    } else {
   385→      st.respawnAt = Date.now() + ITEM_RESPAWN_MS
   386→    }
   387→    this.emitRoom('game:items', this.itemsArray())
   388→    return true
   389→  }

   400→  /** Rate limiter: max 5 events/s per player per channel. */
   401→  rateLimit(p: PlayerState, channel: 'noiseTimes' | 'emoteTimes' | 'giftTimes' | 'inviteTimes' | 'kudosTimes'): boolean {
   402→    const now = Date.now()
   403→    const arr = p[channel]
   404→    while (arr.length && now - arr[0] > RATE_LIMIT_WINDOW_MS) arr.shift()
   405→    if (arr.length >= RATE_LIMIT_MAX) return false
   406→    arr.push(now)
   407→    return true
   408→  }

   434→  /**
   435→   * Gift an item to a nearby player (§11 eligible item sharing).
   436→   * Server validates: type whitelist, rate limit, neither detained, distance ≤ GIFT_RADIUS.
   437→   * Inventory lives client-side, so the receiver may bounce the gift back if full.
   438→   */
   439→  gift(byId: string, targetId: unknown, itemType: unknown): boolean {
   440→    const by = this.players.get(byId)
   441→    const target = typeof targetId === 'string' ? this.players.get(targetId) : undefined
   442→    if (!by || !target || by.detained || target.detained) return false
   443→    if (by.id === target.id) return false
   444→    const type = String(itemType ?? '')
   445→    if (!GIFT_TYPES.includes(type)) return false
   446→    if (!this.rateLimit(by, 'giftTimes')) return false
   447→    if (dist2d(by.x, by.z, target.x, target.z) > GIFT_RADIUS) return false
   448→    // targeted emit — only the receiver sees the incoming gift
   449→    this.io.to(target.id).emit('game:gift', { fromId: by.id, fromName: by.name, type })
   450→    return true
   451→  }

   485→  bossDeposit(playerId: string): boolean {
   486→    const p = this.players.get(playerId)
   487→    if (!p || p.detained || p.dazed) return false
   488→    const ok = depositBallot(this.boss, p.x, p.z)
   489→    if (ok) {
   490→      p.carrying = false
   491→      this.emitBoss()
   492→    }
   493→    return ok
   494→  }
   495→
   496→  private checkGate() {
   497→    if (!this.gateOpen && this.tokens >= TOKENS_NEEDED) {
   498→      this.gateOpen = true
   499→      this.emitRoom('game:gate', true)
   500→    }
   501→  }
   502→
   503→  // ---------- damage / capture / detention ----------
   504→
   505→  /** Server-owned health. Emits game:damage + game:health. 0 hp → dazed. */
   506→  damagePlayer(playerId: string, amount: number, kind: string, now: number) {
   507→    const p = this.players.get(playerId)
   508→    if (!p || p.dazed || p.detained) return
   509→    p.health = Math.max(0, p.health - amount)
   510→    this.emitRoom('game:damage', p.id, amount, kind)
   511→    if (p.health <= 0) {
   512→      p.dazed = true
   513→      p.dazedUntil = now + DAMAGE.dazedDuration * 1000
   514→      p.carrying = false
   515→    }
   516→    this.emitRoom('game:health', p.id, p.health, p.dazed)
   517→  }
   518→
   519→  private capturePlayer(playerId: string, now: number) {
   520→    const p = this.players.get(playerId)
   521→    if (!p || p.detained || p.dazed) return
   522→    p.detained = true
   523→    p.detainUntil = now + DETENTION_SECONDS * 1000 * (p.archetype === 'accidental' ? 0.75 : 1)
   524→    p.carrying = false
   525→    p.anim = 'detained'
   526→    this.emitRoom('game:capture', p.id, p.detainUntil, randomOf(CAPTURE_MESSAGES))
   527→  }
   528→
   529→  /**
   530→   * Release to a random validated point (Phase 6): never the same point twice in a
   531→   * row, ≥30 m from non-detained squadmates while alternatives remain, never inside
   532→   * the boss arena. Detention duration semantics are unchanged.
   533→   */
   534→  private releasePlayer(p: PlayerState) {
   535→    const squad = this.squadmatePositions
   536→      ? this.squadmatePositions(p.id)
   537→      : [...this.humans()].filter((h) => h.id !== p.id && !h.detained).map((h) => ({ x: h.x, z: h.z }))
   538→    const dest = pickReleasePoint(p.lastReleaseId, squad)
   539→    p.lastReleaseId = dest.id
   540→    p.detained = false
   541→    p.detainUntil = 0
   542→    p.x = dest.x
   543→    p.z = dest.z
   544→    p.y = 0
   545→    p.lastX = dest.x
   546→    p.lastZ = dest.z
   547→    p.lastStateTime = Date.now()
   548→    p.anim = 'idle'
   549→    // 4th arg `label` is additive — existing clients ignore extra args
   550→    this.emitRoom('game:release', p.id, p.x, p.z, dest.label)
   551→  }

   553→  // ---------- tick ----------
   554→
   555→  private tick() {
   556→    if (this.disposed) return
   557→    const now = Date.now()
   558→    const dt = TICK_MS / 1000
   559→    this.tickCount++
   560→
   561→    // Eligible player views: NPCs and boss ignore detained/dazed players.
   562→    // Fresh joiners get an 8s spawn grace so nobody is captured while loading in.
   563→    const views: NpcPlayerView[] = []
   564→    const bossViews: NpcPlayerView[] = [] // humans only — bots never start the boss fight
   565→    for (const p of this.players.values()) {
   566→      if (!p.detained && !p.dazed) {
   567→        const grace = now - p.joinedAt < 10000 && dist2d(p.x, p.z, WORLD.spawn.x, WORLD.spawn.z) < 16
   568→        const view: NpcPlayerView = { id: p.id, x: p.x, z: p.z, detained: p.detained, dazed: p.dazed, hidden: grace, bot: p.bot, health: p.health }
   569→        views.push(view)
   570→        if (!p.bot) bossViews.push(view)
   571→      }
   572→    }
   573→
   574→    // 1. NPC brains
   575→    for (const n of this.npcs) {
   576→      const res = tickNpc(n, views, WORLD.colliders, dt, now)
   577→      for (const ev of res.events) {
   578→        if (ev.kind === 'capture') this.capturePlayer(ev.playerId, now)
   579→        else if (ev.kind === 'hit') {
   580→          // Phase 5 police melee — damagePlayer emits game:damage (kind 'police') + game:health,
   581→          // and 0 hp → dazed. No extra event: clients react to game:damage.
   582→          this.damagePlayer(ev.playerId, ev.amount ?? POLICE_ATTACK.fallbackDmg, 'police', now)
   583→        }
   584→      }
   585→    }
   586→
   587→    // 2. Searchlight rotation
   588→    for (const sl of this.searchlights) sl.ry += sl.speed * dt
   589→
   590→    // 3. Vehicle collisions (positions deterministic from matchStartedAt)
   591→    this.vehicleDamage(now)
   592→
   593→    // 4. Daze auto-recovery
   594→    for (const p of this.players.values()) {
   595→      if (p.dazed && now >= p.dazedUntil) {
   596→        p.dazed = false
   597→        p.dazedUntil = 0
   598→        p.health = DAMAGE.dazedRecoverHealth
   599→        this.emitRoom('game:health', p.id, p.health, p.dazed)
   600→      }
   601→    }
   602→
   603→    // 5. Detention timers → release to a random validated point (Phase 6)
   604→    for (const p of this.players.values()) {
   605→      if (p.detained && now >= p.detainUntil) this.releasePlayer(p)
   606→    }
   607→
   608→    // 6. Checkpoint activation
   609→    if (!this.ended) {
   610→      for (const p of this.players.values()) {
   611→        if (p.detained) continue
   612→        for (const cp of WORLD.checkpoints) {
   613→          if (this.checkpoints.has(cp.id)) continue
   614→          if (dist2d(p.x, p.z, cp.x, cp.z) <= cp.radius) {
   615→            this.checkpoints.add(cp.id)
   616→            this.emitRoom('game:checkpoint', cp.id, p.name, this.checkpoints.size)
   617→          }
   618→        }
   619→      }
   620→    }
   621→
   622→    // 7. Boss brain — only real protesters can wake the Vote Collector
   623→    const bossEv = tickBoss(this.boss, bossViews, dt, now)
   624→    for (const d of bossEv.damage) this.damagePlayer(d.playerId, d.amount, d.kind, now)
   625→
   626→    // 8. Match end (victory or 900s cap) → emit once, then reset the room
   627→    const elapsed = (now - this.matchStartedAt) / 1000
   628→    if (!this.ended && (bossEv.defeated || elapsed >= MATCH_TIME_LIMIT)) {
   629→      this.ended = true
   630→      this.emitMatchEnd(bossEv.defeated, now)
   631→      this.resetMatch()
   632→      this.ended = false
   633→    }
   634→
   635→    // 9. Item respawns
   636→    let itemsChanged = false
   637→    for (const st of this.itemStates.values()) {
   638→      if (st.taken && st.respawnAt !== undefined && now >= st.respawnAt) {
   639→        st.taken = false
   640→        st.takenBy = undefined
   641→        st.respawnAt = undefined
   642→        itemsChanged = true
   643→      }
   644→    }
   645→    if (itemsChanged) this.emitRoom('game:items', this.itemsArray())
   646→
   647→    // 10. NPC snapshots at 10 Hz (searchlights included)
   648→    if (this.players.size > 0) this.emitRoom('game:npcs', this.npcsArray())
   649→
   650→    // 11. Boss broadcasts: 10 Hz while active, 1 Hz idle
   651→    if (this.boss.started) this.emitBoss()
   652→    else if (this.tickCount % 10 === 0) this.emitBoss()
   653→  }
   654→
   655→  private vehicleDamage(now: number) {
   656→    const elapsed = (now - this.matchStartedAt) / 1000
   657→    for (const v of WORLD.vehicles) {
   658→      const [vx, vz] = vehiclePosition(v.path, v.speed, elapsed)
   659→      const radius = VEHICLE_RADIUS[v.kind] ?? 1.7
   660→      const dmg = VEHICLE_DAMAGE[v.kind] ?? 35
   661→      for (const p of this.players.values()) {
   662→        if (p.detained || p.dazed) continue
   663→        const last = this.vehicleHitAt.get(p.id) ?? 0
   664→        if (now - last < VEHICLE_HIT_COOLDOWN_MS) continue
   665→        if (dist2d(p.x, p.z, vx, vz) < radius) {
   666→          this.vehicleHitAt.set(p.id, now)
   667→          this.damagePlayer(p.id, dmg, v.kind, now)
   668→        }
   669→      }
   670→    }
   671→  }

   673→  private emitMatchEnd(victory: boolean, now: number) {
   674→    const playersOnline = Object.keys(this.io.sockets.sockets).length
   675→    const payload: MatchEndPayload = {
   676→      matchId: 'm_' + Math.random().toString(36).slice(2, 10),
   677→      victory,
   678→      duration: Math.round((now - this.matchStartedAt) / 1000),
   679→      position: 1,
   680→      tokens: this.tokens,
   681→      checkpoints: this.checkpoints.size,
   682→      distance: 0, // per-player distance is tracked client-side
   683→      rescues: 0,
   684→      revives: 0,
   685→      itemsCollected: 0,
   686→      bossContribution: 0,
   687→      detentions: 0,
   688→      playersOnline,
   689→    }
   690→    this.emitRoom('game:matchend', payload)
   691→  }
   692→
   693→  /** Fresh match state: new clock, reset items/npcs/boss/checkpoints/tokens/gate. */
   694→  private resetMatch() {
   695→    this.matchStartedAt = Date.now()
   696→    this.checkpoints.clear()
   697→    this.tokens = 0
   698→    this.gateOpen = false
   699→    this.npcs = createNpcStates(WORLD.npcs)
   700→    this.boss = createBossBrain(WORLD.bossArena)
   701→    this.itemStates = new Map(WORLD.items.map(i => [i.id, { taken: false } as ItemState]))
   702→    this.vehicleHitAt.clear()
   703→    for (const p of this.players.values()) {
   704→      if (p.detained) {
   705→        p.detained = false
   706→        p.detainUntil = 0
   707→      }
   708→      if (p.dazed) {
   709→        p.dazed = false
   710→        p.dazedUntil = 0
   711→      }
   712→      p.health = p.maxHealth
   713→      p.carrying = false
   714→    }
   715→    this.emitRoom('game:items', this.itemsArray())
   716→    this.emitRoom('game:tokens', this.tokens)
   717→    this.emitRoom('game:gate', false)
   718→    this.emitBoss()
   719→  }

   723→  playersArray(includeCosmetics = false): PlayerNetState[] {
   724→    const out: PlayerNetState[] = []
   725→    for (const p of this.players.values()) out.push(playerNet(p, includeCosmetics))
   726→    return out

   729→  npcsArray(): NpcNetState[] {
   730→    const out: NpcNetState[] = []
   731→    const now = Date.now()
   732→    for (const n of this.npcs) {
   733→      const net: NpcNetState = { id: n.id, x: round2(n.x), z: round2(n.z), ry: round2(n.ry), state: n.state, alert: round2(n.alert) }
   734→      // Phase 5 telegraph fields — only sent while meaningful (keeps the 10 Hz payload small):
   735→      // 'windup' + attackAt (absolute server ms the strike lands → client times impact FX),
   736→      // or 'stagger' for the 0.35 s window after a landed strike.
   737→      if (n.attackTelegraphUntil > now) {
   738→        net.anim = 'windup'
   739→        net.attackAt = n.attackTelegraphUntil
   740→      } else if (n.staggerUntil > now) {
   741→        net.anim = 'stagger'
   742→      }
   743→      out.push(net)
   744→    }
   745→    for (const sl of this.searchlights) {
   746→      out.push({ id: sl.id, x: WORLD.searchlights.find(s => s.id === sl.id)?.x ?? 0, z: WORLD.searchlights.find(s => s.id === sl.id)?.z ?? 0, ry: round2(sl.ry), state: 'idle', alert: 0 })
   747→    }
   748→    return out
   749→  }

   751→  itemsArray(): ItemNetState[] {
   752→    const out: ItemNetState[] = []
   753→    for (const [id, st] of this.itemStates) {
   754→      out.push({ id, taken: st.taken, takenBy: st.takenBy, respawnAt: st.respawnAt })
   755→    }
   756→    return out
   757→  }

   759→  bossPayload(): { boss: BossNetState; ballots: { id: string; x: number; z: number; taken: boolean }[] } {
   760→    const b = this.boss
   761→    return {
   762→      boss: {
   763→        active: b.active,
   764→        phase: b.phase,
   765→        health: b.health,
   766→        maxHealth: b.maxHealth,
   767→        stateName: b.stateName,
   768→        timer: round2(b.timer),
   769→        speechProgress: round2(b.speech),
   770→        interruptPoints: round2(b.interruptPoints),
   771→        documents: round2(b.documents),
   772→        hits: b.hits,
   773→        dizzyTimer: round2(b.dizzyTimer),
   774→        teamworkProgress: round2(b.teamworkProgress),
   775→        votes: b.votes,
   776→        votesNeeded: b.votesNeeded,
   777→        x: b.x,
   778→        z: b.z,
   779→        ry: b.ry,
   780→        telegraphs: b.telegraphs,
   781→        stations: b.stations,
   782→        paperwork: b.paperwork.map(h => ({ id: h.id, x: round2(h.x), z: round2(h.z), vx: round2(h.vx), vz: round2(h.vz) })),
   783→      },
   784→      ballots: b.ballots.map(bl => ({ id: bl.id, x: bl.x, z: bl.z, taken: bl.taken })),
   785→    }
   786→  }
   787→
   788→  emitBoss() {
   789→    const { boss, ballots } = this.bossPayload()
   790→    this.emitRoom('game:boss', boss, ballots)
   791→  }
   792→
   793→  emitPlayers() {
   794→    if (this.disposed || this.players.size === 0) return
   795→    if (this.playersBroadcaster) { this.playersBroadcaster(); return }
   796→    this.emitRoom('game:players', this.playersArray())
   797→  }
   798→}

   800→/** Wire format for one player/bot — shared by the legacy and AOI broadcast paths. */
   801→export function playerNet(p: PlayerState, includeCosmetics = false): PlayerNetState {
   802→  const net: PlayerNetState = {
   803→    id: p.id,
   804→    name: p.name,
   805→    archetype: p.archetype,
   806→    x: round2(p.x),
   807→    y: round2(p.y),
   808→    z: round2(p.z),
   809→    ry: round2(p.ry),
   810→    anim: p.anim,
   811→    health: p.health,
   812→    detained: p.detained,
   813→    dazed: p.dazed,
   814→    tokens: p.tokens,
   815→    carrying: p.carrying,
   816→  } as PlayerNetState
   817→  if (includeCosmetics && p.cosmetics) net.cosmetics = p.cosmetics
   818→  return net
   819→}

   821→// ---------- deterministic vehicles (mirrors client VehicleSystem ping-pong) ----------
   822→
   823→/** 2-point linear ping-pong — identical formula to client npcs.ts VehicleSystem. */
   824→function vehiclePosition(path: Array<[number, number]>, speed: number, worldTime: number): [number, number] {
   825→  const a = path[0]
   826→  const b = path[path.length - 1]
   827→  const len = Math.hypot(b[0] - a[0], b[1] - a[1])
   828→  const period = (len / speed) * 2
   829→  const phase = ((worldTime % period) + period) % period
   830→  const half = period / 2
   831→  const t = phase < half ? phase / half : 2 - phase / half // ping-pong 0..1..0
   832→  return [a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t]
   833→}
   834→
   835→function round2(v: number): number {
   836→  return Math.round(v * 100) / 100
   837→}
