    27→  /** OPT-IN far-LOD (game.ts keys it to the performance-visibility setting). Default
    28→   *  false = every NPC rig always drawn — the proven v25 rendering path. */
    29→  farLod = false
    30→  group = new THREE.Group()
    31→  bubbles = new SpeechBubbleSystem()

    39→  constructor(
    40→    world: WorldDef,
    41→    private physics: PhysicsWorld,
    42→    quality: 'low' | 'medium' | 'high',
    43→  ) {
    44→    const npcs = quality === 'low' ? world.npcs : world.npcs
    45→    // patrol passed RAW — createNpcStates handles [x,z] tuples properly; pre-spreading
    46→    // ({...p}) exploded tuples into {0:x,1:z}, making wp.x undefined → NaN-frozen NPCs offline
    47→    const brains = createNpcStates(npcs.map(n => ({ id: n.id, type: n.type, x: n.x, z: n.z, patrol: n.patrol })))
    48→    for (const b of brains) {
    49→      if (b.type === 'searchlight') continue
    50→      const rig = new CharacterRig({ colors: NPC_COLORS[b.type] ?? NPC_COLORS.police, npcType: b.type as 'police', scale: b.type === 'security' ? 1.04 : 1 })
    51→      rig.group.position.set(b.x, 0, b.z)
    52→      rig.setNameVisible(false)
    53→      this.group.add(rig.group)
    54→      const v: NpcVisual = {
    55→        brain: b, rig,
    56→        prev: { x: b.x, z: b.z, ry: b.ry },
    57→        target: { x: b.x, z: b.z, ry: b.ry },
    58→        alertShown: false,
    59→      }
    60→      this.visuals.push(v)
    61→      this.brainById.set(b.id, b)
    62→    }
    63→    this.group.add(this.bubbles.group)
    64→  }

    66→  /** Local offline simulation (solo, server unreachable) */
    67→  simulateLocal(dt: number, player: NpcPlayerView, colliders: AABB[], now: number) {
    68→    this.localSim = true
    69→    this.simAccum += dt
    70→    const step = 1 / 10
    71→    while (this.simAccum >= step) {
    72→      this.simAccum -= step
    73→      const events: import('./shared/npc-brain').NpcEvent[] = []
    74→      for (const v of this.visuals) {
    75→        const r = tickNpc(v.brain, [player], colliders, step, now)
    76→        events.push(...r.events)
    77→        v.prev = { ...v.target }
    78→        v.target = { x: v.brain.x, z: v.brain.z, ry: v.brain.ry }
    79→      }
    80→      for (const ev of events) {
    81→        if (ev.kind === 'capture') this.onLocalCapture?.(ev.npcId)
    82→        else if (ev.kind === 'hit') this.onLocalHit?.(ev.amount ?? 12)
    83→      }
    84→    }
    85→  }
    86→
    87→  onLocalCapture: ((npcId: string) => void) | null = null
    88→  onLocalHit: ((amount: number) => void) | null = null
    89→  onNoise: ((x: number, z: number) => void) | null = null
    90→
    91→  distraction(x: number, z: number, duration: number) {
    92→    applyNoise(this.visuals.map(v => v.brain), x, z, duration)
    93→    this.onNoise?.(x, z)
    94→  }

    96→  /** Apply server snapshot (10Hz) */
    97→  applyServerState(states: NpcNetState[]) {
    98→    this.localSim = false
    99→    const now = performance.now()
   100→    this.lastNetTime = now
   101→    for (const s of states) {
   102→      const v = this.visuals.find(v => v.brain.id === s.id)
   103→      if (!v) continue
   104→      v.prev = { ...v.target }
   105→      v.target = { x: s.x, z: s.z, ry: s.ry }
   106→      // brain mirrors the latest server truth — minimapData(), nearestChaserDistance(),
   107→      // speech-bubble proximity and hit-feel knockback all read brain.x/z, which were
   108→      // silently stuck at world-def SPAWN positions in online mode (latent bug found
   109→      // while fixing the WANTED banner: distance checks compared against stale spawns)
   110→      v.brain.x = s.x
   111→      v.brain.z = s.z
   112→      v.brain.ry = s.ry
   113→      v.brain.state = s.state
   114→      v.brain.alert = s.alert
   115→      // chase target — lets nearestChaserDistance() tell "I am the quarry" from
   116→      // "some officer across town is chasing a bot" (WANTED banner correctness)
   117→      v.brain.targetId = s.target ?? null
   118→    }
   119→  }

   121→  update(dt: number, alpha: number, t: number, px?: number, pz?: number) {
   122→    // ---- far LOD (OPT-IN via game.ts, keyed to the performance-visibility setting):
   123→    // NPC rigs beyond 70m are sub-pixel silhouettes in fog — hide the body and skip
   124→    // its pose update entirely (AI/minimaps/speech logic untouched: everything reads
   125→    // brain.x/z, and speech targets are always <34m away). A hidden rig saves ~25
   126→    // draw calls + a full procedural pose pass per NPC. Default OFF = the proven
   127→    // v25 rendering path where every rig is always drawn.
   128→    const LOD_DIST = 70
   129→    const lod = this.farLod && px !== undefined && pz !== undefined
   130→    for (const v of this.visuals) {
   131→      if (lod) {
   132→        const far = Math.hypot(v.target.x - px, v.target.z - pz) > LOD_DIST
   133→        if (far) {
   134→          if (v.rig.group.visible) v.rig.group.visible = false
   135→          continue
   136→        } else if (!v.rig.group.visible) {
   137→          v.rig.group.visible = true
   138→        }
   139→      }
   140→      // interpolate
   141→      const x = v.prev.x + (v.target.x - v.prev.x) * alpha
   142→      const z = v.prev.z + (v.target.z - v.prev.z) * alpha
   143→      let dry = v.target.ry - v.prev.ry
   144→      while (dry > Math.PI) dry -= Math.PI * 2
   145→      while (dry < -Math.PI) dry += Math.PI * 2
   146→      const ry = v.prev.ry + dry * alpha
   147→      const dx = v.target.x - v.prev.x, dz = v.target.z - v.prev.z
   148→      const speed = Math.hypot(dx, dz) * 10 // per second
   149→      const st: NpcStateName = v.brain.state
   150→      const anim = st === 'chase' ? 'run' : st === 'investigate' || st === 'return' ? 'walk' : st === 'capture' ? 'throw' : speed > 0.3 ? 'walk' : 'idle'
   151→      v.rig.update(dt, anim as 'idle', Math.min(9, speed))
   152→      v.rig.group.position.set(x, v.rig.group.position.y * 0 + 0, z)
   153→      v.rig.group.rotation.y = ry
   154→      // alert indicator
   155→      const showAlert = st === 'chase' || (st !== 'patrol' && v.brain.alert > 0.25)
   156→      if (showAlert !== v.alertShown) { v.rig.setAlert(showAlert); v.alertShown = showAlert }
   157→    }
   158→    // ambient satire speech bubbles (chasing officers get their own lines)
   159→    this.bubbles.update(dt)
   160→    if (px !== undefined && pz !== undefined) {
   161→      this.speechCooldown -= dt
   162→      if (this.speechCooldown <= 0) {
   163→        this.speechCooldown = 3.6 + Math.random() * 3.4
   164→        const nearby = this.visuals.filter(v => Math.hypot(v.brain.x - px, v.brain.z - pz) < 34)
   165→        if (nearby.length) {
   166→          const pick = nearby[Math.floor(Math.random() * nearby.length)]
   167→          const chasing = pick.brain.state === 'chase'
   168→          const lines = chasing ? NPC_LINES.chase : NPC_LINES[pick.brain.type] ?? NPC_LINES.police
   169→          const text = lines[Math.floor(Math.random() * lines.length)]
   170→          const id = pick.brain.id
   171→          this.bubbles.say(`npc-${id}`, text, () => {
   172→            const vv = this.visuals.find(v2 => v2.brain.id === id)
   173→            return vv ? vv.rig.group.position : null
   174→          }, chasing ? 2.6 : 4.2)
   175→        }
   176→      }
   177→    }
   178→  }
   179→
   180→  /** Info for minimap: positions + chase states */
   181→  minimapData() {
   182→    return this.visuals.map(v => ({ id: v.brain.id, x: v.brain.x, z: v.brain.z, chasing: v.brain.state === 'chase', state: v.brain.state }))
   183→  }
   184→
   185→  /**
   186→   * Distance to the nearest officer ACTIVELY CHASING `targetId` (omit for "any chaser").
   187→   * Target-aware since round 25: a Season officer chasing a bot across town must not
   188→   * flash the WANTED banner on an unrelated player's screen.
   189→   */
   190→  nearestChaserDistance(px: number, pz: number, targetId?: string): number {
   191→    let best = 1e9
   192→    for (const v of this.visuals) {
   193→      if (v.brain.state !== 'chase') continue
   194→      if (targetId !== undefined && v.brain.targetId !== targetId) continue
   195→      const d = Math.hypot(v.brain.x - px, v.brain.z - pz)
   196→      if (d < best) best = d
   197→    }
   198→    return best
   199→  }
   200→}

   240→const matCache = new Map<string, THREE.MeshStandardMaterial>()
   241→function flatMat(color: string, opts: Partial<THREE.MeshStandardMaterialParameters> = {}) {
   242→  const key = color + '|' + JSON.stringify(opts)
   243→  let m = matCache.get(key)
   244→  if (!m) { m = new THREE.MeshStandardMaterial({ color, flatShading: true, ...opts }); matCache.set(key, m) }
   245→  return m
   246→}

   255→const wheelGeoms = new Map<string, THREE.CylinderGeometry>()
   256→function cachedCyl(key: string, make: () => THREE.CylinderGeometry) {
   257→  let g = wheelGeoms.get(key)
   258→  if (!g) { g = make(); wheelGeoms.set(key, g) }
   259→  return g
   260→}
   261→function wheelGeo(r: number, w: number) { return cachedCyl(`t${r}:${w}`, () => new THREE.CylinderGeometry(r, r, w, 10)) }
   262→function rimGeo(r: number, w: number) { return cachedCyl(`r${r}:${w}`, () => new THREE.CylinderGeometry(r * 0.52, r * 0.52, w + 0.03, 8)) }

   264→/** buildWheels — cylinder wheels rolled around X (rotation.z=π/2, spin = rotation.x), grouped per axle. */
   265→function buildWheels(
   266→  count: number,
   267→  positions: Array<[number, number]>, // [x, z]; front axle = largest z (steers when opts.steer)
   268→  radius: number,
   269→  opts: { width?: number; steer?: boolean; rims?: boolean | 'front' } = {},
   270→): { group: THREE.Group; wheels: VehicleWheel[] } {
   271→  const width = opts.width ?? radius * 0.68
   272→  const frontZ = Math.max(...positions.map(p => p[1]))
   273→  const group = new THREE.Group()
   274→  const axles = new Map<number, THREE.Group>()
   275→  const wheels: VehicleWheel[] = []
   276→  for (let i = 0; i < count; i++) {
   277→    const [wx, wz] = positions[i % positions.length]
   278→    let axle = axles.get(wz)
   279→    if (!axle) { axle = new THREE.Group(); axle.position.z = wz; group.add(axle); axles.set(wz, axle) }
   280→    const isFront = opts.steer === true && wz === frontZ
   281→    const pivot = new THREE.Group()
   282→    pivot.position.x = wx
   283→    axle.add(pivot)
   284→    const tire = new THREE.Mesh(wheelGeo(radius, width), tireMat)
   285→    tire.rotation.z = Math.PI / 2
   286→    tire.castShadow = false // perf: bodies only
   287→    pivot.add(tire)
   288→    if (opts.rims === true || (opts.rims === 'front' && isFront)) {
   289→      const rim = new THREE.Mesh(rimGeo(radius, width), rimMat)
   290→      rim.rotation.z = Math.PI / 2
   291→      rim.castShadow = false
   292→      tire.add(rim) // child of the tire → inherits spin + steer
   293→    }
   294→    wheels.push({ pivot, spin: tire, radius, steers: isFront })
   295→  }
   296→  return { group, wheels }
   297→}

   299→// ---- primitive-baking helpers (build-time only; everything merges into few meshes per vehicle) ----
   300→function box(w: number, h: number, d: number, x = 0, y = 0, z = 0, rx = 0): THREE.BoxGeometry {
   301→  const g = new THREE.BoxGeometry(w, h, d)
   302→  if (rx) g.rotateX(rx)
   303→  g.translate(x, y, z)
   304→  return g
   305→}
   306→function taperTop(g: THREE.BoxGeometry, sx: number, sz: number) {
   307→  const pos = g.attributes.position
   308→  for (let i = 0; i < pos.count; i++) if (pos.getY(i) > 0) { pos.setX(i, pos.getX(i) * sx); pos.setZ(i, pos.getZ(i) * sz) }
   309→  pos.needsUpdate = true
   310→  g.computeVertexNormals()
   311→  return g
   312→}
   313→function merged(parts: THREE.BufferGeometry[], mat: THREE.Material, castShadow = true): THREE.Mesh {
   314→  const geo = mergeGeometries(parts, false)
   315→  const mesh = new THREE.Mesh(geo ?? parts[0], mat)
   316→  if (geo) for (const p of parts) p.dispose()
   317→  mesh.castShadow = castShadow
   318→  return mesh
   319→}

   595→export class VehicleSystem {
   596→  group = new THREE.Group()
   597→  vehicles: VehicleVisual[] = []
   598→  onPlayerHit: ((kind: string, x: number, z: number, dmg: number) => void) | null = null
   599→  private hitCooldown = 0
   600→
   601→  constructor(world: WorldDef) {
   602→    for (let i = 0; i < world.vehicles.length; i++) {
   603→      const v = world.vehicles[i]
   604→      const built = v.kind === 'bus' ? buildBus(i)
   605→        : v.kind === 'cow' ? buildCow()
   606→        : v.kind === 'auto' ? buildAuto()
   607→        : v.kind === 'bike' ? buildBike(i)
   608→        : buildCar(i) // 'car' kind (and safe fallback for any future kind)
   609→      built.group.name = `vehicle-${v.kind}-${v.id}`
   610→      this.group.add(built.group)
   611→      this.vehicles.push({ id: v.id, kind: v.kind, group: built.group, path: v.path.map(([px, pz]) => ({ x: px, z: pz })), speed: v.speed, wheels: built.wheels, prevHeading: NaN, steer: 0 })
   612→    }
   613→  }
   614→
   615→  /** Deterministic update from shared world time (seconds since match start) */
   616→  update(worldTime: number, dt: number, playerPos: THREE.Vector3 | null, localAuth: boolean) {
   617→    this.hitCooldown = Math.max(0, this.hitCooldown - dt)
   618→    const step = Math.max(dt, 1e-4)
   619→    for (const v of this.vehicles) {
   620→      const a = v.path[0], b = v.path[1]
   621→      const len = Math.hypot(b.x - a.x, b.z - a.z)
   622→      const period = (len / v.speed) * 2
   623→      const phase = ((worldTime % period) + period) % period
   624→      const half = period / 2
   625→      const t = phase < half ? phase / half : 2 - phase / half // ping-pong 0..1..0
   626→      const x = a.x + (b.x - a.x) * t
   627→      const z = a.z + (b.z - a.z) * t
   628→      const dir = phase < half ? 1 : -1
   629→      v.group.position.set(x, 0, z)
   630→      const heading = Math.atan2(b.x - a.x, b.z - a.z) * dir
   631→      // steer angle ≈ signed turn rate of heading, clamped ±0.5 rad (paths are straight → ~0, kicks at U-turns)
   632→      if (Number.isFinite(v.prevHeading)) {
   633→        let dh = heading - v.prevHeading
   634→        while (dh > Math.PI) dh -= Math.PI * 2
   635→        while (dh < -Math.PI) dh += Math.PI * 2
   636→        const turn = Math.max(-0.5, Math.min(0.5, dh / step))
   637→        v.steer += (turn - v.steer) * Math.min(1, step * 7)
   638→      }
   639→      v.prevHeading = heading
   640→      v.group.rotation.y = heading
   641→      // wheels spin by actual travel speed; front pivots take the steer angle
   642→      for (const w of v.wheels) {
   643→        w.spin.rotation.x += (v.speed * dt) / w.radius
   644→        if (w.steers) w.pivot.rotation.y = v.steer
   645→      }
   646→      // collision with local player (per-kind ring + damage, mirrored server-side)
   647→      if (playerPos && localAuth && this.hitCooldown <= 0) {
   648→        const hit = VEHICLE_HIT[v.kind] ?? { r: 1.7, dmg: 35 }
   649→        const d = Math.hypot(playerPos.x - x, playerPos.z - z)
   650→        if (d < hit.r) {
   651→          this.hitCooldown = 1.2
   652→          this.onPlayerHit?.(v.kind, x, z, hit.dmg)
   653→        }
   654→      }
   655→    }
   656→  }
   657→
   658→  /** (2-d contract) One heading-agnostic solid AABB per vehicle — movement workstream consumes these. */
   659→  getSolidAABBs(): AABB[] {
   660→    const pad = 0.05
   661→    const out: AABB[] = []
   662→    for (const v of this.vehicles) {
   663→      const e = VEHICLE_SOLID_EXTENTS[v.kind] ?? VEHICLE_SOLID_EXTENTS.car
   664→      const x = v.group.position.x
   665→      const z = v.group.position.z
   666→      out.push({ minX: x - e.hx - pad, minY: 0, minZ: z - e.hz - pad, maxX: x + e.hx + pad, maxY: e.maxY, maxZ: z + e.hz + pad })
   667→    }
   668→    return out
   669→  }
