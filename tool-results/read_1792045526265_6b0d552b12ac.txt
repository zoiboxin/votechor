   122→   122→function mat(color: string, flat = true): THREE.MeshStandardMaterial {
   123→   123→  return new THREE.MeshStandardMaterial({ color, roughness: 0.85, metalness: 0.02, flatShading: flat })
   124→   124→}
   125→   125→
   126→   126→/** Lighten/darken a hex color by ±0-255 per channel (string in, string out). */
   127→   127→function shade(hex: string, amt: number): string {
   128→   128→  const c = new THREE.Color(hex)
   129→   129→  const f = (v: number) => Math.max(0, Math.min(1, v + amt / 255))
   130→   130→  return `#${new THREE.Color(f(c.r), f(c.g), f(c.b)).getHexString()}`
   131→   131→}
   132→   132→
   133→   133→// ---- realistic face features (round 23) ----
   134→   134→// Eyes + brows + nose + mouth are merged into ONE vertex-colored geometry per
   135→   135→// unique (skin, hair, gender) triple — a single extra draw call per rig.
   136→   136→function tintedClone(geo: THREE.BufferGeometry, color: string): THREE.BufferGeometry {
   137→   137→  const c = new THREE.Color(color)
   138→   138→  const src = geo.clone()
   139→   139→  const count = src.attributes.position.count
   140→   140→  const arr = new Float32Array(count * 3)
   141→   141→  for (let i = 0; i < count; i++) {
   142→   142→    arr[i * 3] = c.r; arr[i * 3 + 1] = c.g; arr[i * 3 + 2] = c.b
   143→   143→  }
   144→   144→  src.setAttribute('color', new THREE.BufferAttribute(arr, 3))
   145→   145→  return src
   146→   146→}

   158→   158→/** Build (or fetch from cache) the merged face-features geometry for a look. */
   159→   159→function faceGeometryFor(skinHex: string, hairHex: string, female: boolean): THREE.BufferGeometry {
   160→   160→  const key = `${skinHex}|${hairHex}|${female ? 'f' : 'm'}`
   161→   161→  const hit = faceGeoCache.get(key)
   162→   162→  if (hit) return hit
   163→   163→  const parts: THREE.BufferGeometry[] = []
   164→   164→  const place = (
   165→   165→    geo: THREE.BufferGeometry, color: string,
   166→   166→    px: number, py: number, pz: number,
   167→   167→    sx = 1, sy = 1, sz = 1, rz = 0,
   168→   168→  ) => {
   169→   169→    const g = tintedClone(geo, color)
   170→   170→    const m = new THREE.Matrix4()
   171→   171→    m.compose(
   172→   172→      new THREE.Vector3(px, py, pz),
   173→   173→      new THREE.Quaternion().setFromEuler(new THREE.Euler(0, 0, rz)),
   174→   174→      new THREE.Vector3(sx, sy, sz),
   175→   175→    )
   176→   176→    g.applyMatrix4(m)
   177→   177→    parts.push(g)
   178→   178→  }
   179→   179→  const dark = '#241a12'
   180→   180→  // eyes — white sclera flattened into the head, dark pupil just proud of it
   181→   181→  for (const side of [1, -1]) {
   182→   182→    place(G.eye, '#f7f3ea', side * 0.056, 0.112, 0.126, 1, 0.85, 0.55)
   183→   183→    place(G.pupil, dark, side * 0.056, 0.112, 0.139)
   184→   184→  }
   185→   185→  // brows — hair colored, slight expressive tilt
   186→   186→  for (const side of [1, -1]) {
   187→   187→    place(G.brow, shade(hairHex, 10), side * 0.056, 0.153, 0.132, 1, 1, 1, side * -0.13)
   188→   188→  }
   189→   189→  // nose — skin shaded down, gentle snub shape
   190→   190→  place(G.nose, shade(skinHex, -26), 0, 0.082, 0.147, 0.9, 1.2, 0.85)
   191→   191→  // mouth — fuller red lips for female, thinner shaded line for male
   192→   192→  if (female) place(G.lip, '#b2596b', 0, 0.034, 0.138, 1, 1, 0.8)
   193→   193→  else place(G.mouth, shade(skinHex, -48), 0, 0.04, 0.141, 1, 1, 0.8)
   194→   194→  const merged = mergeGeometries(parts, false) ?? new THREE.BufferGeometry()
   195→   195→  for (const p of parts) p.dispose()
   196→   196→  faceGeoCache.set(key, merged)
   197→   197→  return merged
   198→   198→}

   844→   844→  private makeTextTexture(text: string): THREE.CanvasTexture {
   845→   845→    const canvas = document.createElement('canvas')

   858→   858→    for (const w of words) {

   862→   862→    if (cur.trim()) lines.push(cur.trim())
   863→   863→    lines.forEach((l, i) => ctx.fillText(l, 128, 62 - (lines.length - 1) * 17 + i * 34))

   865→   865→    tex.colorSpace = THREE.SRGBColorSpace

   924→   924→  update(dt: number, anim: AnimName, moveSpeed: number) {
   925→   925→    this.anim = anim
   926→   926→    if (this.forceAnim) anim = this.forceAnim
   927→   927→    this.speedRatio = Math.min(1, moveSpeed / 9)
   928→   928→    const p = (this.phase += dt * (anim === 'run' ? 11 : anim === 'walk' ? 7.2 : 1.6))
   929→   929→    const s = Math.sin(p), c = Math.cos(p)
   930→   930→    const swing = anim === 'run' ? 1.05 : anim === 'walk' ? 0.62 : 0
   931→   931→
   932→   932→    const set = (o: THREE.Object3D, x: number, z = 0) => { o.rotation.x += (x - o.rotation.x) * Math.min(1, dt * 14); if (z !== 0 || o.rotation.z !== 0) o.rotation.z += (0 - o.rotation.z) * Math.min(1, dt * 14) }
   933→   933→
   934→   934→    switch (anim) {
   935→   935→      case 'walk':
   936→   936→      case 'run': {
   937→   937→        const lean = anim === 'run' ? 0.22 : 0.08
   938→   938→        this.torsoPivot.rotation.x += (lean - this.torsoPivot.rotation.x) * Math.min(1, dt * 8)
   939→   939→        set(this.legL, s * swing)
   940→   940→        set(this.legR, -s * swing)
   941→   941→        if (!this.armBlockR) {
   942→   942→          set(this.armR, s * swing * 0.9)
   943→   943→        }
   944→   944→        set(this.armL, this.armBlockL ? 0.15 : -s * swing * 0.9)
   945→   945→        this.group.position.y = this.baseY + Math.abs(s) * (anim === 'run' ? 0.06 : 0.03)
   946→   946→        break
   947→   947→      }
   948→   948→      case 'jump': {
   949→   949→        this.torsoPivot.rotation.x += (0.1 - this.torsoPivot.rotation.x) * Math.min(1, dt * 8)
   950→   950→        set(this.legL, -0.5); set(this.legR, 0.25)
   951→   951→        set(this.armL, -2.6); set(this.armR, -2.6)
   952→   952→        // double-jump front-flip flourish (procedural, rides on top of the jump pose)
   953→   953→        if (this.flipTimer > 0) {
   954→   954→          this.flipTimer = Math.max(0, this.flipTimer - dt)
   955→   955→          const t = 1 - this.flipTimer / this.flipDur
   956→   956→          const eased = t < 0.7 ? (t / 0.7) * (Math.PI * 2) : Math.PI * 2 // flip then hold
   957→   957→          this.torsoPivot.rotation.x = eased
   958→   958→        }
   959→   959→        break
   960→   960→      }
   961→   961→      case 'slide': {
   962→   962→        this.torsoPivot.rotation.x += (-1.15 - this.torsoPivot.rotation.x) * Math.min(1, dt * 10)
   963→   963→        set(this.legL, 0.4); set(this.legR, -0.15)
   964→   964→        set(this.armL, -2.9); set(this.armR, -2.7)
   965→   965→        break
   966→   966→      }
   967→   967→      case 'dazed':
   968→   968→      case 'down': {
   969→   969→        this.torsoPivot.rotation.x += (-0.55 - this.torsoPivot.rotation.x) * Math.min(1, dt * 6)
   970→   970→        this.headPivot.rotation.z = Math.sin(p * 2) * 0.25
   971→   971→        set(this.legL, 0.35); set(this.legR, -0.3)

   973→   973→        break
   974→   974→      }
   975→   975→      case 'detained': {
   976→   976→        this.torsoPivot.rotation.x += (0.12 - this.torsoPivot.rotation.x) * Math.min(1, dt * 6)
   977→   977→        set(this.armL, 0.55, 0); set(this.armR, 0.55, 0)
   978→   978→        this.armL.rotation.z = 0.5; this.armR.rotation.z = -0.5
   979→   979→        this.headPivot.rotation.x = 0.35
   980→   980→        break
   981→   981→      }
   982→   982→      case 'celebrate': {
   983→   983→        this.torsoPivot.rotation.x += (0 - this.torsoPivot.rotation.x) * Math.min(1, dt * 8)
   984→   984→        set(this.armL, -2.9 + s * 0.3); set(this.armR, -2.9 - s * 0.3)
   985→   985→        this.group.position.y = this.baseY + Math.abs(c) * 0.14
   986→   986→        this.headPivot.rotation.x = 0
   987→   987→        break
   988→   988→      }
   989→   989→      case 'throw': {
   990→   990→        set(this.armR, -2.4); set(this.armL, 0.4)
   991→   991→        this.torsoPivot.rotation.x += (0.15 - this.torsoPivot.rotation.x) * dt * 10
   992→   992→        break
   993→   993→      }
   994→   994→      case 'heal': {
   995→   995→        set(this.armL, -1.7); set(this.armR, -1.7)
   996→   996→        this.headPivot.rotation.x = 0.1
   997→   997→        break
   998→   998→      }
   999→   999→      default: { // idle
  1000→  1000→        this.torsoPivot.rotation.x += (0.03 - this.torsoPivot.rotation.x) * Math.min(1, dt * 6)
  1001→  1001→        this.headPivot.rotation.x = Math.sin(p * 0.5) * 0.04
  1002→  1002→        set(this.legL, 0); set(this.legR, 0)
  1003→  1003→        if (!this.armBlockL) {
  1004→  1004→          set(this.armL, 0.04 + Math.sin(p * 0.8) * 0.03)
  1005→  1005→        }
  1006→  1006→        if (!this.armBlockR) {
  1007→  1007→          set(this.armR, 0.04 - Math.sin(p * 0.8) * 0.03)
  1008→  1008→        }
  1009→  1009→        this.group.position.y = this.baseY + Math.sin(p * 0.9) * 0.012
  1010→  1010→      }
  1011→  1011→    }
  1012→  1012→
  1013→  1013→    // cloth life: flag ripples, banner sways with the stride
  1014→  1014→    if (this.flagClothRef) {
  1015→  1015→      this.flagClothRef.rotation.y = Math.sin(p * 1.4) * 0.35 + 0.12
  1016→  1016→      this.flagClothRef.rotation.z = Math.sin(p * 0.9) * 0.06
  1017→  1017→    }
  1018→  1018→    if (this.bannerClothRef) {
  1019→  1019→      this.bannerClothRef.rotation.z = Math.sin(p * 0.8) * 0.05
  1020→  1020→      this.bannerClothRef.rotation.x = Math.sin(p * 1.1) * 0.04
  1021→  1021→    }
  1022→  1022→  }
