    41→  constructor(
    42→    world: WorldDef,
    43→    private physics: PhysicsWorld,
    44→    quality: 'low' | 'medium' | 'high',
    45→  ) {
    46→    const npcs = quality === 'low' ? world.npcs : world.npcs
    47→    // patrol passed RAW — createNpcStates handles [x,z] tuples properly; pre-spreading
    48→    // ({...p}) exploded tuples into {0:x,1:z}, making wp.x undefined → NaN-frozen NPCs offline
    49→    const brains = createNpcStates(npcs.map(n => ({ id: n.id, type: n.type, x: n.x, z: n.z, patrol: n.patrol })))
    50→    for (const b of brains) {
    51→      if (b.type === 'searchlight') continue
    52→      const rig = new CharacterRig({ colors: NPC_COLORS[b.type] ?? NPC_COLORS.police, npcType: b.type as 'police', scale: b.type === 'security' ? 1.04 : 1 })
    53→      rig.group.position.set(b.x, 0, b.z)
    54→      rig.setNameVisible(false)
    55→      this.group.add(rig.group)
    56→      const v: NpcVisual = {
    57→        brain: b, rig,
    58→        prev: { x: b.x, z: b.z, ry: b.ry },
    59→        target: { x: b.x, z: b.z, ry: b.ry },
    60→        alertShown: false,
    61→      }
    62→      this.visuals.push(v)
    63→      this.brainById.set(b.id, b)
    64→    }
    65→    this.group.add(this.bubbles.group)
    66→  }

    68→  /** Local offline simulation (solo, server unreachable) */
    69→  simulateLocal(dt: number, player: NpcPlayerView, colliders: AABB[], now: number) {
    70→    this.localSim = true
    71→    this.simAccum += dt
    72→    const step = 1 / 10
    73→    while (this.simAccum >= step) {
    74→      this.simAccum -= step
    75→      const events: import('./shared/npc-brain').NpcEvent[] = []
    76→      for (const v of this.visuals) {
    77→        const r = tickNpc(v.brain, [player], colliders, step, now)
    78→        events.push(...r.events)
    79→        v.prev = { ...v.target }
    80→        v.target = { x: v.brain.x, z: v.brain.z, ry: v.brain.ry }
    81→      }
    82→      for (const ev of events) {
    83→        if (ev.kind === 'capture') this.onLocalCapture?.(ev.npcId)
    84→        else if (ev.kind === 'hit') this.onLocalHit?.(ev.amount ?? 12)
    85→      }
    86→    }
    87→  }

    93→  distraction(x: number, z: number, duration: number) {

    98→  /** Apply server snapshot (10Hz) */
    99→  applyServerState(states: NpcNetState[]) {
   100→    this.localSim = false
   101→    const now = performance.now()
   102→    this.lastNetTime = now
   103→    for (const s of states) {
   104→      const v = this.visuals.find(v => v.brain.id === s.id)
   105→      if (!v) continue
   106→      v.prev = { ...v.target }
   107→      v.target = { x: s.x, z: s.z, ry: s.ry }
   108→      // brain mirrors the latest server truth — minimapData(), nearestChaserDistance(),
   109→      // speech-bubble proximity and hit-feel knockback all read brain.x/z, which were
   110→      // silently stuck at world-def SPAWN positions in online mode (latent bug found
   111→      // while fixing the WANTED banner: distance checks compared against stale spawns)
   112→      v.brain.x = s.x
   113→      v.brain.z = s.z
   114→      v.brain.ry = s.ry
   115→      v.brain.state = s.state
   116→      v.brain.alert = s.alert
   117→      // chase target — lets nearestChaserDistance() tell "I am the quarry" from
   118→      // "some officer across town is chasing a bot" (WANTED banner correctness)
   119→      v.brain.targetId = s.target ?? null
   120→    }
   121→  }

   123→  update(dt: number, alpha: number, t: number, px?: number, pz?: number) {
   124→    // ---- far LOD (OPT-IN via game.ts, keyed to the performance-visibility setting):
   125→    // NPC rigs beyond 70m are sub-pixel silhouettes in fog — hide the body and skip
   126→    // its pose update entirely (AI/minimaps/speech logic untouched: everything reads
   127→    // brain.x/z, and speech targets are always <34m away). A hidden rig saves ~25
   128→    // draw calls + a full procedural pose pass per NPC. Default OFF = the proven
   129→    // v25 rendering path where every rig is always drawn.
   130→    const LOD_DIST = 70
   131→    const lod = this.farLod && px !== undefined && pz !== undefined
   132→    for (const v of this.visuals) {
   133→      if (lod) {
   134→        const far = Math.hypot(v.target.x - px, v.target.z - pz) > LOD_DIST
   135→        if (far) {
   136→          if (v.rig.group.visible) v.rig.group.visible = false
   137→          continue
   138→        } else if (!v.rig.group.visible) {
   139→          v.rig.group.visible = true
   140→        }
   141→      }
   142→      // interpolate
   143→      const x = v.prev.x + (v.target.x - v.prev.x) * alpha
   144→      const z = v.prev.z + (v.target.z - v.prev.z) * alpha
   145→      let dry = v.target.ry - v.prev.ry
   146→      while (dry > Math.PI) dry -= Math.PI * 2
   147→      while (dry < -Math.PI) dry += Math.PI * 2
   148→      const ry = v.prev.ry + dry * alpha
   149→      const dx = v.target.x - v.prev.x, dz = v.target.z - v.prev.z
   150→      const speed = Math.hypot(dx, dz) * 10 // per second
   151→      const st: NpcStateName = v.brain.state
   152→      const anim = st === 'chase' ? 'run' : st === 'investigate' || st === 'return' ? 'walk' : st === 'capture' ? 'throw' : speed > 0.3 ? 'walk' : 'idle'
   153→      v.rig.update(dt, anim as 'idle', Math.min(9, speed))
   154→      v.rig.group.position.set(x, v.rig.group.position.y * 0 + 0, z)
   155→      v.rig.group.rotation.y = ry
   156→      // alert indicator
   157→      const showAlert = st === 'chase' || (st !== 'patrol' && v.brain.alert > 0.25)
   158→      if (showAlert !== v.alertShown) { v.rig.setAlert(showAlert); v.alertShown = showAlert }
   159→    }
   160→    // ambient satire speech bubbles (chasing officers get their own lines)
   161→    this.bubbles.update(dt)
   162→    if (px !== undefined && pz !== undefined) {
   163→      this.speechCooldown -= dt
   164→      if (this.speechCooldown <= 0) {
   165→        this.speechCooldown = 3.6 + Math.random() * 3.4
   166→        const nearby = this.visuals.filter(v => Math.hypot(v.brain.x - px, v.brain.z - pz) < 34)
   167→        if (nearby.length) {
   168→          const pick = nearby[Math.floor(Math.random() * nearby.length)]
   169→          const chasing = pick.brain.state === 'chase'
   170→          const lines = chasing ? NPC_LINES.chase : NPC_LINES[pick.brain.type] ?? NPC_LINES.police
   171→          const text = lines[Math.floor(Math.random() * lines.length)]
   172→          const id = pick.brain.id
   173→          this.bubbles.say(`npc-${id}`, text, () => {
   174→            const vv = this.visuals.find(v2 => v2.brain.id === id)
   175→            return vv ? vv.rig.group.position : null
   176→          }, chasing ? 2.6 : 4.2)
   177→        }
   178→      }
   179→    }
   180→  }
   181→
   182→  /** Info for minimap: positions + chase states */
   183→  minimapData() {
   184→    return this.visuals.map(v => ({ id: v.brain.id, x: v.brain.x, z: v.brain.z, chasing: v.brain.state === 'chase', state: v.brain.state }))
   185→  }
   186→
   187→  /**
   188→   * Distance to the nearest officer ACTIVELY CHASING `targetId` (omit for "any chaser").
   189→   * Target-aware since round 25: a Season officer chasing a bot across town must not
   190→   * flash the WANTED banner on an unrelated player's screen.
   191→   */
   192→  nearestChaserDistance(px: number, pz: number, targetId?: string): number {
   193→    let best = 1e9
   194→    for (const v of this.visuals) {
   195→      if (v.brain.state !== 'chase') continue
   196→      if (targetId !== undefined && v.brain.targetId !== targetId) continue
   197→      const d = Math.hypot(v.brain.x - px, v.brain.z - pz)
   198→      if (d < best) best = d
   199→    }
   200→    return best
   201→  }
   202→}

   267→let vehicleEnvTex: THREE.Texture | null = null
   268→let vehicleEnvRenderer: THREE.WebGLRenderer | null = null
   269→/** Bake a small neutral-studio environment once per renderer; null on any failure. */
   270→function ensureVehicleEnv(renderer: THREE.WebGLRenderer): THREE.Texture | null {
   271→  if (vehicleEnvRenderer === renderer) return vehicleEnvTex
   272→  try {
   273→    const pmrem = new THREE.PMREMGenerator(renderer)
   274→    const rt = pmrem.fromScene(new RoomEnvironment(), 0.04)
   275→    vehicleEnvTex = rt.texture
   276→    pmrem.dispose()
   277→    vehicleEnvRenderer = renderer
   278→  } catch {
   279→    vehicleEnvTex = null
   280→  }
   281→  return vehicleEnvTex
   282→}
   283→/** Assign the env map to every vehicle material (shared caches → guarded by envDone). */
   284→function applyVehicleEnv(root: THREE.Object3D) {
   285→  if (!vehicleEnvTex) return
   286→  root.traverse((o) => {
   287→    const mesh = o as THREE.Mesh
   288→    if (!mesh.isMesh) return
   289→    const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material]
   290→    for (const m of mats) {
   291→      const mat = m as THREE.MeshStandardMaterial & { userData: { envDone?: boolean; envInt?: number } }
   292→      if (!mat || !mat.isMaterial || mat.userData.envDone) continue
   293→      mat.envMap = vehicleEnvTex
   294→      mat.envMapIntensity = typeof mat.userData.envInt === 'number' ? mat.userData.envInt : 0.45
   295→      mat.userData.envDone = true
   296→      mat.needsUpdate = true
   297→    }
   298→  })
   299→}

   313→const wheelGeoms = new Map<string, THREE.CylinderGeometry>()
   314→function cachedCyl(key: string, make: () => THREE.CylinderGeometry) {
   315→  let g = wheelGeoms.get(key)
   316→  if (!g) { g = make(); wheelGeoms.set(key, g) }
   317→  return g
   318→}
   319→function wheelGeo(r: number, w: number) { return cachedCyl(`t${r}:${w}`, () => new THREE.CylinderGeometry(r, r, w, 10)) }
   320→function rimGeo(r: number, w: number) { return cachedCyl(`r${r}:${w}`, () => new THREE.CylinderGeometry(r * 0.52, r * 0.52, w + 0.03, 8)) }
   321→// v36 torus tire — rounded cross-section reads far more real than a plain cylinder
   322→const torusCache = new Map<string, THREE.TorusGeometry>()
   323→function torusGeo(r: number, w: number) {
   324→  let g = torusCache.get(`tt${r}:${w}`)
   325→  if (!g) {
   326→    const tube = w * 0.42
   327→    g = new THREE.TorusGeometry(r - tube, tube, 8, 18)
   328→    g.rotateY(Math.PI / 2) // axle → X (mesh rotation stays identity; spin = rotation.x)
   329→    torusCache.set(`tt${r}:${w}`, g)
   330→  }
   331→  return g
   332→}

   334→/** buildWheels — cylinder wheels rolled around X (rotation.z=π/2, spin = rotation.x), grouped per axle. */
   335→function buildWheels(
   336→  count: number,
   337→  positions: Array<[number, number]>, // [x, z]; front axle = largest z (steers when opts.steer)
   338→  radius: number,
   339→  opts: { width?: number; steer?: boolean; rims?: boolean | 'front'; torus?: boolean } = {},
   340→): { group: THREE.Group; wheels: VehicleWheel[] } {
   341→  const width = opts.width ?? radius * 0.68
   342→  const frontZ = Math.max(...positions.map(p => p[1]))
   343→  const group = new THREE.Group()
   344→  const axles = new Map<number, THREE.Group>()
   345→  const wheels: VehicleWheel[] = []
   346→  for (let i = 0; i < count; i++) {
   347→    const [wx, wz] = positions[i % positions.length]
   348→    let axle = axles.get(wz)
   349→    if (!axle) { axle = new THREE.Group(); axle.position.z = wz; group.add(axle); axles.set(wz, axle) }
   350→    const isFront = opts.steer === true && wz === frontZ
   351→    const pivot = new THREE.Group()
   352→    pivot.position.x = wx
   353→    axle.add(pivot)
   354→    const useTorus = opts.torus === true
   355→    const tire = new THREE.Mesh(useTorus ? torusGeo(radius, width) : wheelGeo(radius, width), tireMat)
   356→    if (!useTorus) tire.rotation.z = Math.PI / 2
   357→    tire.castShadow = false // perf: bodies only
   358→    pivot.add(tire)
   359→    if (opts.rims === true || (opts.rims === 'front' && isFront)) {
   360→      const rim = new THREE.Mesh(rimGeo(radius, width), rimMat)
   361→      rim.rotation.z = Math.PI / 2
   362→      rim.castShadow = false
   363→      tire.add(rim) // child of the tire → inherits spin + steer
   364→    }
   365→    wheels.push({ pivot, spin: tire, radius, steers: isFront })
   366→  }
   367→  return { group, wheels }
   368→}

   371→function box(w: number, h: number, d: number, x = 0, y = 0, z = 0, rx = 0): THREE.BoxGeometry {
   372→  const g = new THREE.BoxGeometry(w, h, d)
   373→  if (rx) g.rotateX(rx)

   377→function taperTop(g: THREE.BoxGeometry, sx: number, sz: number) {
   378→  const pos = g.attributes.position
   379→  for (let i = 0; i < pos.count; i++) if (pos.getY(i) > 0) { pos.setX(i, pos.getX(i) * sx); pos.setZ(i, pos.getZ(i) * sz) }
   380→  pos.needsUpdate = true
   381→  g.computeVertexNormals()
   382→  return g
   383→}
   384→function merged(parts: THREE.BufferGeometry[], mat: THREE.Material, castShadow = true): THREE.Mesh {
   385→  // mergeGeometries requires uniform indexedness — RoundedBoxGeometry (v36) is
   386→  // non-indexed, so weld any non-indexed part into an indexed geometry first
   387→  // (mergeVertices hashes positions and builds an index; no toIndexed in three).
   388→  const norm = parts.map(p => (p.index ? p : mergeVertices(p, 1e-4) ?? p))
   389→  const geo = mergeGeometries(norm, false)
   390→  const mesh = new THREE.Mesh(geo ?? parts[0], mat)
   391→  if (geo) for (const p of norm) p.dispose()
   392→  mesh.castShadow = castShadow
   393→  return mesh
   394→}

   396→const CAR_PALETTES = ['#d9a441', '#b5543a', '#93a67e', '#e8e2d4', '#6b4a3a'] // taxi-gold, dusty red, sage, cream, walnut (poster palette)

   399→/** buildCar v3 (v36) — real automotive construction via ExtrudeGeometry:
   400→ *  the three-box side profile (bumper faces, curved nose, bonnet, cowl dip,
   401→ *  rear deck, tail drop) is drawn as a Shape with WHEEL ARCHES cut into the
   402→ *  outline, then extruded across the body width with a bevel and welded
   403→ *  vertices → smooth rounded panel edges. Greenhouse is a separate blacked-out
   404→ *  glass extrusion (modern stylized A/B/C pillars) with a painted roof panel on
   405→ *  top. Clearcoat paint + env reflections finish it. Details kit kept from v35. */
   406→function buildCar(variant: number): { group: THREE.Group; wheels: VehicleWheel[] } {
   407→  const L = 3.9 + (variant % 3) * 0.25 // per-instance length variance
   408→  const paint = carPaint(CAR_PALETTES[variant % CAR_PALETTES.length])
   409→  const W = 1.58
   410→  const belt = 0.96 // bonnet / boot top line — the cabin sits clearly above it
   411→  // --- lower body: side-profile silhouette with real wheel arches ---
   412→  const archZ = L * 0.315
   413→  const archR = 0.42 // tire r=0.34 → tight 0.08 arch gap (real-car stance)
   414→  const archY = 0.32
   415→  const s = new THREE.Shape()
   416→  s.moveTo(-L / 2 + 0.10, archY)
   417→  s.lineTo(-archZ - archR, archY)

   482→/** v36 helper — extrude a side-profile Shape across the body width with a bevel,
   483→ *  weld coincident vertices and recompute smooth normals (UVs dropped: untextured). */
   484→function smoothExtrude(shape: THREE.Shape, depth: number, bevel: number): THREE.BufferGeometry {
   485→  let geo: THREE.BufferGeometry = new THREE.ExtrudeGeometry(shape, {
   486→    depth,
   487→    bevelEnabled: true,
   488→    bevelThickness: bevel,
   489→    bevelSize: bevel * 0.9,
   490→    bevelSegments: 3,
   491→    curveSegments: 10,
   492→  })
   493→  geo.deleteAttribute('uv')
   494→  const welded = mergeVertices(geo, 1e-4)
   495→  if (welded) { geo.dispose(); geo = welded }
   496→  geo.computeVertexNormals()
   497→  geo.rotateY(-Math.PI / 2) // profile length (shape x) → world +Z (front), extrusion → X
   498→  geo.computeBoundingBox()
   499→  const bb = geo.boundingBox
   500→  if (bb) geo.translate(-(bb.min.x + bb.max.x) / 2, 0, 0) // center across the width
   501→  return geo
   502→}

   721→export class VehicleSystem {
   722→  group = new THREE.Group()
   723→  vehicles: VehicleVisual[] = []
   724→  onPlayerHit: ((kind: string, x: number, z: number, dmg: number) => void) | null = null
   725→  private hitCooldown = 0
   726→
   727→  constructor(world: WorldDef, renderer?: THREE.WebGLRenderer) {
   728→    for (let i = 0; i < world.vehicles.length; i++) {
   729→      const v = world.vehicles[i]
   730→      const built = v.kind === 'bus' ? buildBus(i)
   731→        : v.kind === 'cow' ? buildCow()
   732→        : v.kind === 'auto' ? buildAuto()
   733→        : v.kind === 'bike' ? buildBike(i)
   734→        : buildCar(i) // 'car' kind (and safe fallback for any future kind)
   735→      built.group.name = `vehicle-${v.kind}-${v.id}`
   736→      this.group.add(built.group)
   737→      this.vehicles.push({ id: v.id, kind: v.kind, group: built.group, path: v.path.map(([px, pz]) => ({ x: px, z: pz })), speed: v.speed, wheels: built.wheels, prevHeading: NaN, steer: 0 })
   738→    }
   739→    // v36 — env reflections on vehicle materials only (world look untouched)
   740→    if (renderer) {
   741→      ensureVehicleEnv(renderer)
   742→      applyVehicleEnv(this.group)
   743→    }
   744→  }
   745→
   746→  /** Deterministic update from shared world time (seconds since match start) */
   747→  update(worldTime: number, dt: number, playerPos: THREE.Vector3 | null, localAuth: boolean) {
   748→    this.hitCooldown = Math.max(0, this.hitCooldown - dt)
   749→    const step = Math.max(dt, 1e-4)
   750→    for (const v of this.vehicles) {
   751→      const a = v.path[0], b = v.path[1]
   752→      const len = Math.hypot(b.x - a.x, b.z - a.z)
   753→      const period = (len / v.speed) * 2
   754→      const phase = ((worldTime % period) + period) % period
   755→      const half = period / 2
   756→      const t = phase < half ? phase / half : 2 - phase / half // ping-pong 0..1..0
   757→      const x = a.x + (b.x - a.x) * t
   758→      const z = a.z + (b.z - a.z) * t
   759→      const dir = phase < half ? 1 : -1
   760→      v.group.position.set(x, 0, z)
   761→      const heading = Math.atan2(b.x - a.x, b.z - a.z) * dir
   762→      // steer angle ≈ signed turn rate of heading, clamped ±0.5 rad (paths are straight → ~0, kicks at U-turns)
   763→      if (Number.isFinite(v.prevHeading)) {
   764→        let dh = heading - v.prevHeading
   765→        while (dh > Math.PI) dh -= Math.PI * 2
   766→        while (dh < -Math.PI) dh += Math.PI * 2
   767→        const turn = Math.max(-0.5, Math.min(0.5, dh / step))
   768→        v.steer += (turn - v.steer) * Math.min(1, step * 7)
   769→      }
   770→      v.prevHeading = heading
   771→      v.group.rotation.y = heading
   772→      // wheels spin by actual travel speed; front pivots take the steer angle
   773→      for (const w of v.wheels) {
   774→        w.spin.rotation.x += (v.speed * dt) / w.radius
   775→        if (w.steers) w.pivot.rotation.y = v.steer
   776→      }
   777→      // collision with local player (per-kind ring + damage, mirrored server-side)
   778→      if (playerPos && localAuth && this.hitCooldown <= 0) {
   779→        const hit = VEHICLE_HIT[v.kind] ?? { r: 1.7, dmg: 35 }
   780→        const d = Math.hypot(playerPos.x - x, playerPos.z - z)
   781→        if (d < hit.r) {
   782→          this.hitCooldown = 1.2
   783→          this.onPlayerHit?.(v.kind, x, z, hit.dmg)
   784→        }
   785→      }
   786→    }
   787→  }
   788→
   789→  /** (2-d contract) One heading-agnostic solid AABB per vehicle — movement workstream consumes these. */
   790→  getSolidAABBs(): AABB[] {
   791→    const pad = 0.05
   792→    const out: AABB[] = []
   793→    for (const v of this.vehicles) {
   794→      const e = VEHICLE_SOLID_EXTENTS[v.kind] ?? VEHICLE_SOLID_EXTENTS.car
   795→      const x = v.group.position.x
   796→      const z = v.group.position.z
   797→      out.push({ minX: x - e.hx - pad, minY: 0, minZ: z - e.hz - pad, maxX: x + e.hx + pad, maxY: e.maxY, maxZ: z + e.hz + pad })
   798→    }
   799→    return out
   800→  }
   801→}
