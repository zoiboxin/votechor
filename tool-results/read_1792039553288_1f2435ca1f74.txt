   562→   562→  private makeTextTexture(text: string): THREE.CanvasTexture {
   563→   563→    const canvas = document.createElement('canvas')
   564→   564→    canvas.width = 256; canvas.height = 148
   565→   565→    const ctx = canvas.getContext('2d')!
   566→   566→    ctx.fillStyle = '#f5efdc'
   567→   567→    ctx.fillRect(0, 0, 256, 148)
   568→   568→    ctx.strokeStyle = '#3a2d1a'; ctx.lineWidth = 6
   569→   569→    ctx.strokeRect(4, 4, 248, 140)
   570→   570→    ctx.fillStyle = '#8c1f28'
   571→   571→    ctx.font = 'bold 30px Arial'
   572→   572→    ctx.textAlign = 'center'
   573→   573→    const words = text.split(' ')
   574→   574→    const lines: string[] = []
   575→   575→    let cur = ''
   576→   576→    for (const w of words) {
   577→   577→      if ((cur + ' ' + w).trim().length > 14) { lines.push(cur.trim()); cur = w }
   578→   578→      else cur += ' ' + w
   579→   579→    }
   580→   580→    if (cur.trim()) lines.push(cur.trim())
   581→   581→    lines.forEach((l, i) => ctx.fillText(l, 128, 62 - (lines.length - 1) * 17 + i * 34))
   582→   582→    const tex = new THREE.CanvasTexture(canvas)
   583→   583→    tex.colorSpace = THREE.SRGBColorSpace
   584→   584→    return tex
   585→   585→  }

   642→   642→  update(dt: number, anim: AnimName, moveSpeed: number) {
   643→   643→    this.anim = anim
   644→   644→    if (this.forceAnim) anim = this.forceAnim
   645→   645→    this.speedRatio = Math.min(1, moveSpeed / 9)
   646→   646→    const p = (this.phase += dt * (anim === 'run' ? 11 : anim === 'walk' ? 7.2 : 1.6))
   647→   647→    const s = Math.sin(p), c = Math.cos(p)
   648→   648→    const swing = anim === 'run' ? 1.05 : anim === 'walk' ? 0.62 : 0
   649→   649→
   650→   650→    const set = (o: THREE.Object3D, x: number, z = 0) => { o.rotation.x += (x - o.rotation.x) * Math.min(1, dt * 14); if (z !== 0 || o.rotation.z !== 0) o.rotation.z += (0 - o.rotation.z) * Math.min(1, dt * 14) }
   651→   651→
   652→   652→    switch (anim) {
   653→   653→      case 'walk':
   654→   654→      case 'run': {
   655→   655→        const lean = anim === 'run' ? 0.22 : 0.08
   656→   656→        this.torsoPivot.rotation.x += (lean - this.torsoPivot.rotation.x) * Math.min(1, dt * 8)
   657→   657→        set(this.legL, s * swing)
   658→   658→        set(this.legR, -s * swing)
   659→   659→        if (!this.armBlockR) {
   660→   660→          set(this.armR, s * swing * 0.9)
   661→   661→        }
   662→   662→        set(this.armL, this.armBlockL ? 0.15 : -s * swing * 0.9)
   663→   663→        this.group.position.y = this.baseY + Math.abs(s) * (anim === 'run' ? 0.06 : 0.03)
   664→   664→        break
   665→   665→      }
   666→   666→      case 'jump': {
   667→   667→        this.torsoPivot.rotation.x += (0.1 - this.torsoPivot.rotation.x) * Math.min(1, dt * 8)
   668→   668→        set(this.legL, -0.5); set(this.legR, 0.25)
   669→   669→        set(this.armL, -2.6); set(this.armR, -2.6)
   670→   670→        // double-jump front-flip flourish (procedural, rides on top of the jump pose)
   671→   671→        if (this.flipTimer > 0) {
   672→   672→          this.flipTimer = Math.max(0, this.flipTimer - dt)
   673→   673→          const t = 1 - this.flipTimer / this.flipDur
   674→   674→          const eased = t < 0.7 ? (t / 0.7) * (Math.PI * 2) : Math.PI * 2 // flip then hold
   675→   675→          this.torsoPivot.rotation.x = eased
   676→   676→        }
   677→   677→        break
   678→   678→      }
   679→   679→      case 'slide': {
   680→   680→        this.torsoPivot.rotation.x += (-1.15 - this.torsoPivot.rotation.x) * Math.min(1, dt * 10)
   681→   681→        set(this.legL, 0.4); set(this.legR, -0.15)
   682→   682→        set(this.armL, -2.9); set(this.armR, -2.7)
   683→   683→        break
   684→   684→      }
   685→   685→      case 'dazed':
   686→   686→      case 'down': {
   687→   687→        this.torsoPivot.rotation.x += (-0.55 - this.torsoPivot.rotation.x) * Math.min(1, dt * 6)
   688→   688→        this.headPivot.rotation.z = Math.sin(p * 2) * 0.25
   689→   689→        set(this.legL, 0.35); set(this.legR, -0.3)
   690→   690→        set(this.armL, 0.6); set(this.armR, -0.6)
   691→   691→        break
   692→   692→      }
   693→   693→      case 'detained': {
   694→   694→        this.torsoPivot.rotation.x += (0.12 - this.torsoPivot.rotation.x) * Math.min(1, dt * 6)
   695→   695→        set(this.armL, 0.55, 0); set(this.armR, 0.55, 0)
   696→   696→        this.armL.rotation.z = 0.5; this.armR.rotation.z = -0.5
   697→   697→        this.headPivot.rotation.x = 0.35
   698→   698→        break
   699→   699→      }
   700→   700→      case 'celebrate': {
   701→   701→        this.torsoPivot.rotation.x += (0 - this.torsoPivot.rotation.x) * Math.min(1, dt * 8)
   702→   702→        set(this.armL, -2.9 + s * 0.3); set(this.armR, -2.9 - s * 0.3)
   703→   703→        this.group.position.y = this.baseY + Math.abs(c) * 0.14
   704→   704→        this.headPivot.rotation.x = 0
   705→   705→        break
   706→   706→      }
   707→   707→      case 'throw': {
   708→   708→        set(this.armR, -2.4); set(this.armL, 0.4)
   709→   709→        this.torsoPivot.rotation.x += (0.15 - this.torsoPivot.rotation.x) * dt * 10
   710→   710→        break
   711→   711→      }
   712→   712→      case 'heal': {
   713→   713→        set(this.armL, -1.7); set(this.armR, -1.7)
   714→   714→        this.headPivot.rotation.x = 0.1
   715→   715→        break
   716→   716→      }
   717→   717→      default: { // idle
   718→   718→        this.torsoPivot.rotation.x += (0.03 - this.torsoPivot.rotation.x) * Math.min(1, dt * 6)
   719→   719→        this.headPivot.rotation.x = Math.sin(p * 0.5) * 0.04
   720→   720→        set(this.legL, 0); set(this.legR, 0)
   721→   721→        if (!this.armBlockL) {
   722→   722→          set(this.armL, 0.04 + Math.sin(p * 0.8) * 0.03)
   723→   723→        }
   724→   724→        if (!this.armBlockR) {
   725→   725→          set(this.armR, 0.04 - Math.sin(p * 0.8) * 0.03)
   726→   726→        }
   727→   727→        this.group.position.y = this.baseY + Math.sin(p * 0.9) * 0.012
   728→   728→      }
   729→   729→    }
   730→   730→
   731→   731→    // cloth life: flag ripples, banner sways with the stride
   732→   732→    if (this.flagClothRef) {
   733→   733→      this.flagClothRef.rotation.y = Math.sin(p * 1.4) * 0.35 + 0.12
   734→   734→      this.flagClothRef.rotation.z = Math.sin(p * 0.9) * 0.06
   735→   735→    }
   736→   736→    if (this.bannerClothRef) {
   737→   737→      this.bannerClothRef.rotation.z = Math.sin(p * 0.8) * 0.05
   738→   738→      this.bannerClothRef.rotation.x = Math.sin(p * 1.1) * 0.04
   739→   739→    }
   740→   740→  }
   741→   741→
   742→   742→  setBaseY(y: number) {
   743→   743→    this.baseY = y
   744→   744→  }
   745→   745→}
