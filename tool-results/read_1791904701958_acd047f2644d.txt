   338→    this.carryingBallot = false

   340→    this.matchTime = 0
   341→    this.humorTimer = 24

   344→    this.squadAutoRejoin = false
   345→    this.rallyReadyAt = 0
   346→    useGameStore.getState().setSquad(null)
   347→
   348→    const profile = store.profile
   349→    this.archetype = profile?.archetype ?? 'student'
   350→    const name = profile?.name ?? 'Protester'
   351→    this.refreshUnlockedEmotes()
   352→
   353→    // reset world states
   354→    this.worldRenderer.setBossGateOpen(false)
   355→    this.bossClient.hideAll()
   356→    this.fx.clear()
   357→    this.playerBubbles.clear()
   358→    for (const it of this.items.items) { it.taken = false; it.group.visible = true; it.respawnAt = 0 }
   359→    for (const [id] of this.remotePlayers) this.removeRemote(id)
   360→    this.localBoss = createBossBrain(this.world.bossArena)
   361→    this.bossActiveLocal = false
   362→
   363→    // fresh player
   364→    if (this.player) { this.engine.scene.remove(this.player.rig.group) }
   365→    const cosmetics = profile?.cosmetics
   366→    this.player = new PlayerController(this.archetype, name, this.world.spawn, this.physics, {
   367→      hat: cosmetics?.hat as string | undefined,
   368→      glasses: cosmetics?.glasses,
   369→      signText: cosmetics?.signText,
   370→      cosmetics,
   371→    })
   372→    this.player.onFootstep = (running) => AudioManager.play(running ? 'footstepRun' : 'footstep')
   373→    this.player.onLand = (hard) => AudioManager.play(hard ? 'land' : 'footstep')
   374→    this.player.onJump = () => AudioManager.play('jump')
   375→    this.player.onSlide = () => AudioManager.play('slide')
   376→    this.player.onDazed = () => { AudioManager.play('hit'); this.store.notify(randomOf(RANDOM_NOTIFICATIONS), 'warning') }
   377→    this.engine.scene.add(this.player.rig.group)
   378→    this.camRig.yaw = 0 // camera behind player, looking down the road (-z)
   379→
   380→    // reset local npc brains
   381→    this.localNpcBrains = createNpcStates(this.world.npcs.map(n => ({ id: n.id, type: n.type, x: n.x, z: n.z, patrol: n.patrol.map(p => ({ ...p })) })))
   382→
   383→    // connect to authoritative server (both modes use it; fallback offline)
   384→    this.offline = true
   385→    store.patchHud({ connection: 'connecting' })
   386→    const connected = await this.net.connect()
   387→    if (connected) {
   388→      this.net.join({
   389→        name,
   390→        archetype: this.archetype,
   391→        roomCode: mode === 'multiplayer' ? roomCode : null,
   392→        cosmetics,
   393→      })
   394→      // wait briefly for join ack; ack sets offline=false
   395→      await new Promise<void>((resolve) => {
   396→        const check = setInterval(() => {
   397→          if (this.net.playerId || !this.net.connected) { clearInterval(check); resolve() }
   398→        }, 100)
   399→        setTimeout(() => { clearInterval(check); resolve() }, 4000)
   400→      })
   401→      this.offline = !this.net.playerId
   402→    }
   403→    if (this.offline) {
   404→      store.patchHud({ connection: 'offline', playersOnline: 1 })
   405→      store.notify('Offline mode — local patrols active', 'info')
   406→    } else {
   407→      store.patchHud({ connection: 'connected' })
   408→    }
   409→
   410→    AudioManager.setTheme('ambient')
   411→    AudioManager.setCrowdLevel(0.7)
   412→    store.setUIMode('playing')
   413→    this.input.setEnabled(true)
   414→    this.running = true
   415→    store.notify('The votes are missing. The protest is on. Everybody run!', 'info')
   416→    this.maybeOnboard()
   417→  }

   419→  /** One-time contextual onboarding hints for brand-new protesters. */
   420→  private maybeOnboard() {
   421→    try {
   422→      if (localStorage.getItem('votechor_onboarded')) return
   423→      const tips: [number, string, 'info' | 'squad' | 'warning' | 'boss'][] = [
   424→        [2500, 'Tip: follow the compass up top — gold markers are stolen vote tokens', 'info'],
   425→        [9500, 'Tip: press E near tokens & items to grab them · Q throws a distraction', 'squad'],
   426→        [16500, 'Tip: checkpoints save progress, heal you, and free detained squadmates', 'warning'],
   427→        [23500, 'Tip: 12 tokens unlock the boss gate in the Government District. Good luck!', 'boss'],
   428→      ]
   429→      for (const [delay, text, kind] of tips) {
   430→        this.onboardingTimers.push(setTimeout(() => {
   431→          if (this.running) useGameStore.getState().notify(text, kind)
   432→        }, delay))
   433→      }
   434→      localStorage.setItem('votechor_onboarded', '1')
   435→    } catch { /* private browsing — skip */ }
   436→  }
   437→
   438→  private clearOnboarding() {
   439→    for (const t of this.onboardingTimers) clearTimeout(t)
   440→    this.onboardingTimers = []
   441→  }

   443→  /** Refresh the unlocked-emote set (level from profile + medal codes from the API). */
   444→  private refreshUnlockedEmotes() {
   445→    const store = useGameStore.getState()
   446→    const level = store.profile?.level ?? 1
   447→    const seed = new Set<string>()
   448→    for (const e of EMOTE_CATALOG) if (e.unlock.type === 'free') seed.add(e.id)
   449→    this.unlockedEmotes = seed
   450→    try {
   451→      const raw = localStorage.getItem('votechor_profile')
   452→      const token = raw ? (JSON.parse(raw) as { token?: string }).token : null
   453→      fetch('/api/achievements', { headers: token ? { 'x-vc-token': token } : {} })
   454→        .then((r) => (r.ok ? r.json() : null))
   455→        .then((data: { unlocked?: string[] } | null) => {
   456→          if (!data?.unlocked) return
   457→          this.unlockedEmotes = new Set([...unlockedEmoteIds(level, data.unlocked)])
   458→        })
   459→        .catch(() => { /* offline — free emotes remain usable */ })
   460→    } catch { /* private browsing — free emotes remain usable */ }
   461→  }

   657→  /**
   658→   * Gift the most useful inventory item to the nearest live teammate (F key).
   659→   * Server validates distance/rate/type; a rejected gift is refunded via game:giftRefund,
   660→   * and a full-pocket receiver bounces the item back via game:giftBounce.
   661→   */
   662→  giftNearest() {
   663→    const store = useGameStore.getState()
   664→    if (!this.player || !this.running) return
   665→    if (this.offline) { store.notify('Gifting needs teammates — go multiplayer!', 'warning'); return }
   666→    if (this.player.isDetained || this.player.isDazed) { store.notify('Cannot gift while detained or dazed', 'warning'); return }
   667→    if (this.inventory.length === 0) { store.notify('Pockets are empty — nothing to gift!', 'info'); return }
   668→    // nearest live player within gift range (interpolated positions ≈ server positions)
   669→    let best: { id: string; name: string; d: number; health: number } | null = null
   670→    for (const [id, r] of this.remotePlayers) {
   671→      if (r.detained) continue
   672→      const d = Math.hypot(r.target.x - this.player.pos.x, r.target.z - this.player.pos.z)
   673→      if (d <= GIFT_RADIUS && (!best || d < best.d)) best = { id, name: r.name, d, health: r.health }
   674→    }
   675→    if (!best) { store.notify(`No teammate within ${GIFT_RADIUS}m — get closer to share supplies`, 'info'); return }
   676→    // pick the most useful item: firstaid when they are hurt, chai, other consumables, then throwables
   677→    const has = (t: string) => this.inventory.find((i) => i.type === t)
   678→    const pick =
   679→      (best.health < 55 ? has('firstaid') : undefined) ??
   680→      has('chai') ?? has('firstaid') ?? has('snack') ?? has('joke') ??
   681→      this.inventory.find((i) => GIFT_TYPES.includes(i.type))
   682→    if (!pick) { store.notify('That item cannot be gifted', 'info'); return }
   683→    const idx = this.inventory.findIndex((i) => i.uid === pick.uid)
   684→    this.inventory.splice(idx, 1)
   685→    this.net.gift(best.id, pick.type)
   686→    this.player.playAnim('heal', 0.35)
   687→    this.fx.popup(`🎁 GIFT → ${best.name}`, '#f2c14e', this.player.pos.x, 2.2, this.player.pos.z)
   688→    this.sayBubble('me', randomOf(GIFT_QUIPS))
   689→    AudioManager.play('pickup')
   690→    this.stats.gifts++
   691→    this.store.notify(`Gifted ${ITEM_CONFIG[pick.type as ItemType]?.name ?? pick.type} to ${best.name} — support +1`, 'squad')
   692→    this.syncHud()
   693→  }

   780→  private doThrow() {
   781→    if (!this.player || this.player.isDetained || this.player.isDazed) return
   782→    const idx = this.inventory.findIndex(i => ['bottle', 'sign', 'foam'].includes(i.type))

   786→    const camDir = new THREE.Vector3()
   787→    this.engine.camera.getWorldDirection(camDir)
   788→    const dir = { x: camDir.x, z: camDir.z }
   789→    const len = Math.hypot(dir.x, dir.z) || 1
   790→    const power = 11
   791→    const from = this.player.pos
   792→    const impact = (ix: number, _iy: number, iz: number) => {
   793→      AudioManager.play('impact')

   798→        this.net.noiseEvent(ix, iz)
   799→      }
   800→      // boss hit check
   801→      if (this.offline && this.localBoss && this.localBoss.phase === 3 && this.localBoss.dizzyTimer > 0) {
   802→        const d = Math.hypot(ix - this.localBoss.x, iz - this.localBoss.z)

   806→          this.fx.popup('BONK! -1 SHIELD', '#ff9f1c', ix, 3, iz)
   807→        }
   808→      }
   809→    }
   810→    this.items.throwItem(item.type, from, { x: dir.x / len, z: dir.z / len }, power, 'me', this.offline, impact)
   811→    if (!this.offline) this.net.throwItem(item.type, from.x, from.y, from.z, (dir.x / len) * power, 5.2, (dir.z / len) * power)
   812→    this.player.playAnim('throw', 0.35)
   813→    AudioManager.play('throw')
   814→    this.syncHud()
   815→  }
