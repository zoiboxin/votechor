     1→// VoteChor Seasons — persistent 100-protester worlds for the solo queue.
     2→//
     3→// Design (user request):
     4→//  • A solo player joins ONE season. Every season holds up to 100 occupants.
     5→//  • Every free slot is filled by a server-simulated bot protester, so a fresh
     6→//    season immediately feels like a living city (you + 99 bots).
     7→//  • When a NEW human joins a season, one bot steps aside (removed + announced)
     8→//    so the human takes its place — population stays at 100.
     9→//  • When ANY occupant leaves, a fresh bot walks in from the plaza to backfill.
    10→//  • When a season is "full" of humans (SEASON_MAX_HUMANS), matchmaking creates
    11→//    the next season instead.
    12→//  • Bots are realistic: they wander the streets on a nav graph, idle, chat
    13→//    (quick messages), emote, socialize with each other AND with humans, flee
    14→//    police, join protest marches to checkpoints, and pick up supplies.
    15→//    Police will still chase them (they are real NPC targets), the boss ignores
    16→//    them, and they get detained / released like everyone else.
    17→//
    18→// Network: bots are regular PlayerState entries inside the Room, so snapshots,
    19→// NPC brains, vehicles, checkpoints, detention and the client renderer all work
    20→// on them with zero special-casing. A per-recipient graded-AOI broadcaster keeps
    21→// the 12 Hz player snapshot cheap: NEAR (≤150 m) every tick at full fidelity,
    22→// MID (150–300 m) at 1/3 rate with 1-decimal payloads, FAR (>300 m) on the
    23→// rotating far-window sweep. Bot wire payloads are cached once per tick.

    70→function angleLerp(from: number, to: number, t: number): number {
    71→  let d = to - from
    72→  while (d > Math.PI) d -= Math.PI * 2
    73→  while (d < -Math.PI) d += Math.PI * 2
    74→  return from + d * Math.min(1, t)
    75→}

    77→// ---------- nav graph (built once from the shared deterministic world) ----------
    78→interface NavNode { x: number; z: number; links: number[] }

    80→const NAV_GRID = 10
    81→const NAV_PAD = 1.7 // keep this far from building walls
    82→const NAV_Z_MAX = -474 // bots never wander into the Government District / boss arena

    84→function pointBlocked(x: number, z: number, pad: number): boolean {
    85→  for (const c of WORLD.colliders) {
    86→    if (c.walkable) continue // low walkable platforms (stage) are fine
    87→    if (x >= c.minX - pad && x <= c.maxX + pad && z >= c.minZ - pad && z <= c.maxZ + pad) return true
    88→  }
    89→  return false
    90→}

    92→function clearPath(ax: number, az: number, bx: number, bz: number): boolean {
    93→  const d = dist2d(ax, az, bx, bz)
    94→  const steps = Math.max(1, Math.ceil(d / 2))
    95→  for (let i = 0; i <= steps; i++) {
    96→    const t = i / steps
    97→    if (pointBlocked(ax + (bx - ax) * t, az + (bz - az) * t, 0.8)) return false
    98→  }
    99→  return true
   100→}
   101→
   102→function buildNavGraph(): NavNode[] {
   103→  const nodes: NavNode[] = []
   104→  const index = new Map<string, number>()
   105→  for (let x = -45; x <= 45; x += NAV_GRID) {
   106→    for (let z = NAV_Z_MAX; z <= 6; z += NAV_GRID) {
   107→      if (pointBlocked(x, z, NAV_PAD)) continue
   108→      index.set(`${x}|${z}`, nodes.length)
   109→      nodes.push({ x, z, links: [] })
   110→    }
   111→  }
   112→  // 8-neighborhood edges with line-of-sight (no cutting through buildings)
   113→  for (let i = 0; i < nodes.length; i++) {
   114→    const n = nodes[i]
   115→    for (const [dx, dz] of [[NAV_GRID, 0], [0, NAV_GRID], [NAV_GRID, NAV_GRID], [NAV_GRID, -NAV_GRID]] as const) {
   116→      const j = index.get(`${n.x + dx}|${n.z + dz}`)
   117→      if (j === undefined) continue
   118→      const m = nodes[j]
   119→      if (clearPath(n.x, n.z, m.x, m.z)) { n.links.push(j); m.links.push(i) }
   120→    }
   121→  }
   122→  return nodes
   123→}

   127→function nearestNode(x: number, z: number): number {
   128→  let best = 0
   129→  let bestD = Infinity
   130→  for (let i = 0; i < NAV.length; i++) {
   131→    const d = (NAV[i].x - x) ** 2 + (NAV[i].z - z) ** 2
   132→    if (d < bestD) { bestD = d; best = i }
   133→  }
   134→  return best
   135→}

   181→// ---------- graded AOI bands (pure helpers — unit-testable without sockets) ----------
   182→
   183→/** NEAR band: dist² ≤ 150² → included EVERY tick with the round2 payload. */
   184→export const AOI_NEAR_SQ = BOT_AOI * BOT_AOI
   185→/** MID band: 150² < dist² ≤ 300² → included at 1/3 rate, 1-decimal payload. */
   186→export const AOI_MID = BOT_MID_AOI
   187→export const AOI_MID_SQ = BOT_MID_AOI * BOT_MID_AOI
   188→
   189→/** 1-decimal quantization (10 cm) for MID/FAR payloads — plenty at that range. */
   190→export function round1(v: number): number {
   191→  return Math.round(v * 10) / 10
   192→}
   193→
   194→/** Minimal shape the AOI selection needs — plain literals work in unit tests. */
   195→export interface AoiEntity { id: string; x: number; z: number }
   196→
   197→/** Per-tick context for selectAoiPayload (payload caches are built once per tick). */
   198→export interface AoiCtx {
   199→  tick: number // broadcast tick counter — drives the 3-slot MID rotation
   200→  farCursor: number // informational: far-window cursor (the window is already resolved)
   201→  windowIds: ReadonlySet<string> // bot ids refreshed by the FAR sweep this tick
   202→  near: readonly PlayerNetState[] // per-bot round2 payloads (index-aligned with `bots`)
   203→  mid: readonly PlayerNetState[] // per-bot round1 payloads (index-aligned with `bots`)
   204→  humans: readonly PlayerNetState[] // full-detail human entries, prepended to every payload
   205→}
   206→
   207→/**
   208→ * Per-recipient AOI selection (pure):
   209→ *  • NEAR (≤150 m): every tick, round2 payload.
   210→ *  • MID (150–300 m): 1/3 rate via `botIndex % 3 === tick % 3`, round1 payload (~4 Hz each).
   211→ *  • FAR (>300 m): ONLY via the rotating BOT_FAR_WINDOW sweep, round1 payload.
   212→ *  • Humans: always full detail (and never duplicated).
   213→ * Client prunes rigs unseen for 6 s — every band stays far inside that window.
   214→ */
   215→export function selectAoiPayload(bots: readonly AoiEntity[], human: AoiEntity, ctx: AoiCtx): PlayerNetState[] {
   216→  const out: PlayerNetState[] = ctx.humans.slice()
   217→  const midSlot = ctx.tick % 3
   218→  for (let i = 0; i < bots.length; i++) {
   219→    const b = bots[i]
   220→    const dx = b.x - human.x
   221→    const dz = b.z - human.z
   222→    const d2 = dx * dx + dz * dz
   223→    if (d2 <= AOI_NEAR_SQ) {
   224→      out.push(ctx.near[i])
   225→    } else if (d2 <= AOI_MID_SQ) {
   226→      if (i % 3 === midSlot) out.push(ctx.mid[i])
   227→    } else if (ctx.windowIds.has(b.id)) {
   228→      out.push(ctx.mid[i])
   229→    }
   230→  }
   231→  return out
   232→}
   233→
   234→/**
   235→ * Rotating far-window sweep (semantics unchanged): the ids of the BOT_FAR_WINDOW
   236→ * bots starting at `cursor`, plus the next cursor. Pure so tests can verify the
   237→ * ≈ every (bots / BOT_FAR_WINDOW)-tick refresh cadence without sockets.
   238→ */
   239→export function computeFarWindow(bots: readonly AoiEntity[], cursor: number): { ids: Set<string>; nextCursor: number } {
   240→  const ids = new Set<string>()
   241→  const n = bots.length
   242→  if (n === 0) return { ids, nextCursor: 0 }
   243→  for (let i = 0; i < Math.min(BOT_FAR_WINDOW, n); i++) ids.add(bots[(cursor + i) % n].id)
   244→  return { ids, nextCursor: (cursor + BOT_FAR_WINDOW) % n }
   245→}

   268→const EST_BYTES_PER_ENTRY = 90 // rough wire estimate per player entry — an ESTIMATE, not measured bytes
   269→export interface BroadcastStats {
   270→  ticksPerSec: number // broadcast passes / sec across all seasons (1 s sampler)
   271→  avgEntriesPerTick: number // EMA of payload entries summed over recipients
   272→  avgTickMs: number // EMA of full broadcast duration
   273→  estBytesPerSec: number // ESTIMATE: avgEntriesPerTick × ticksPerSec × 90
   274→}
   275→export const broadcastStats: BroadcastStats = {
   276→  ticksPerSec: 0,
   277→  avgEntriesPerTick: 0,
   278→  avgTickMs: 0,
   279→  estBytesPerSec: 0,
   280→}
   281→const tickWindow = { count: 0, at: Date.now() }
   282→
   283→/** Fold one completed broadcast pass (one season, all recipients) into the stats. */
   284→function recordBroadcastTick(entriesSent: number, durationMs: number): void {
   285→  const k = 0.15
   286→  broadcastStats.avgEntriesPerTick += (entriesSent - broadcastStats.avgEntriesPerTick) * k
   287→  broadcastStats.avgTickMs += (durationMs - broadcastStats.avgTickMs) * k
   288→  tickWindow.count++
   289→}
   290→
   291→// 1 Hz sampler: real ticks/sec + wire estimate (decays to 0 when idle). unref'd:
   292→// never keeps the process (or a test run) alive on its own.
   293→setInterval(() => {
   294→  const now = Date.now()
   295→  const dt = (now - tickWindow.at) / 1000
   296→  if (dt <= 0) return
   297→  broadcastStats.ticksPerSec = tickWindow.count / dt
   298→  broadcastStats.estBytesPerSec = Math.round(broadcastStats.avgEntriesPerTick * broadcastStats.ticksPerSec * EST_BYTES_PER_ENTRY)
   299→  tickWindow.count = 0
   300→  tickWindow.at = now
   301→}, 1000).unref?.()

   424→  private spawnPoint(mode: 'seed' | 'arrive'): { x: number; z: number } {

   451→      node: nearestNode(pos.x, pos.z),
   452→      wp: null,
   453→      goal: mode === 'arrive' ? { x: rand(-30, 30), z: rand(-60, -10) } : randomGoal(),
   454→      state: mode === 'arrive' ? 'arrive' : (chance(0.3) ? 'idle' : 'wander'),
   455→      stateUntil: Date.now() + (mode === 'arrive' ? 0 : rand(1000, 5000)),
   456→      speed: rand(2.3, 3.4),
   457→      partner: null,
   458→      marchCp: null,
   459→      fleeFrom: null,
   460→      emoteUntil: 0,
   461→      emoteAnim: 'idle',
   462→      nextEmoteAt: Date.now() + rand(8000, 50_000),
   463→      nextQuipAt: Date.now() + rand(20_000, 120_000),
   464→      nextPickupAt: Date.now() + rand(6000, 40_000),
   465→    }
   466→    this.bots.set(id, bot)
   467→    if (announce) this.io.to(this.room.id).emit('game:playerjoined', id, name)
   468→    return bot
   469→  }
   470→
   471→  /** Remove a bot (announced to the room so clients drop the rig + feed the event). */
   472→  private removeBot(bot: Bot, announce: boolean) {
   473→    this.bots.delete(bot.p.id)
   474→    this.usedNames.delete(bot.p.name)
   475→    this.room.removePlayer(bot.p.id)
   476→    if (announce) this.io.to(this.room.id).emit('game:playerleft', bot.p.id, bot.p.name)
   477→  }
   478→
   479→  /** A bot to sacrifice for an incoming human — prefer one NOT standing next to a human. */
   480→  private pickRemovableBot(): Bot | null {
   481→    const all = [...this.bots.values()]
   482→    if (all.length === 0) return null
   483→    const humans = this.room.humans()
   484→    const away = all.filter((b) => !humans.some((h) => dist2d(h.x, h.z, b.p.x, b.p.z) < 9))
   485→    return pick(away.length > 0 ? away : all)
   486→  }

   488→  // ---------- simulation ----------
   489→
   490→  private tick() {
   491→    if (this.disposed) return
   492→    const now = Date.now()
   493→    const dt = TICK_MS / 1000
   494→
   495→    for (const bot of this.bots.values()) this.stepBot(bot, dt, now)
   496→
   497→    // churn — the world breathes: someone leaves, someone new arrives
   498→    if (now >= this.churnAt && this.room.humansCount() > 0) {
   499→      const victim = this.pickRemovableBot()
   500→      if (victim) {
   501→        this.removeBot(victim, true)
   502→        const t = setTimeout(() => {
   503→          if (this.disposed) return
   504→          if (this.room.humansCount() + this.bots.size < SEASON_CAP) this.createBot('arrive', true)
   505→          this.emitSeason()
   506→        }, rand(2500, 7000))
   507→        t.unref?.()
   508→        this.churnTimers.push(t)
   509→      }
   510→      this.churnAt = now + rand(CHURN_MIN_MS, CHURN_MAX_MS)
   511→    }
   512→
   513→    // protest marches to the next checkpoint
   514→    if (now >= this.marchAt && this.room.humansCount() > 0) {
   515→      this.startMarch()
   516→      this.marchAt = now + rand(MARCH_MIN_MS, MARCH_MAX_MS)
   517→    }
   518→  }
   519→
   520→  private startMarch() {
   521→    const inactive = WORLD.checkpoints.filter((c) => !this.room.checkpoints.has(c.id))
   522→    if (inactive.length === 0) return
   523→    const cp = pick(inactive)
   524→    const crowd = [...this.bots.values()].filter((b) => b.state === 'wander' && !b.p.detained && !b.p.dazed)
   525→    if (crowd.length < 3) return
   526→    const n = Math.min(crowd.length, randInt(3, 8))
   527→    for (let i = 0; i < n; i++) {
   528→      const b = crowd.splice(randInt(0, crowd.length - 1), 1)[0]
   529→      b.state = 'march'
   530→      b.marchCp = cp.id
   531→      b.goal = { x: cp.x + rand(-5, 5), z: cp.z + rand(-5, 5) }
   532→      b.wp = null
   533→      b.speed = rand(3.8, 4.4)
   534→    }
   535→    // a march leader shouts the plan
   536→    const leader = this.bots.get(crowd[0]?.p.id ?? '')
   537→    if (leader) this.io.to(this.room.id).emit('game:quickmsg', leader.p.id, randInt(5, 8)) // "To the flyover!" / "Form a squad!" / "The gate needs votes!"
   538→  }

   540→  private stepBot(bot: Bot, dt: number, now: number) {
   541→    const p = bot.p
   542→    if (p.detained || p.dazed) return // capture/daze systems own the bot
   543→
   544→    // --- threat scan: police chasing me, or an officer breathing down my neck ---
   545→    if (bot.state !== 'flee') {
   546→      for (const n of this.room.npcs) {
   547→        if (n.type === 'searchlight') continue
   548→        const d = dist2d(n.x, n.z, p.x, p.z)
   549→        if ((n.state === 'chase' && n.targetId === p.id && d < 20) || d < 5.5) {
   550→          bot.state = 'flee'
   551→          bot.fleeFrom = { x: n.x, z: n.z }
   552→          bot.stateUntil = now + rand(2200, 3600)
   553→          bot.partner = null
   554→          bot.marchCp = null
   555→          break
   556→        }
   557→      }
   558→    }
   559→
   560→    // --- emote chatter (bots stand still and perform) ---
   561→    if (now < bot.emoteUntil) {
   562→      p.anim = bot.emoteAnim
   563→      p.lastStateTime = now
   564→      return
   565→    }
   566→    if (now >= bot.nextEmoteAt && bot.state !== 'flee' && bot.state !== 'arrive') {
   567→      const emote = pick(BOT_EMOTES)
   568→      bot.emoteUntil = now + rand(1500, 2200)
   569→      bot.emoteAnim = EMOTE_ANIM[emote] ?? 'celebrate'
   570→      p.anim = bot.emoteAnim
   571→      bot.nextEmoteAt = now + rand(26_000, 80_000)
   572→      this.io.to(this.room.id).emit('game:emote', p.id, emote)
   573→      return
   574→    }
   575→
   576→    // --- one-liners ---
   577→    if (now >= bot.nextQuipAt && bot.state !== 'flee') {
   578→      bot.nextQuipAt = now + rand(50_000, 170_000)
   579→      this.io.to(this.room.id).emit('game:quickmsg', p.id, randInt(0, 9))
   580→    }
   581→
   582→    // --- supply runs: grab a nearby non-token item ---
   583→    if (now >= bot.nextPickupAt && bot.state !== 'flee') {
   584→      bot.nextPickupAt = now + rand(30_000, 80_000)
   585→      for (const def of WORLD.items) {
   586→        if (def.type === 'token') continue
   587→        if (dist2d(def.x, def.z, p.x, p.z) > 2.5) continue
   588→        const st = this.room.itemStates.get(def.id)
   589→        if (st && !st.taken && this.room.pickup(p.id, def.id)) break
   590→      }
   591→    }
   592→
   593→    // --- state machine ---
   594→    switch (bot.state) {
   595→      case 'arrive': {
   596→        // walk from the spawn gate into the plaza, then blend into the crowd
   597→        if (this.moveToward(bot, bot.goal!.x, bot.goal!.z, dt) || dist2d(p.x, p.z, bot.goal!.x, bot.goal!.z) < 4) {
   598→          bot.state = 'wander'
   599→          bot.goal = randomGoal()
   600→          bot.wp = null
   601→        }
   602→        break
   603→      }
   604→      case 'wander': {
   605→        if (!bot.wp) { this.pickWaypoint(bot); break }
   606→        if (this.moveToward(bot, bot.wp.x, bot.wp.z, dt)) {
   607→          bot.wp = null
   608→          // arrived at the wider goal?
   609→          if (bot.goal && dist2d(p.x, p.z, bot.goal.x, bot.goal.z) < 7) {
   610→            bot.goal = randomGoal()
   611→            if (chance(0.5)) { this.enterIdle(bot, now); break }
   612→            if (chance(0.45)) { this.trySocialize(bot, now); break }
   613→          }
   614→        }
   615→        break
   616→      }
   617→      case 'idle': {
   618→        p.anim = 'idle'
   619→        p.ry += Math.sin(now / 900 + p.x) * dt * 0.6 // look around
   620→        if (now >= bot.stateUntil) {
   621→          if (chance(0.35)) this.trySocialize(bot, now)
   622→          else { bot.state = 'wander'; bot.goal = randomGoal(); bot.wp = null }
   623→        }
   624→        break
   625→      }
   626→      case 'social': {
   627→        const partner = partnerPos(this, bot)
   628→        if (!partner) { bot.state = 'wander'; bot.wp = null; bot.goal = randomGoal(); break }
   629→        const d = dist2d(p.x, p.z, partner.x, partner.z)
   630→        if (d > 4.5) {
   631→          this.moveToward(bot, partner.x, partner.z, dt)
   632→        } else {
   633→          p.anim = 'idle'
   634→          p.ry = angleLerp(p.ry, Math.atan2(partner.x - p.x, partner.z - p.z), dt * 6)
   635→        }
   636→        if (now >= bot.stateUntil) { bot.state = 'wander'; bot.partner = null; bot.wp = null; bot.goal = randomGoal() }
   637→        break
   638→      }
   639→      case 'flee': {
   640→        if (bot.fleeFrom) this.moveAway(bot, bot.fleeFrom.x, bot.fleeFrom.z, dt)
   641→        if (now >= bot.stateUntil) {
   642→          bot.state = 'wander'
   643→          bot.fleeFrom = null
   644→          bot.node = nearestNode(p.x, p.z)
   645→          bot.goal = randomGoal()
   646→          bot.wp = null
   647→          bot.speed = rand(2.3, 3.4)
   648→        }
   649→        break
   650→      }
   651→      case 'march': {
   652→        const cp = WORLD.checkpoints.find((c) => c.id === bot.marchCp)
   653→        if (!cp || this.room.checkpoints.has(cp.id)) {
   654→          // checkpoint reached (by anyone) — celebrate then disperse
   655→          if (chance(0.6)) { bot.emoteUntil = now + 2200; bot.emoteAnim = 'celebrate' }
   656→          bot.state = 'wander'
   657→          bot.marchCp = null
   658→          bot.goal = randomGoal()
   659→          bot.wp = null
   660→          break
   661→        }
   662→        if (this.moveToward(bot, bot.goal!.x, bot.goal!.z, dt)) {
   663→          if (chance(0.5)) { bot.emoteUntil = now + 2000; bot.emoteAnim = 'celebrate' }
   664→          bot.state = 'wander'
   665→          bot.marchCp = null
   666→          bot.goal = randomGoal()
   667→          bot.wp = null
   668→        }
   669→        break
   670→      }
   671→    }
   672→
   673→    p.lastX = p.x
   674→    p.lastZ = p.z
   675→    p.lastStateTime = now
   676→  }
   677→
   678→  private enterIdle(bot: Bot, now: number) {
   679→    bot.state = 'idle'
   680→    bot.stateUntil = now + rand(2000, 6000)
   681→    bot.wp = null
   682→  }

   684→  private trySocialize(bot: Bot, now: number) {
   685→    const p = bot.p
   686→    // prefer a nearby human (bots gravitate toward real protesters), else another bot
   687→    let target: { id: string; x: number; z: number } | null = null
   688→    for (const h of this.room.humans()) {
   689→      if (h.detained || h.id === p.id) continue
   690→      const d = dist2d(h.x, h.z, p.x, p.z)
   691→      if (d < 10 && d > 1 && chance(0.5)) { target = { id: h.id, x: h.x, z: h.z }; break }
   692→    }
   693→    if (!target) {
   694→      for (const b of this.bots.values()) {
   695→        if (b.p.id === p.id || b.p.detained || b.p.dazed || b.state === 'march' || b.state === 'flee') continue
   696→        const d = dist2d(b.p.x, b.p.z, p.x, p.z)
   697→        if (d < 9 && d > 1.2) { target = { id: b.p.id, x: b.p.x, z: b.p.z }; break }
   698→      }
   699→    }
   700→    if (!target) { this.enterIdle(bot, now); return }
   701→    bot.state = 'social'
   702→    bot.partner = target.id
   703→    bot.stateUntil = now + rand(4500, 10_000)
   704→    bot.wp = null
   705→    if (chance(0.5)) this.io.to(this.room.id).emit('game:quickmsg', p.id, randInt(0, 9))
   706→    // mutual: if the partner is a bot, it turns to face us too
   707→    const other = this.bots.get(target.id)
   708→    if (other && other.state !== 'flee' && other.state !== 'march') {
   709→      other.state = 'social'
   710→      other.partner = p.id
   711→      other.stateUntil = bot.stateUntil
   712→      if (chance(0.5)) this.io.to(this.room.id).emit('game:quickmsg', other.p.id, randInt(0, 9))
   713→    }
   714→  }

   716→  /** Pick the next nav-graph waypoint, biased toward the current goal. */
   717→  private pickWaypoint(bot: Bot) {
   718→    const node = NAV[bot.node]
   719→    if (!node || node.links.length === 0) { bot.node = nearestNode(bot.p.x, bot.p.z); return }
   720→    let next: number
   721→    if (bot.goal && chance(0.68)) {
   722→      let best = node.links[0]
   723→      let bestD = Infinity
   724→      for (const l of node.links) {
   725→        const d = (NAV[l].x - bot.goal.x) ** 2 + (NAV[l].z - bot.goal.z) ** 2
   726→        if (d < bestD) { bestD = d; best = l }
   727→      }
   728→      next = best
   729→    } else {
   730→      next = pick(node.links)
   731→    }
   732→    bot.node = next
   733→    const n = NAV[next]
   734→    // small lateral jitter so bots don't walk in perfect single file
   735→    const jx = n.x + rand(-2.2, 2.2)
   736→    const jz = n.z + rand(-2.2, 2.2)
   737→    bot.wp = pointBlocked(jx, jz, 1.0) ? { x: n.x, z: n.z } : { x: jx, z: jz }
   738→  }
   739→
   740→  /** Move toward a point; returns true on arrival. Handles anim + facing. */
   741→  private moveToward(bot: Bot, tx: number, tz: number, dt: number): boolean {
   742→    const p = bot.p
   743→    const dx = tx - p.x
   744→    const dz = tz - p.z
   745→    const d = Math.hypot(dx, dz)
   746→    if (d < 0.6) { p.anim = 'idle'; return true }
   747→    const step = bot.speed * dt
   748→    if (d <= step) { p.x = tx; p.z = tz; p.anim = 'idle'; return true }
   749→    p.x += (dx / d) * step
   750→    p.z += (dz / d) * step
   751→    p.ry = angleLerp(p.ry, Math.atan2(dx, dz), dt * 8)
   752→    p.anim = bot.speed > 4.6 ? 'run' : 'walk'
   753→    return false
   754→  }
   755→
   756→  /** Run away from a threat, steering around walls. */
   757→  private moveAway(bot: Bot, fx: number, fz: number, dt: number) {
   758→    const p = bot.p
   759→    const speed = 6.6
   760→    let dx = p.x - fx
   761→    let dz = p.z - fz
   762→    const d = Math.hypot(dx, dz) || 1
   763→    dx /= d; dz /= d
   764→    const step = speed * dt
   765→    // try the direct route, then swing up to ±100° if a wall is in the way
   766→    for (const ang of [0, 0.5, -0.5, 1.0, -1.0, 1.6, -1.6]) {
   767→      const ca = Math.cos(ang), sa = Math.sin(ang)
   768→      const nx = p.x + (dx * ca - dz * sa) * step
   769→      const nz = p.z + (dx * sa + dz * ca) * step
   770→      if (nx < -45 || nx > 45 || nz > 7 || nz < NAV_Z_MAX + 2) continue
   771→      if (!pointBlocked(nx, nz, 1.1)) {
   772→        p.x = nx
   773→        p.z = nz
   774→        p.ry = angleLerp(p.ry, Math.atan2(dx * ca - dz * sa, dx * sa + dz * ca), dt * 10)
   775→        p.anim = 'run'
   776→        return
   777→      }
   778→    }
   779→    p.anim = 'run' // cornered — keep running on the spot
   780→  }

   782→  // ---------- AOI broadcast (replaces Room's flat game:players blast) ----------
   783→
   784→  private broadcastPlayers() {
   785→    if (this.disposed) return
   786→    const started = performance.now()
   787→    const humans: PlayerState[] = this.room.humans()
   788→    if (humans.length === 0) return
   789→    const bots: PlayerState[] = []
   790→    for (const b of this.bots.values()) bots.push(b.p)
   791→    const n = bots.length
   792→    // Per-tick payload cache: each bot's wire state is built exactly ONCE per tick
   793→    // (was O(humans × bots) recomputation + allocation at 12 Hz). Near cache keeps
   794→    // the full round2 fidelity; mid cache quantizes to 1 decimal (10 cm) for the
   795→    // MID band and the FAR sweep.
   796→    const nearCache: PlayerNetState[] = new Array(n)
   797→    const midCache: PlayerNetState[] = new Array(n)
   798→    for (let i = 0; i < n; i++) {
   799→      nearCache[i] = playerNet(bots[i])
   800→      midCache[i] = botNet1(bots[i])
   801→    }
   802→    // rotating far-window sweep — cadence unchanged (≈ every n / BOT_FAR_WINDOW ticks)
   803→    const { ids: windowIds, nextCursor } = computeFarWindow(bots, this.farCursor)
   804→    this.farCursor = nextCursor
   805→    this.tickCount++
   806→    const ctx: AoiCtx = {
   807→      tick: this.tickCount,
   808→      farCursor: this.farCursor,
   809→      windowIds,
   810→      near: nearCache,
   811→      mid: midCache,
   812→      humans: humans.map(playerNet), // humans: always full detail in every payload
   813→    }
   814→    let entriesSent = 0
   815→    for (const h of humans) {
   816→      const payload = selectAoiPayload(bots, h, ctx)
   817→      entriesSent += payload.length
   818→      this.io.to(h.id).emit('game:players', payload)
   819→    }
   820→    // cheap EMAs on this season + fold into the module-level aggregate for /health
   821→    const dur = performance.now() - started
   822→    const gap = this.lastBroadcastAt > 0 ? started - this.lastBroadcastAt : TICK_MS
   823→    this.lastBroadcastAt = started
   824→    const k = 0.15
   825→    this.statRate += (1000 / Math.max(1, gap) - this.statRate) * k
   826→    this.statEntries += (entriesSent - this.statEntries) * k
   827→    this.statMs += (dur - this.statMs) * k
   828→    recordBroadcastTick(entriesSent, dur)
   829→  }
   830→}

   832→/** Partner position for a socializing bot (human or bot). */
   833→function partnerPos(season: Season, bot: Bot): { x: number; z: number } | null {
   834→  if (!bot.partner) return null
   835→  const other = season.room.getPlayer(bot.partner)
   836→  if (!other || other.detained || other.dazed) return null
   837→  return { x: other.x, z: other.z }
   838→}

   842→export class SeasonManager {
   843→  private seasons = new Map<string, Season>()
   844→  private counter = 0
   845→
   846→  constructor(private io: SocketIOServer, private rooms: Map<string, Room>) {}
   847→
   848→  /**
   849→   * Find the liveliest season with room for one more human, or create the next one.
   850→   * "Full" = SEASON_MAX_HUMANS real players — bots always backfill the rest.
   851→   */
   852→  assign(): Season {
   853→    let best: Season | null = null
   854→    for (const s of this.seasons.values()) {
   855→      if (s.room.humansCount() >= SEASON_MAX_HUMANS) continue
   856→      if (!best) { best = s; continue }
   857→      const bh = best.room.humansCount()
   858→      const sh = s.room.humansCount()
   859→      if (sh > bh || (sh === bh && s.createdAt > best.createdAt)) best = s
   860→    }
   861→    if (best) return best
   862→    this.counter++
   863→    const season = new Season(this.io, `sea_${this.counter}`, `SEASON #${this.counter}`, (id) => {
   864→      const s = this.seasons.get(id)
   865→      if (s) { s.dispose(); this.seasons.delete(id); this.rooms.delete(id) }
   866→    })
   867→    this.seasons.set(season.id, season)
   868→    this.rooms.set(season.id, season.room)
   869→    return season
   870→  }
   871→
   872→  get(roomId: string): Season | undefined {
   873→    return this.seasons.get(roomId)
   874→  }
   875→
   876→  /** Cheap live snapshot of every season — consumed by the /health endpoint. */
   877→  list(): SeasonInfo[] {
   878→    const out: SeasonInfo[] = []
   879→    for (const s of this.seasons.values()) out.push(s.info())
   880→    return out
   881→  }
   882→
   883→  /** Real (socket-backed) players across every season — for the global join cap. */
   884→  totalHumans(): number {
   885→    let n = 0
   886→    for (const s of this.seasons.values()) n += s.room.humansCount()
   887→    return n
   888→  }
   889→}
