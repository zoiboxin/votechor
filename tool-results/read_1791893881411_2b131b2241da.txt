   221→  private async startMatch(mode: 'solo' | 'multiplayer', roomCode: string | null) {
   222→    const store = useGameStore.getState()
   223→    store.setUIMode('loading')
   224→    this.mode = mode
   225→    store.setMatchMode(mode)
   226→    this.lastRoomCode = roomCode
   227→    this.reconnectNotified = false
   228→    this.clearOnboarding()
   229→    this.matchEnded = false
   230→    this.endHandled = false
   231→    this.tokens = 0
   232→    this.gateOpen = false
   233→    this.activatedCheckpoints.clear()
   234→    this.inventory = []
   235→    this.stats = { tokens: 0, checkpoints: 0, rescues: 0, revives: 0, itemsCollected: 0, detentions: 0, bossHits: 0, distractions: 0 }
   236→    this.carryingBallot = false
   237→    this.carryingBallotId = null
   238→    this.matchTime = 0
   239→    this.humorTimer = 24
   240→
   241→    const profile = store.profile
   242→    this.archetype = profile?.archetype ?? 'student'
   243→    const name = profile?.name ?? 'Protester'
   244→
   245→    // reset world states
   246→    this.worldRenderer.setBossGateOpen(false)
   247→    this.bossClient.hideAll()
   248→    this.fx.clear()
   249→    this.bossCinematic = { active: false, t: 0, fired: false }
   250→    for (const it of this.items.items) { it.taken = false; it.group.visible = true; it.respawnAt = 0 }
   251→    for (const [id] of this.remotePlayers) this.removeRemote(id)
   252→    this.localBoss = createBossBrain(this.world.bossArena)
   253→    this.bossActiveLocal = false
   254→
   255→    // fresh player
   256→    if (this.player) { this.engine.scene.remove(this.player.rig.group) }
   257→    const cosmetics = profile?.cosmetics
   258→    this.player = new PlayerController(this.archetype, name, this.world.spawn, this.physics, {
   259→      hat: cosmetics?.hat as string | undefined,
   260→      glasses: cosmetics?.glasses,
   261→      signText: cosmetics?.signText,
   262→      cosmetics,
   263→    })
   264→    this.player.onFootstep = (running) => AudioManager.play(running ? 'footstepRun' : 'footstep')
   265→    this.player.onLand = (hard) => AudioManager.play(hard ? 'land' : 'footstep')
   266→    this.player.onJump = () => AudioManager.play('jump')
   267→    this.player.onSlide = () => AudioManager.play('slide')
   268→    this.player.onDazed = () => { AudioManager.play('hit'); this.store.notify(randomOf(RANDOM_NOTIFICATIONS), 'warning') }
   269→    this.engine.scene.add(this.player.rig.group)
   270→    this.camRig.yaw = 0 // camera behind player, looking down the road (-z)
   271→
   272→    // reset local npc brains
   273→    this.localNpcBrains = createNpcStates(this.world.npcs.map(n => ({ id: n.id, type: n.type, x: n.x, z: n.z, patrol: n.patrol.map(p => ({ ...p })) })))
   274→
   275→    // connect to authoritative server (both modes use it; fallback offline)
   276→    this.offline = true
   277→    store.patchHud({ connection: 'connecting' })
   278→    const connected = await this.net.connect()
   279→    if (connected) {
   280→      this.net.join({
   281→        name,
   282→        archetype: this.archetype,
   283→        roomCode: mode === 'multiplayer' ? roomCode : null,
   284→        cosmetics,
   285→      })
   286→      // wait briefly for join ack; ack sets offline=false
   287→      await new Promise<void>((resolve) => {
   288→        const check = setInterval(() => {
   289→          if (this.net.playerId || !this.net.connected) { clearInterval(check); resolve() }
   290→        }, 100)
   291→        setTimeout(() => { clearInterval(check); resolve() }, 4000)
   292→      })
   293→      this.offline = !this.net.playerId
   294→    }
   295→    if (this.offline) {
   296→      store.patchHud({ connection: 'offline', playersOnline: 1 })
   297→      store.notify('Offline mode — local patrols active', 'info')
   298→    } else {
   299→      store.patchHud({ connection: 'connected' })
   300→    }
   301→
   302→    AudioManager.setTheme('ambient')
   303→    AudioManager.setCrowdLevel(0.7)
   304→    store.setUIMode('playing')
   305→    this.input.setEnabled(true)
   306→    this.running = true
   307→    store.notify('The votes are missing. The protest is on. Everybody run!', 'info')
   308→    this.maybeOnboard()
   309→  }

   312→  private maybeOnboard() {
   313→    try {
   314→      if (localStorage.getItem('votechor_onboarded')) return
   315→      const tips: [number, string, 'info' | 'squad' | 'warning' | 'boss'][] = [
   316→        [2500, 'Tip: follow the compass up top — gold markers are stolen vote tokens', 'info'],
   317→        [9500, 'Tip: press E near tokens & items to grab them · Q throws a distraction', 'squad'],
   318→        [16500, 'Tip: checkpoints save progress, heal you, and free detained squadmates', 'warning'],
   319→        [23500, 'Tip: 12 tokens unlock the boss gate in the Government District. Good luck!', 'boss'],
   320→      ]
   321→      for (const [delay, text, kind] of tips) {
   322→        this.onboardingTimers.push(setTimeout(() => {

   324→        }, delay))
   325→      }
   326→      localStorage.setItem('votechor_onboarded', '1')
   327→    } catch { /* private browsing — skip */ }
   328→  }

   350→  private onJoinAck(ack: JoinAck) {
   351→    if (!ack.ok) {
   352→      useGameStore.getState().setJoining(false, ack.error ?? 'Join failed')
   353→      useGameStore.getState().notify(ack.error ?? 'Join failed', 'warning')
   354→      return
   355→    }
   356→    this.offline = false
   357→    this.net.playerId = ack.playerId
   358→    this.serverMatchStart = ack.matchStartedAt
   359→    useGameStore.getState().setRoomCode(ack.roomCode)
   360→    useGameStore.getState().patchHud({ playersOnline: ack.players.length })
   361→    this.tokens = ack.tokensCollected
   362→    this.tokensNeeded = ack.tokensNeeded
   363→    for (const p of ack.players) {
   364→      if (p.id !== ack.playerId) this.addRemote(p)
   365→    }
   366→    // sync world state
   367→    if (ack.checkpoints) ack.checkpoints.forEach(cp => this.activatedCheckpoints.add(cp))
   368→    if (ack.boss && ack.boss.active) this.bossActiveLocal = true
   369→    useGameStore.getState().notify(`Joined room ${ack.roomCode}`, 'squad')
   370→  }

   372→  private onServerPlayers(players: import('./shared/types').PlayerNetState[]) {
   373→    if (!this.player) return
   374→    const seen = new Set<string>()
   375→    let online = 0
   376→    for (const p of players) {
   377→      online++
   378→      seen.add(p.id)
   379→      if (p.id === this.net.playerId) continue
   380→      let r = this.remotePlayers.get(p.id)
   381→      if (!r) { r = this.addRemote(p) }
   382→      r.target = { x: p.x, y: p.y, z: p.z, ry: p.ry }
   383→      r.anim = p.anim
   384→      r.health = p.health
   385→      r.detained = p.detained
   386→      r.rig.group.visible = true
   387→    }
   388→    for (const [id, r] of this.remotePlayers) {
   389→      if (!seen.has(id)) { this.removeRemote(id) }
   390→    }
   391→    useGameStore.getState().patchHud({ playersOnline: online })
   392→  }

   394→  private addRemote(p: import('./shared/types').PlayerNetState) {
   395→    const rig = new CharacterRig({ archetype: p.archetype, name: p.name })
   396→    rig.group.position.set(p.x, p.y, p.z)
   397→    this.engine.scene.add(rig.group)
   398→    const r = { rig, target: { x: p.x, y: p.y, z: p.z, ry: p.ry }, anim: p.anim as AnimName, name: p.name, health: p.health, detained: p.detained }
   399→    this.remotePlayers.set(p.id, r)
   400→    return r
   401→  }

   403→  private removeRemote(id: string) {
   404→    const r = this.remotePlayers.get(id)
   405→    if (r) { this.engine.scene.remove(r.rig.group); this.remotePlayers.delete(id) }
   406→  }

   408→  private onServerBoss(boss: BossNetState, ballots: { id: string; x: number; z: number; taken: boolean }[]) {
   409→    this.bossClient.applyState(boss, this.worldTime)
   410→    this.bossClient.setBallots(ballots, this.worldTime)
   411→    const store = useGameStore.getState()
   412→    if (boss.active) {
   413→      let progress = 0
   414→      let objective = ''
   415→      switch (boss.phase) {
   416→        case 0: objective = 'The Committee arrives...'; break
   417→        case 1: progress = boss.interruptPoints / 120; objective = 'Interrupt the speech — hold megaphone stations!'; break
   418→        case 2: progress = boss.documents / 240; objective = 'Sort documents at the desks!'; break
   419→        case 3: progress = boss.hits / 6; objective = boss.dizzyTimer > 0 ? 'BOSS DIZZY — Throw items at him! (Q)' : 'Dodge the giant stamp!'; break
   420→        case 4: progress = boss.teamworkProgress / 12; objective = 'Hold BOTH team stations!'; break
   421→        case 5: progress = boss.votes / boss.votesNeeded; objective = `Grab ballots → deposit at the box! (${boss.votes}/${boss.votesNeeded})`; break
   422→        case 6: objective = 'DEMOCRACY RESTORED!'; break
   423→      }
   424→      store.patchBoss({
   425→        active: true, phase: boss.phase, phaseName: BOSS_PHASE_NAMES[boss.phase] ?? '',
   426→        health: boss.health, maxHealth: boss.maxHealth, objectiveText: objective,
   427→        progress: Number.isFinite(progress) ? Math.min(1, Math.max(0, progress)) : 0,
   428→        votes: boss.votes, votesNeeded: boss.votesNeeded,
   429→        dizzy: boss.stateName === 'dizzy',
   430→      })
   431→    } else {
   432→      store.patchBoss({ active: false })
   433→    }
   434→  }

   436→  private onCapture(playerId: string, until: number, message: string) {
   437→    if (!this.player) return
   438→    if (playerId === this.net.playerId || this.offline) {
   439→      this.player.detain()
   440→      this.stats.detentions++
   441→      this.detainUntil = until
   442→      this.store.notify(message || randomOf(CAPTURE_MESSAGES), 'capture')
   443→      AudioManager.play('capture')
   444→      this.camRig.addShake(0.7)
   445→    } else {
   446→      const r = this.remotePlayers.get(playerId)
   447→      if (r) r.detained = true
   448→      this.store.notify(`Teammate detained: ${r?.name ?? 'someone'}`, 'squad')
   449→    }
   450→  }

   452→  private onRelease(playerId: string, x: number, z: number) {
   453→    if (!this.player) return
   454→    if (playerId === this.net.playerId || this.offline) {
   455→      this.player.release(x, z)
   456→      this.store.notify('Released! The paperwork was, tragically, in order.', 'info')
   457→      AudioManager.play('release')
   458→    } else {
   459→      const r = this.remotePlayers.get(playerId)

   503→    if (!this.player || this.player.isDetained || this.player.isDazed) return

   513→    const from = this.player.pos

   532→    this.items.throwItem(item.type, from, { x: dir.x / len, z: dir.z / len }, power, 'me', this.offline, impact)
   533→    if (!this.offline) this.net.throwItem(item.type, from.x, from.y, from.z, (dir.x / len) * power, 5.2, (dir.z / len) * power)

   621→  private fixedUpdate(dt: number) {
   622→    if (!this.running || !this.player) return
   623→    const store = useGameStore.getState()
   624→    if (store.uiMode !== 'playing') { return }
   625→    this.worldTime += dt
   626→    this.matchTime += dt
   627→
   628→    // input
   629→    const [ldx, ldy] = this.input.consumeLook()
   630→    const oneShot = this.pendingOneShot
   631→    this.pendingOneShot = null
   632→    const boosted = this.worldTime < this.player.speedBoostUntil
   633→    this.player.applyInput(dt, this.input.state, this.camRig.yaw, this.archetype, oneShot, boosted)
   634→
   635→    // world bounds safety
   636→    const b = this.world.bounds
   637→    this.player.pos.x = Math.max(b.minX, Math.min(b.maxX, this.player.pos.x))
   638→    this.player.pos.z = Math.max(b.minZ, Math.min(b.maxZ, this.player.pos.z))
   639→    if (this.player.pos.y < -5) { this.player.teleport(this.world.spawn.x, this.world.spawn.z) }
   640→
   641→    // boss gate: block player when closed
   642→    const gateZ = this.world.bossArena.gateZ
   643→    if (!this.gateOpen && this.player.pos.z < gateZ + 0.5 && this.player.pos.z > gateZ - 2 && Math.abs(this.player.pos.x) < 10.5) {
   644→      this.player.pos.z = gateZ + 0.55
   645→      if (this.player.pos.z < gateZ) this.player.pos.z = gateZ + 0.55
   646→    }
   647→
   648→    // detention countdown
   649→    if (this.player.isDetained) {
   650→      const remain = Math.max(0, (this.detainUntil - Date.now()) / 1000)
   651→      if (remain <= 0) {
   652→        const cp = this.activatedCheckpoints.size > 0 ? this.world.checkpoints[Math.min(this.activatedCheckpoints.size, this.world.checkpoints.length) - 1] : this.world.checkpoints[0]
   653→        const dp = cp.detention
   654→        if (this.offline) this.player.release(dp.x, dp.z)
   655→        // server path: server sends game:release
   656→      }
   657→    }
   658→
   659→    // checkpoints (offline: local activation; online: server detects from positions)
   660→    if (this.offline) {
   661→      for (const cp of this.world.checkpoints) {
   662→        if (this.activatedCheckpoints.has(cp.id)) continue
   663→        if (Math.hypot(this.player.pos.x - cp.x, this.player.pos.z - cp.z) < cp.radius) {
   664→          this.activatedCheckpoints.add(cp.id)
   665→          this.stats.checkpoints = this.activatedCheckpoints.size
   666→          AudioManager.play('checkpoint')
   667→          this.fx.popup('CHECKPOINT +60 XP', '#4e9b6e', this.player.pos.x, 2.1, this.player.pos.z)
   668→          this.store.notify(`Checkpoint ${cp.id} — ${cp.name} unlocked! (${this.activatedCheckpoints.size}/4)`, 'achievement')
   669→        }
   670→      }
   671→    }
   672→
   673→    // offline NPC sim + capture (8s spawn grace so players can orient)
   674→    if (this.offline) {
   675→      const pv: NpcPlayerView = {
   676→        id: 'me', x: this.player.pos.x, z: this.player.pos.z,
   677→        detained: this.player.isDetained, dazed: this.player.isDazed,
   678→        hidden: this.matchTime < 8 && Math.hypot(this.player.pos.x - this.world.spawn.x, this.player.pos.z - this.world.spawn.z) < 16,
   679→      }
   680→      this.npcs.simulateLocal(dt, pv, this.world.colliders, Date.now())
   681→      // searchlight rotation local
   682→      for (const sl of this.world.searchlights) {
   683→        this.localSearchlightAngles[sl.id] += sl.speed * dt
   684→      }
   685→    }
   686→
   687→    // vehicles (clock synced to server when online so damage matches visuals)
   688→    const vehicleTime = this.offline || !this.serverMatchStart ? this.matchTime : (Date.now() - this.serverMatchStart) / 1000
   689→    this.vehicles.update(vehicleTime, dt, this.player.pos, true)
   690→
   691→    // offline boss tick
   692→    if (this.offline && this.localBoss) {
   693→      this.localBossAccum += dt
   694→      const step = 1 / 10
   695→      let guard = 0
   696→      while (this.localBossAccum >= step && guard++ < 4) {
   697→        this.localBossAccum -= step
   698→        const pv: import('./shared/boss-brain').BossPlayerView = { id: 'me', x: this.player.pos.x, z: this.player.pos.z, detained: this.player.isDetained, dazed: this.player.isDazed }
   699→        const ev = tickBoss(this.localBoss, [pv], step, Date.now())
   700→        for (const d of ev.damage) this.applyDamage(d.amount, d.kind)
   701→        if (ev.speechShock) { AudioManager.play('shockwave'); this.camRig.addShake(0.6) }
   702→        if (ev.phaseDone >= 1) {
   703→          AudioManager.play('checkpoint')
   704→          if (ev.defeated) { this.doVictory() }
   705→          else if (ev.phaseDone === 5) AudioManager.play('gate')
   706→        }
   707→        // ballot proximity (grab)
   708→        if (this.localBoss.phase === 5 && !this.carryingBallot) {
   709→          for (const bl of this.localBoss.ballots) {
   710→            if (!bl.taken && Math.hypot(bl.x - this.player.pos.x, bl.z - this.player.pos.z) < 2) {
   711→              bl.taken = true
   712→              this.carryingBallot = true
   713→              this.carryingBallotId = bl.id
   714→              AudioManager.play('pickup')
   715→              break
   716→            }
   717→          }
   718→        }
   719→        if (this.localBoss.phase === 5 && this.carryingBallot && Math.hypot(this.player.pos.x - this.localBoss.x, this.player.pos.z - (this.localBoss.z + 2)) < 3.4) {
   720→          this.carryingBallot = false
   721→          this.carryingBallotId = null
   722→          this.localBoss.votes++
   723→          AudioManager.play('cash')
   724→        }
   725→        // render local boss state
   726→        const bossNet = this.localBossToNet(this.localBoss)
   727→        this.bossClient.applyState(bossNet, this.worldTime)
   728→        this.bossClient.setBallots(this.localBoss.ballots.map(bl => ({ ...bl })), this.worldTime)
   729→        this.updateBossHudLocal(this.localBoss)
   730→        if (this.localBoss.active) this.bossActiveLocal = true
   731→      }
   732→    }
   733→
   734→    // item pickup (proximity + E / auto for tokens)
   735→    const near = this.items.nearestAvailable(this.player.pos.x, this.player.pos.z, this.player.pos.y)
   736→    if (near) {
   737→      const isToken = near.def.type === 'token'
   738→      const inRange = true
   739→      if (inRange) {
   740→        const label = isToken ? `E — Grab ${ITEM_CONFIG.token.name}` : `E — Pick up ${ITEM_CONFIG[near.def.type].name}`
   741→        if (this.store.hud.interactPrompt !== label) this.store.patchHud({ interactPrompt: label })
   742→        if (this.input.state.interact) {
   743→          if (!this.interacting) {
   744→            this.interacting = true
   745→            this.pickupItem(near.def.id, near.def.type)
   746→          }
   747→        } else this.interacting = false
   748→      }
   749→    } else {
   750→      // revive / heal teammates / deposit ballots
   751→      let prompt: string | null = null
   752→      if (this.carryingBallot && this.bossActiveLocal) prompt = 'E — Deposit ballot'
   753→      if (this.input.state.interact && this.carryingBallot) {
   754→        if (this.offline && this.localBoss) {
   755→          if (Math.hypot(this.player.pos.x - this.localBoss.x, this.player.pos.z - (this.localBoss.z + 2)) < 3.4 && this.carryingBallot) {
   756→            this.carryingBallot = false; this.localBoss.votes++; AudioManager.play('cash')
   757→          }
   758→        } else this.net.bossDeposit(this.player.pos.x, this.player.pos.z)
   759→      }
   760→      if (this.store.hud.interactPrompt !== prompt) this.store.patchHud({ interactPrompt: prompt })
   761→    }
   762→
   763→    // network send 12Hz
   764→    this.stateSendAccum += dt
   765→    if (this.stateSendAccum >= 1 / 12) {
   766→      this.stateSendAccum = 0
   767→      if (!this.offline && this.net.connected && !this.matchEnded) {
   768→        const anim = this.player.rig.anim
   769→        this.net.sendState(this.player.pos.x, this.player.pos.y, this.player.pos.z, this.player.ry, anim, {
   770→          tokens: this.tokens,
   771→          carrying: this.carryingBallot,
   772→        })
   773→      }
   774→    }
   775→
   776→    // boss ballot proximity (server mode)
   777→    if (!this.offline && this.bossActiveLocal && this.carryingBallot) {
   778→      // deposit via E handled above; grab via proximity:
   779→    }
   780→    if (!this.offline && this.bossActiveLocal && !this.carryingBallot) {
   781→      // grabbing handled by server broadcast game:bossBallot? -> simplified: client sends grab on proximity
   782→    }
   783→
   784→    // humor timer
   785→    this.humorTimer -= dt
   786→    if (this.humorTimer <= 0) {
   787→      this.humorTimer = 26 + Math.random() * 20
   788→      this.store.notify(randomOf(RANDOM_NOTIFICATIONS), 'info')
   789→    }
   790→
   791→    // match time limit
   792→    if (this.matchTime > MATCH_TIME_LIMIT && !this.matchEnded) {
   793→      this.doTimeUp()
   794→    }
   795→
   796→    // HUD sync at ~8Hz
   797→    this.hudAccum += dt
   798→    if (this.hudAccum >= 0.125) {
   799→      this.hudAccum = 0
   800→      this.syncHud()
   801→      this.drawMinimap()
   802→    }
   803→  }

   871→  private updateBossHudLocal(b: BossBrain) {
   872→    const store = useGameStore.getState()
   873→    if (!b.active) { store.patchBoss({ active: false }); return }
   874→    let progress = 0, objective = ''
   875→    switch (b.phase) {
   876→      case 0: objective = 'The Committee arrives...'; break
   877→      case 1: progress = b.interruptPoints / 120; objective = 'Interrupt the speech — hold megaphone stations! (E not needed, just stand)'; break
   878→      case 2: progress = b.documents / 240; objective = 'Sort documents at the desks!'; break
   879→      case 3: progress = b.hits / 6; objective = b.dizzyTimer > 0 ? 'BOSS DIZZY — Throw items at him! (Q)' : 'Dodge the giant stamp!'; break
   880→      case 4: progress = b.teamworkProgress / 12; objective = 'Hold the team stations!'; break
   881→      case 5: progress = b.votes / b.votesNeeded; objective = `Grab ballots → deposit at the box! (${b.votes}/${b.votesNeeded})`; break
   882→      case 6: objective = 'DEMOCRACY RESTORED!'; break
   883→    }
   884→    store.patchBoss({
   885→      active: true, phase: b.phase, phaseName: BOSS_PHASE_NAMES[b.phase] ?? '',
   886→      health: b.health, maxHealth: b.maxHealth, objectiveText: objective,
   887→      progress: Math.min(1, Math.max(0, progress)), votes: b.votes, votesNeeded: b.votesNeeded,
   888→      dizzy: b.stateName === 'dizzy',

   910→      while (d > Math.PI) d -= Math.PI * 2

   925→    const focus = this.player.pos
   926→    if (!this.bossCinematic.active) {

   958→  /** 4s cinematic orbit when the boss activates; freezes input. */
   959→  private updateBossCinematic(dt: number) {
   960→    const started = (this.offline && this.localBoss?.started) || (!this.offline && this.bossClient.visible)
   961→    if (started && !this.bossCinematic.fired) {
   962→      this.bossCinematic = { active: true, t: 0, fired: true }
   963→      this.player?.teleport(this.player.pos.x, this.player.pos.z) // stop momentum
   964→      if (this.player) this.player.state = 'frozen'
   965→      useGameStore.getState().patchHud({ cinematic: true })
   966→      AudioManager.setTheme('boss')
   967→      AudioManager.play('gate')
   968→    }
   969→    if (!this.bossCinematic.active) return
   970→    this.bossCinematic.t += dt
   971→    const t = this.bossCinematic.t
   972→    const DUR = 4.2
   973→    const b = this.world.bossArena
   974→    const bossZ = b.z - 14
   975→    // orbit from the gate side toward the side view, slightly rising
   976→    const ang = Math.PI * 0.5 + Math.sin((t / DUR) * Math.PI * 0.5) * 0.9
   977→    const r = 16 - (t / DUR) * 3
   978→    const cx = Math.sin(ang) * r
   979→    const cz = bossZ + Math.cos(ang) * r
   980→    const cy = 4.5 + (t / DUR) * 2.5
   981→    this.engine.camera.position.set(cx, cy, cz)
   982→    this.engine.camera.lookAt(b.x, 6.5, bossZ)
   983→    if (t >= DUR || useGameStore.getState().uiMode !== 'playing') {
   984→      this.bossCinematic.active = false
   985→      if (this.player && this.player.state === 'frozen') this.player.state = 'normal'
   986→      useGameStore.getState().patchHud({ cinematic: false })
   987→    }

   991→  private syncHud() {
   992→    if (!this.player) return
   993→    const p = this.player
   994→    const zone = this.world.zones.find(z => p.pos.z <= z.zStart && p.pos.z > z.zEnd)
   995→    const store = useGameStore.getState()
   996→    let objective = 'Recover the stolen vote tokens!'
   997→    if (this.tokens >= this.tokensNeeded) objective = 'Enter the Government District — defeat the Vote Collector!'
   998→    else if (this.activatedCheckpoints.size >= 4) objective = `Collect vote tokens (${this.tokens}/${this.tokensNeeded}) to open the boss gate`
   999→    const squad = [...this.remotePlayers.entries()].map(([id, r]) => ({
  1000→      id, name: r.name, health: r.health, detained: r.detained, dazed: r.anim === 'dazed', archetype: '', distance: Math.hypot(r.target.x - p.pos.x, r.target.z - p.pos.z),
  1001→    }))
  1002→    // compass markers (relative to camera yaw so up = where you look)
  1003→    const yawDeg = ((this.camRig.yaw * 180) / Math.PI + 360) % 360
  1004→    const rel = (x: number, z: number) => {
  1005→      let a = Math.atan2(x - p.pos.x, -(z - p.pos.z)) * (180 / Math.PI) - yawDeg
  1006→      a = ((a + 540) % 360) - 180
  1007→      return { deg: a, dist: Math.round(Math.hypot(x - p.pos.x, z - p.pos.z)) }
  1008→    }
  1009→    const markers: import('@/lib/game-store').CompassMarker[] = []
  1010→    // nearest untaken token within 90m
  1011→    let bestTok: { d: number; x: number; z: number } | null = null
  1012→    for (const it of this.items.items) {
  1013→      if (it.taken || it.def.type !== 'token') continue
  1014→      const d = Math.hypot(it.def.x - p.pos.x, it.def.z - p.pos.z)
  1015→      if (d < 90 && (!bestTok || d < bestTok.d)) bestTok = { d, x: it.def.x, z: it.def.z }
  1016→    }
  1017→    if (bestTok) markers.push({ kind: 'token', ...rel(bestTok.x, bestTok.z) })
  1018→    // next unactivated checkpoint
  1019→    const nextCp = this.world.checkpoints.find(cp => !this.activatedCheckpoints.has(cp.id))
  1020→    if (nextCp) markers.push({ kind: 'checkpoint', ...rel(nextCp.x, nextCp.z) })
  1021→    // boss arena once gate open
  1022→    if (this.gateOpen) markers.push({ kind: 'boss', ...rel(this.world.bossArena.x, this.world.bossArena.z) })
  1023→    // squad
  1024→    for (const s of squad) {
  1025→      const r = [...this.remotePlayers.entries()].find(([id]) => id === s.id)?.[1]
  1026→      if (r) markers.push({ kind: 'squad', ...rel(r.target.x, r.target.z) })
  1027→    }
  1028→    store.patchHud({
  1029→      health: Math.round(p.health), maxHealth: p.maxHealth, stamina: Math.round(p.stamina),
  1030→      zoneName: zone?.name ?? 'Unknown', zoneHint: zone?.hint ?? '',
  1031→      objective, checkpoint: this.activatedCheckpoints.size, totalCheckpoints: this.world.checkpoints.length,
  1032→      tokens: this.tokens, tokensNeeded: this.tokensNeeded,
  1033→      inventory: this.inventory.map(i => ({ uid: i.uid, type: i.type })), inventoryMax: INVENTORY_MAX,
  1034→      detained: p.isDetained, detainRemain: p.isDetained ? Math.max(0, (this.detainUntil - Date.now()) / 1000) : 0,
  1035→      dazed: p.isDazed,
  1036→      squad, matchTime: this.matchTime, distance: Math.round(p.distance),
  1037→      speedBoostUntil: p.speedBoostUntil, carryingBallot: this.carryingBallot,
  1038→      score: this.stats.tokens * 45 + this.stats.checkpoints * 60,
  1039→      compass: { yawDeg, markers },
  1040→    })
  1041→  }

  1045→  private drawMinimap() {
  1046→    const canvas = this.minimapCanvas
  1047→    if (!canvas || !this.player) return
  1048→    const ctx = canvas.getContext('2d')
  1049→    if (!ctx) return
  1050→    const W = canvas.width, H = canvas.height
  1051→    ctx.clearRect(0, 0, W, H)
  1052→    // world: z from 10 to -572, x -48..48 → vertical strip
  1053→    const toX = (x: number) => ((x + 48) / 96) * W
  1054→    const toY = (z: number) => ((10 - z) / 582) * H
  1055→    // zones
  1056→    const zoneColors = ['rgba(201,111,74,0.25)', 'rgba(184,92,62,0.25)', 'rgba(127,154,140,0.25)', 'rgba(201,168,106,0.25)', 'rgba(154,123,95,0.3)']
  1057→    this.world.zones.forEach((z, i) => {
  1058→      ctx.fillStyle = zoneColors[i]
  1059→      ctx.fillRect(0, toY(z.zStart), W, toY(z.zEnd) - toY(z.zStart))
  1060→    })
  1061→    // road
  1062→    ctx.fillStyle = 'rgba(80,80,80,0.5)'
  1063→    ctx.fillRect(toX(-8), 0, toX(8) - toX(-8), H)
  1064→    // checkpoints
  1065→    for (const cp of this.world.checkpoints) {
  1066→      ctx.fillStyle = this.activatedCheckpoints.has(cp.id) ? '#4e9b6e' : '#c9482e'
  1067→      ctx.beginPath(); ctx.arc(toX(cp.x), toY(cp.z), 3.4, 0, 7); ctx.fill()
  1068→    }
  1069→    // boss
  1070→    ctx.fillStyle = this.gateOpen ? '#8c1f28' : 'rgba(140,31,40,0.4)'
  1071→    ctx.beginPath(); ctx.arc(toX(this.world.bossArena.x), toY(this.world.bossArena.z), 5, 0, 7); ctx.fill()
  1072→    // tokens
  1073→    for (const t of this.items.tokenData()) {
  1074→      if (t.taken) continue
  1075→      ctx.fillStyle = '#f2c14e'
  1076→      ctx.fillRect(toX(t.x) - 1.5, toY(t.z) - 1.5, 3, 3)
  1077→    }
  1078→    // npcs (journalist sees far, others 40m)
  1079→    const seeDist = this.archetype === 'journalist' ? 70 : 42
  1080→    for (const n of this.npcs.minimapData()) {
  1081→      if (Math.hypot(n.x - this.player.pos.x, n.z - this.player.pos.z) > seeDist) continue
  1082→      ctx.fillStyle = n.chasing ? '#ff3b30' : '#3a6ea5'
  1083→      ctx.beginPath(); ctx.arc(toX(n.x), toY(n.z), 2.2, 0, 7); ctx.fill()
  1084→    }
  1085→    // squad
  1086→    for (const [, r] of this.remotePlayers) {
  1087→      ctx.fillStyle = '#4ecdc4'
  1088→      ctx.beginPath(); ctx.arc(toX(r.target.x), toY(r.target.z), 2.6, 0, 7); ctx.fill()
  1089→    }
  1090→    // player arrow
  1091→    const px = toX(this.player.pos.x), py = toY(this.player.pos.z)
  1092→    ctx.save()
  1093→    ctx.translate(px, py)
  1094→    ctx.rotate(-this.player.ry + Math.PI)
  1095→    ctx.fillStyle = '#ffffff'
  1096→    ctx.strokeStyle = '#1c1a17'
  1097→    ctx.lineWidth = 1.2
  1098→    ctx.beginPath()
  1099→    ctx.moveTo(0, -5); ctx.lineTo(3.6, 4); ctx.lineTo(0, 2); ctx.lineTo(-3.6, 4)
  1100→    ctx.closePath(); ctx.fill(); ctx.stroke()
  1101→    ctx.restore()
  1102→  }

  1134→  private async finalizeMatch(victory: boolean, serverPayload?: import('./shared/types').MatchEndPayload) {
  1135→    const p = this.player
  1136→    if (!p) return
  1137→    const store = useGameStore.getState()
  1138→    this.input.setEnabled(false)
  1139→    this.running = false
  1140→    const detentions = this.stats.detentions
  1141→    const stats = {
  1142→      victory, position: serverPayload?.position ?? 1,
  1143→      tokens: serverPayload?.tokens ?? this.stats.tokens,
  1144→      distance: serverPayload?.distance ?? Math.round(p.distance),
  1145→      checkpoints: serverPayload?.checkpoints ?? this.stats.checkpoints,
  1146→      rescues: serverPayload?.rescues ?? this.stats.rescues,
  1147→      revives: serverPayload?.revives ?? this.stats.revives,
  1148→      itemsCollected: serverPayload?.itemsCollected ?? this.stats.itemsCollected,
  1149→      bossContribution: serverPayload?.bossContribution ?? this.stats.bossHits,
  1150→      detentions,
  1151→    }
  1152→    // XP calculation (client display)
  1153→    const xp = Math.round(
  1154→      XP_TABLE.participation + stats.distance * XP_TABLE.perMeter +
  1155→      stats.checkpoints * XP_TABLE.perCheckpoint + stats.tokens * XP_TABLE.perToken +
  1156→      stats.rescues * XP_TABLE.perRescue + stats.revives * XP_TABLE.perRevive +
  1157→      stats.itemsCollected * XP_TABLE.perItem + stats.bossContribution * XP_TABLE.bossHit +
  1158→      (victory ? XP_TABLE.victoryBonus : 0),
  1159→    )
  1160→    const { VICTORY_TITLES } = await import('./shared/constants')
  1161→    const title = VICTORY_TITLES({ detentions, rescues: stats.rescues, revives: stats.revives, items: stats.itemsCollected, distance: stats.distance, victory })
  1162→    // persist
  1163→    let newAchievements: { id: string; name: string }[] = []
  1164→    let leveledUp = false
  1165→    let newLevel = store.profile?.level ?? 1
  1166→    let xpEarned = xp
  1167→    let dailyBonus = 0
  1168→    let dailyDesc: string | undefined
  1169→    let dailyJustCompleted = false
  1170→    const profileRaw = typeof window !== 'undefined' ? localStorage.getItem('votechor_profile') : null
  1171→    const token = profileRaw ? (JSON.parse(profileRaw) as { token: string }).token : null
  1172→    if (token) {
  1173→      try {
  1174→        const res = await fetch('/api/matches', {
  1175→          method: 'POST',
  1176→          headers: { 'Content-Type': 'application/json', 'x-vc-token': token },
  1177→          body: JSON.stringify({
  1178→            matchId: serverPayload?.matchId ?? `offline_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
  1179→            victory, distance: stats.distance, checkpoints: stats.checkpoints,
  1180→            rescues: stats.rescues, revives: stats.revives, itemsCollected: stats.itemsCollected,
  1181→            bossContribution: stats.bossContribution, detentions, tokens: stats.tokens,
  1182→            position: stats.position, multiplayer: !this.offline,
  1183→          }),
  1184→        })
  1185→        if (res.ok) {
  1186→          const data = await res.json() as {
  1187→            xpEarned: number; level: number; leveledUp: boolean
  1188→            newAchievements: { id: string; name: string }[]
  1189→            profile: { xp: number; level: number }
  1190→            dailyBonus?: number
  1191→            daily?: { kind: string; desc: string; target: number; reward: number; progress: number; claimed: boolean; justCompleted: boolean } | null
  1192→          }
  1193→          xpEarned = data.xpEarned
  1194→          newAchievements = data.newAchievements ?? []
  1195→          leveledUp = data.leveledUp
  1196→          newLevel = data.level
  1197→          if (data.daily) {
  1198→            dailyBonus = data.dailyBonus ?? 0
  1199→            dailyDesc = data.daily.desc
  1200→            dailyJustCompleted = !!data.daily.justCompleted
  1201→            useGameStore.getState().setDaily({ kind: data.daily.kind, desc: data.daily.desc, target: data.daily.target, reward: data.daily.reward, progress: data.daily.progress, claimed: data.daily.claimed })
  1202→          }
  1203→          if (data.profile && store.profile) {
  1204→            const updated = { ...store.profile, xp: data.profile.xp, level: data.profile.level }
  1205→            store.setProfile(updated)
  1206→            // keep localStorage in sync so the menu shows fresh progress after reload
  1207→            try {
  1208→              const raw = localStorage.getItem('votechor_profile')
  1209→              if (raw) {
  1210→                const parsed = JSON.parse(raw) as { token: string; profile: unknown }
  1211→                localStorage.setItem('votechor_profile', JSON.stringify({ token: parsed.token, profile: updated }))
  1212→              }
  1213→            } catch { /* non-fatal */ }
  1214→          }
  1215→          for (const a of newAchievements) store.notify(`Achievement: ${a.name}!`, 'achievement')
  1216→          if (dailyJustCompleted) store.notify(`DAILY CHALLENGE COMPLETE! +${dailyBonus} bonus XP!`, 'achievement')
  1217→          if (leveledUp) { store.notify(`LEVEL UP — Level ${newLevel}!`, 'achievement'); AudioManager.play('victory') }
  1218→        }
  1219→      } catch { /* offline persistence failure is fine */ }
  1220→    }
  1221→    const summary: MatchSummary = { ...stats, xpEarned, funnyTitle: title, newAchievements, leveledUp, newLevel, matchId: serverPayload?.matchId ?? 'offline', duration: serverPayload?.duration ?? Math.round(this.matchTime), playersOnline: serverPayload?.playersOnline ?? 1, dailyBonus, dailyDesc, dailyJustCompleted }
  1222→    store.setMatchSummary(summary)
  1223→    store.setUIMode('ended')
  1224→    AudioManager.setTheme('menu')
  1225→    AudioManager.setCrowdLevel(0.2)
  1226→  }
