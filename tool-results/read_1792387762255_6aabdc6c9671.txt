   292→    uiActions().setHudMinimap(minimapRef.current)

   296→  // mouse-wheel over the radar adjusts the minimap zoom (clamped + persisted via the

   298→  useEffect(() => {
   299→    const el = minimapBoxRef.current
   300→    if (!el) return
   301→    const onWheel = (e: WheelEvent) => {
   302→      e.preventDefault()
   303→      e.stopPropagation()
   304→      const st = useGameStore.getState()
   305→      const cur = st.settings.minimapZoom ?? 150
   306→      const next = Math.max(MINIMAP_MIN_ZOOM, Math.min(MINIMAP_MAX_ZOOM, Math.round((cur + (e.deltaY > 0 ? 12 : -12)) / 2) * 2))
   307→      if (next === cur) return
   308→      st.setSettings({ minimapZoom: next })
   309→      try {
   310→        localStorage.setItem('votechor_settings', JSON.stringify({ ...st.settings, minimapZoom: next, keybinds: { ...st.settings.keybinds } }))
   311→      } catch {
   312→        /* private browsing — lives for this session only */
   313→      }
   314→    }
   315→    el.addEventListener('wheel', onWheel, { passive: false })
   316→    return () => el.removeEventListener('wheel', onWheel)
   317→  }, [])
   318→
   319→  // zone banner fades out after a change
   320→  useEffect(() => {
   321→    setZoneVisible(true)
   322→    const t = setTimeout(() => setZoneVisible(false), 2800)
   323→    return () => clearTimeout(t)
   324→  }, [hud.zoneName])
   325→
   326→  // cheap clock for time-limited buffs
   327→  useEffect(() => {
   328→    const t = setInterval(() => setNow(Date.now()), 500)
   329→    return () => clearInterval(t)
   330→  }, [])
   331→
   332→  const conn = CONNECTION_STYLE[hud.connection]
   333→  const speedLeft = Math.max(0, Math.ceil((hud.speedBoostUntil - now) / 1000))
   334→  const lowHealth = hud.health / Math.max(1, hud.maxHealth) < 0.3
   335→
   336→  return (

   338→       spans the whole right half of the screen, so at z-20 it silently swallowed
   339→       every tap on right-side HUD elements (minimap → big map, the collapsed
   340→       squad chip, perf/env chips). The HUD root itself is pointer-events-none —
   341→       only the chips capture events — so look-dragging on empty space still works. */
   342→    <div className="pointer-events-none fixed inset-0 z-[35] select-none font-sans">
   343→      {/* low-health vignette */}
   344→      {lowHealth && !hud.detained && (
   345→        <div className="absolute inset-0 animate-[vc-blink_1.4s_ease-in-out_infinite] shadow-[inset_0_0_90px_rgba(140,31,40,0.6)]" />
   346→      )}
   347→      {/* hunted vignette (police/security chasing) — v51: dimmer + slower pulse while the badge fades out */}
   348→      {hud.hunted && !hud.detained && !lowHealth && (
   349→        <div
   350→          className={hud.wantedFading

   724→/** Bottom-center location chip — always-on, tap to copy "X .. · Z .." for sharing. */
   725→function CoordsChip({ mapX, mapZ }: { mapX: number; mapZ: number }) {
   726→  const [copied, setCopied] = useState(false)
   727→  const copy = async () => {
   728→    const text = `X ${mapX} · Z ${mapZ}`
   729→    try {
   730→      await navigator.clipboard.writeText(text)
   731→    } catch {
   732→      try {
   733→        const ta = document.createElement('textarea')
   734→        ta.value = text
   735→        ta.style.position = 'fixed'
   736→        ta.style.opacity = '0'
   737→        document.body.appendChild(ta)
   738→        ta.select()
   739→        document.execCommand('copy')
   740→        document.body.removeChild(ta)
   741→      } catch { /* clipboard unavailable */ }
   742→    }
   743→    setCopied(true)
   744→    useGameStore.getState().notify(`Coordinates copied — paste them in chat (T) so the squad can find you.`, 'info')
   745→    setTimeout(() => setCopied(false), 1600)
   746→  }
   747→  return (
   748→    <button
   749→      onClick={copy}
   750→      title="Click to copy your coordinates — share them in chat (T) so teammates can route to you"
   751→      aria-label={`Your location: X ${mapX}, Z ${mapZ}. Click to copy.`}
   752→      className="pointer-events-auto flex items-center gap-1.5 rounded-full border-2 border-[#1c1a17]/70 bg-[#1c1a17]/78 px-3 py-0.5 text-[10px] font-black tabular-nums tracking-wide text-[#f5efdc]/90 shadow-[2px_2px_0_rgba(28,26,23,0.55)] backdrop-blur-sm transition-all hover:border-[#f2c14e] hover:text-[#f2c14e]"
   753→    >
   754→      {copied ? <Check className="h-3 w-3 text-[#4e9b6e]" /> : <MapPin className="h-3 w-3 text-[#f2c14e]" />}
   755→      <span>X {mapX} · Z {mapZ}</span>
   756→      <span className="hidden items-center gap-0.5 rounded bg-[#f5efdc]/12 px-1 text-[7px] font-black uppercase tracking-widest text-[#f5efdc]/60 sm:flex">
   757→        <Copy className="h-2 w-2" /> copy
   758→      </span>
   759→    </button>
   760→  )
   761→}

   763→const CHAT_FEED_TTL_MS = 14000
   764→const CHAT_FEED_MAX = 4
   765→
   766→/** Bottom-left chat ticker: the freshest few comms lines, auto-fading after 14 s.
   767→ *  Coordinate lines carry a gold pin chip — clicking it sets the radar waypoint.
   768→ *  v46 — on touch devices it shows at most 2 lines and sits in the clear band
   769→ *  between the inventory (below) and the joystick (above); desktop keeps 4. */
   770→function ChatFeed() {
   771→  const chatLog = useGameStore((s) => s.chat.log)
   772→  const chatOpen = useGameStore((s) => s.chatOpen)
   773→  const [, forceTick] = useState(0)
   774→  const [isTouch] = useState(() => vcIsTouchDevice())
   775→  const feedMax = isTouch ? 2 : CHAT_FEED_MAX
   776→
   777→  // 1 Hz re-render so old lines fade out without any store writes
   778→  useEffect(() => {
   779→    if (chatLog.length === 0) return
   780→    const t = window.setInterval(() => forceTick((n) => n + 1), 1000)
   781→    return () => window.clearInterval(t)
   782→  }, [chatLog.length])
   783→
   784→  if (chatOpen) return null // the comms panel shows the full log — no double feed
   785→  const now = Date.now()
   786→  const fresh = chatLog.filter((c) => now - c.at < CHAT_FEED_TTL_MS).slice(-feedMax)
   787→  if (fresh.length === 0) return null
   788→
   789→  return (
   790→    <div className="pointer-events-none absolute bottom-[5.75rem] left-2 z-10 flex w-56 flex-col gap-1 sm:bottom-30 sm:left-3 sm:w-72">
   791→      <AnimatePresence>
   792→        {fresh.map((c) => (
   793→          <motion.div
   794→            key={c.id}
   795→            initial={{ opacity: 0, x: -14 }}
   796→            animate={{ opacity: 1, x: 0 }}
   797→            exit={{ opacity: 0, x: -10 }}
   798→            transition={{ type: 'spring', stiffness: 320, damping: 26 }}
   799→            className="pointer-events-auto flex items-start gap-1.5 rounded-xl border-2 border-[#1c1a17]/70 bg-[#f5efdc]/92 px-2 py-1 shadow-[2px_2px_0_rgba(28,26,23,0.65)] backdrop-blur-sm"
   800→          >
   801→            <MessageSquare className="mt-0.5 h-3 w-3 shrink-0 text-[#8c1f28]" />
   802→            <span className="min-w-0 text-[10px] leading-snug">
   803→              <span className={`font-black uppercase tracking-tight ${c.mine ? 'text-[#2e6e4e]' : c.bot ? 'text-[#1c1a17]/40' : 'text-[#8c1f28]'}`}>
   804→                {c.mine ? 'You' : c.name}
   805→              </span>
   806→              <span className="ml-1 break-words font-semibold text-[#1c1a17]/85">{c.text}</span>
   807→              {c.text.match(/[Xx]\s*-?\d/) && (
   808→                <span className="ml-1 inline-flex translate-y-px">
   809→                  <CoordChip text={c.text} who={c.mine ? undefined : c.name} />
   810→                </span>
   811→              )}
   812→            </span>
   813→          </motion.div>
   814→        ))}
   815→      </AnimatePresence>
   816→    </div>
   817→  )
   818→}
