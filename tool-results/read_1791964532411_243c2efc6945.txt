   301→              roomCode: this.lastRoomCode,

   308→              this.squadAutoRejoin = true
   309→              this.net.squadJoin(this.lastSquadCode)

   423→  // ================= match lifecycle =================
   424→  async startMatch(mode: 'solo' | 'multiplayer', roomCode: string | null) {
   425→    const store = useGameStore.getState()
   426→    store.setUIMode('loading')
   427→    this.mode = mode
   428→    store.setMatchMode(mode)
   429→    this.lastRoomCode = roomCode
   430→    this.reconnectNotified = false
   431→    this.clearOnboarding()
   432→    this.matchEnded = false
   433→    this.endHandled = false
   434→    // force-close any overlays so a fresh match never boots with stuck suppressed input
   435→    useGameStore.getState().toggleMap(false)
   436→    useGameStore.getState().toggleSquad(false)
   437→    useGameStore.getState().toggleEmoteWheel(false)
   438→    this.input.suppressed = false
   439→    this.tokens = 0
   440→    this.kudosSentTo.clear()

   653→  private onServerPlayers(players: import('./shared/types').PlayerNetState[]) {
   654→    if (!this.player) return
   655→    const seen = new Set<string>()

   669→    for (const [id, r] of this.remotePlayers) {
   670→      // AOI snapshots omit far bots — only prune entries we haven't heard from in 6s
   671→      if (!seen.has(id) && now - r.lastSeen > 6000) { this.removeRemote(id) }
   672→    }

   674→    useGameStore.getState().patchHud({

   681→  /**
   682→   * Rig budgeting: season worlds hold 100 protesters, but full character rigs are
   683→   * expensive. Humans always get rigs; bots get them by distance rank until the
   684→   * budget is spent (data-only bots remain as minimap/roster dots). A few rigs
   685→   * are created per pass to avoid a single-frame spike on arrival.
   686→   */
   687→  private ensureRemoteRigs(now: number) {
   688→    if (now - this.rigsRebuiltAt < 700) return
   689→    this.rigsRebuiltAt = now
   690→    const px = this.player!.pos.x
   691→    const pz = this.player!.pos.z
   692→    const entries = [...this.remotePlayers.entries()]
   693→    const humans = entries.filter(([id]) => !id.startsWith('bot_'))
   694→    const bots = entries
   695→      .filter(([id]) => id.startsWith('bot_'))
   696→      .sort((a, b) =>
   697→        ((a[1].target.x - px) ** 2 + (a[1].target.z - pz) ** 2) -
   698→        ((b[1].target.x - px) ** 2 + (b[1].target.z - pz) ** 2))
   699→    const botBudget = Math.max(4, Game.MAX_RIGS - humans.length)
   700→    let created = 0
   701→    // humans first — always rigged
   702→    for (const [id, r] of humans) if (!r.rig && created < 4) { this.attachRig(id, r); created++ }
   703→    for (const [id, r] of humans) if (!r.rig) this.attachRig(id, r)
   704→    // bots by distance rank
   705→    let rank = 0
   706→    for (const [id, r] of bots) {
   707→      if (rank < botBudget) {
   708→        if (!r.rig && created < 4) { this.attachRig(id, r); created++ }
   709→        else if (!r.rig) this.attachRig(id, r)
   710→      } else if (r.rig) {
   711→        this.engine.scene.remove(r.rig.group)
   712→        r.rig = null
   713→      }
   714→      rank++
   715→    }
   716→  }

   854→  /**
   855→   * Gift the most useful inventory item to the nearest live teammate (F key).
   856→   * Server validates distance/rate/type; a rejected gift is refunded via game:giftRefund,
   857→   * and a full-pocket receiver bounces the item back via game:giftBounce.
   858→   */
   859→  giftNearest() {
   860→    const store = useGameStore.getState()
   861→    if (!this.player || !this.running) return
   862→    if (this.offline) { store.notify('Gifting needs teammates — go multiplayer!', 'warning'); return }
   863→    if (this.player.isDetained || this.player.isDazed) { store.notify('Cannot gift while detained or dazed', 'warning'); return }
   864→    if (this.inventory.length === 0) { store.notify('Pockets are empty — nothing to gift!', 'info'); return }
   865→    // nearest live HUMAN within gift range (interpolated positions ≈ server positions;
   866→    // season bots carry no inventory so they are never gift targets)
   867→    let best: { id: string; name: string; d: number; health: number } | null = null
   868→    for (const [id, r] of this.remotePlayers) {

  1234→  private fixedUpdate(dt: number) {
  1235→    if (!this.running || !this.player) return
  1236→    const store = useGameStore.getState()
  1237→    if (store.uiMode !== 'playing') { return }
  1238→    this.worldTime += dt
  1239→    this.matchTime += dt
  1240→
  1241→    // input — apply look deltas to the camera rig (mouse/touch orbit)
  1242→    const [ldx, ldy] = this.input.consumeLook()
  1243→    this.camRig.yaw -= ldx
  1244→    this.camRig.pitch = this.photoMode
  1245→      ? Math.max(-0.2, Math.min(1.25, this.camRig.pitch + ldy))
  1246→      : Math.max(-0.55, Math.min(1.15, this.camRig.pitch + ldy))
  1247→    const oneShot = this.pendingOneShot
  1248→    this.pendingOneShot = null
  1249→    const boosted = this.worldTime < this.player.speedBoostUntil
  1250→    this.player.applyInput(dt, this.input.state, this.camRig.yaw, this.archetype, oneShot, boosted)
  1251→
  1252→    // world bounds safety
  1253→    const b = this.world.bounds
  1254→    this.player.pos.x = Math.max(b.minX, Math.min(b.maxX, this.player.pos.x))
  1255→    this.player.pos.z = Math.max(b.minZ, Math.min(b.maxZ, this.player.pos.z))
  1256→    if (this.player.pos.y < -5) { this.player.teleport(this.world.spawn.x, this.world.spawn.z) }
  1257→
