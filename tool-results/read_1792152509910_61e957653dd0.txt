   270→        localStorage.setItem('votechor_settings', JSON.stringify({ ...st.settings, minimapZoom: next, keybinds: { ...st.settings.keybinds } }))

   272→        /* private browsing — lives for this session only */
   273→      }

   646→/** Bottom-center location chip — always-on, tap to copy "X .. · Z .." for sharing. */
   647→function CoordsChip({ mapX, mapZ }: { mapX: number; mapZ: number }) {
   648→  const [copied, setCopied] = useState(false)
   649→  const copy = async () => {
   650→    const text = `X ${mapX} · Z ${mapZ}`
   651→    try {
   652→      await navigator.clipboard.writeText(text)
   653→    } catch {
   654→      try {
   655→        const ta = document.createElement('textarea')
   656→        ta.value = text
   657→        ta.style.position = 'fixed'
   658→        ta.style.opacity = '0'
   659→        document.body.appendChild(ta)
   660→        ta.select()
   661→        document.execCommand('copy')
   662→        document.body.removeChild(ta)
   663→      } catch { /* clipboard unavailable */ }
   664→    }
   665→    setCopied(true)
   666→    useGameStore.getState().notify(`Coordinates copied — paste them in chat (T) so the squad can find you.`, 'info')
   667→    setTimeout(() => setCopied(false), 1600)
   668→  }
   669→  return (
   670→    <button
   671→      onClick={copy}
   672→      title="Click to copy your coordinates — share them in chat (T) so teammates can route to you"
   673→      aria-label={`Your location: X ${mapX}, Z ${mapZ}. Click to copy.`}
   674→      className="pointer-events-auto flex items-center gap-1.5 rounded-full border-2 border-[#1c1a17]/70 bg-[#1c1a17]/78 px-3 py-0.5 text-[10px] font-black tabular-nums tracking-wide text-[#f5efdc]/90 shadow-[2px_2px_0_rgba(28,26,23,0.55)] backdrop-blur-sm transition-all hover:border-[#f2c14e] hover:text-[#f2c14e]"
   675→    >
   676→      {copied ? <Check className="h-3 w-3 text-[#4e9b6e]" /> : <MapPin className="h-3 w-3 text-[#f2c14e]" />}
   677→      <span>X {mapX} · Z {mapZ}</span>
   678→      <span className="hidden items-center gap-0.5 rounded bg-[#f5efdc]/12 px-1 text-[7px] font-black uppercase tracking-widest text-[#f5efdc]/60 sm:flex">
   679→        <Copy className="h-2 w-2" /> copy
   680→      </span>
   681→    </button>
   682→  )

   688→/** Bottom-left chat ticker: the freshest few comms lines, auto-fading after 14 s.
   689→ *  Coordinate lines carry a gold pin chip — clicking it sets the radar waypoint. */
   690→function ChatFeed() {
   691→  const chatLog = useGameStore((s) => s.chat.log)
   692→  const chatOpen = useGameStore((s) => s.chatOpen)
   693→  const [, forceTick] = useState(0)
   694→
   695→  // 1 Hz re-render so old lines fade out without any store writes
   696→  useEffect(() => {
   697→    if (chatLog.length === 0) return
   698→    const t = window.setInterval(() => forceTick((n) => n + 1), 1000)
   699→    return () => window.clearInterval(t)
   700→  }, [chatLog.length])
   701→
   702→  if (chatOpen) return null // the comms panel shows the full log — no double feed
   703→  const now = Date.now()
   704→  const fresh = chatLog.filter((c) => now - c.at < CHAT_FEED_TTL_MS).slice(-CHAT_FEED_MAX)
   705→  if (fresh.length === 0) return null
   706→
   707→  return (
   708→    <div className="pointer-events-none absolute bottom-28 left-2 z-10 flex w-64 flex-col gap-1 sm:bottom-30 sm:left-3 sm:w-72">
   709→      <AnimatePresence>
   710→        {fresh.map((c) => (
   711→          <motion.div
   712→            key={c.id}
   713→            initial={{ opacity: 0, x: -14 }}
   714→            animate={{ opacity: 1, x: 0 }}
   715→            exit={{ opacity: 0, x: -10 }}
   716→            transition={{ type: 'spring', stiffness: 320, damping: 26 }}
   717→            className="pointer-events-auto flex items-start gap-1.5 rounded-xl border-2 border-[#1c1a17]/70 bg-[#f5efdc]/92 px-2 py-1 shadow-[2px_2px_0_rgba(28,26,23,0.65)] backdrop-blur-sm"
   718→          >
