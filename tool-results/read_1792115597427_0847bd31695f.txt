    69→    69→const SKY_VERT = /* glsl */`
    70→    70→varying vec3 vDir;
    71→    71→void main() {
    72→    72→  vDir = position; // dome is centered on the camera, so object-space position IS the view direction
    73→    73→  vec4 mv = modelViewMatrix * vec4(position, 1.0);
    74→    74→  gl_Position = projectionMatrix * mv;
    75→    75→}`

    77→    77→const SKY_FRAG = /* glsl */`
    78→    78→uniform vec3 uTop;
    79→    79→uniform vec3 uHor;
    80→    80→uniform vec3 uBottom;
    81→    81→uniform vec3 uSunDir;
    82→    82→uniform vec3 uSunColor;
    83→    83→uniform float uGlow;
    84→    84→uniform float uHaze;
    85→    85→uniform float uSunDisc;
    86→    86→varying vec3 vDir;
    87→    87→void main() {
    88→    88→  vec3 d = normalize(vDir);
    89→    89→  float h = d.y;
    90→    90→  float t = pow(clamp(h, 0.0, 1.0), 0.52);
    91→    91→  vec3 col = mix(uHor, uTop, t);
    92→    92→  // below the horizon line: fade toward the darker ground haze
    93→    93→  col = mix(col, uBottom, smoothstep(0.015, -0.22, h));
    94→    94→  float sd = dot(d, normalize(uSunDir));
    95→    95→  // wide warm haze + tight glow around the sun
    96→    96→  col += uSunColor * pow(max(sd, 0.0), 5.0) * uHaze;
    97→    97→  col += uSunColor * pow(max(sd, 0.0), 60.0) * uGlow;
    98→    98→  // sun disc
    99→    99→  float disc = smoothstep(1.0 - uSunDisc, 1.0 - uSunDisc * 0.55, sd);
   100→   100→  col = mix(col, uSunColor * 1.45 + vec3(0.22), disc * step(-0.06, uSunDir.y));
   101→   101→  gl_FragColor = vec4(col, 1.0);
   102→   102→  #include <tonemapping_fragment>
   103→   103→  #include <colorspace_fragment>
   104→   104→}`
   105→   105→
   106→   106→const RAIN_VERT = /* glsl */`
   107→   107→attribute float seed;
   108→   108→attribute float tip;
   109→   109→uniform float uTime;
   110→   110→uniform vec3 uCam;
   111→   111→uniform float uArea;
   112→   112→uniform float uHeight;
   113→   113→uniform float uSpeed;
   114→   114→uniform float uWind;
   115→   115→uniform float uLen;
   116→   116→varying float vFade;
   117→   117→void main() {
   118→   118→  vec3 p = position;
   119→   119→  float fall = uTime * uSpeed * (0.75 + 0.5 * seed);
   120→   120→  p.y = mod(p.y - fall, uHeight);
   121→   121→  p.x = mod(p.x - uCam.x + uArea * 0.5 + uTime * uWind * 3.0, uArea) - uArea * 0.5 + uCam.x;
   122→   122→  p.z = mod(p.z - uCam.z + uArea * 0.5, uArea) - uArea * 0.5 + uCam.z;
   123→   123→  vec3 vel = normalize(vec3(uWind * 0.45, -1.0, uWind * 0.15));
   124→   124→  p += vel * uLen * (0.6 + 0.8 * seed) * tip;
   125→   125→  float dist = distance(p, uCam);
   126→   126→  vFade = smoothstep(60.0, 22.0, dist) * (0.55 + 0.45 * seed);
   127→   127→  gl_Position = projectionMatrix * viewMatrix * vec4(p, 1.0);
   128→   128→}`
   129→   129→
   130→   130→const RAIN_FRAG = /* glsl */`
   131→   131→uniform vec3 uColor;
   132→   132→uniform float uOpacity;
   133→   133→varying float vFade;
   134→   134→void main() {
   135→   135→  gl_FragColor = vec4(uColor, uOpacity * vFade);
   136→   136→  if (gl_FragColor.a < 0.01) discard;
   137→   137→}`

   358→   358→      const x = 26 + Math.random() * 108
   359→   359→      const y = 40 + Math.random() * 30
   360→   360→      const r = 18 + Math.random() * 22
   361→   361→      const g = ctx.createRadialGradient(x, y, 2, x, y, r)
   362→   362→      g.addColorStop(0, 'rgba(255,255,255,0.85)')
   363→   363→      g.addColorStop(0.65, 'rgba(255,255,255,0.4)')
   364→   364→      g.addColorStop(1, 'rgba(255,255,255,0)')
   365→   365→      ctx.fillStyle = g
   366→   366→      ctx.fillRect(0, 0, 160, 96)
   367→   367→    }
   368→   368→    const tex = new THREE.CanvasTexture(c)
   369→   369→    tex.colorSpace = THREE.SRGBColorSpace
   370→   370→    return tex
   371→   371→  }
   372→   372→
   373→   373→  setQuality(q: ResolvedQuality) {
   374→   374→    this.quality = q
   375→   375→  }

   377→   377→  setTimeMode(m: TimeMode) {
   378→   378→    if (m === this.timeMode) return
   379→   379→    this.timeMode = m
   380→   380→    if (m === 'dawn') this.timeOfDay = 5.9
   381→   381→    else if (m === 'noon') this.timeOfDay = 12.2
   382→   382→    else if (m === 'dusk') this.timeOfDay = 17.9
   383→   383→    else if (m === 'night') this.timeOfDay = 22.5
   384→   384→    // 'cycle' resumes from wherever the sky currently is
   385→   385→  }
   386→   386→
   387→   387→  setWeatherMode(m: WeatherMode) {
   388→   388→    if (m === this.weatherMode) return
   389→   389→    this.weatherMode = m
   390→   390→    if (m !== 'auto') this.targetWeather = m
   391→   391→    this.wTimer = 45 + Math.random() * 70
   392→   392→  }

   394→   394→  /** Human-readable environment info for the HUD chip. */
   395→   395→  envInfo(): { time: string; phase: string; weather: string; icon: 'sun' | 'sunrise' | 'sunset' | 'moon'; wIcon: WeatherKind } {
   396→   396→    const h = Math.floor(this.timeOfDay)
   397→   397→    const m = Math.floor((this.timeOfDay - h) * 60)
   398→   398→    const elev = this.sunElevation()
   399→   399→    let phase = 'Night'
   400→   400→    let icon: 'sun' | 'sunrise' | 'sunset' | 'moon' = 'moon'
   401→   401→    if (elev > 0.32) {
   402→   402→      phase = this.timeOfDay < 15 ? 'Morning' : 'Afternoon'
   403→   403→      icon = 'sun'
   404→   404→    } else if (elev > 0.08) {
   405→   405→      phase = this.timeOfDay < 12 ? 'Sunrise Glow' : 'Golden Hour'
   406→   406→      icon = 'sun'
   407→   407→    } else if (elev > -0.14) {
   408→   408→      const morningSide = this.timeOfDay >= 2 && this.timeOfDay < 12
   409→   409→      phase = morningSide ? 'Dawn' : 'Dusk'
   410→   410→      icon = morningSide ? 'sunrise' : 'sunset'
   411→   411→    }
   412→   412→    return { time: `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`, phase, weather: WEATHER[this.weather].label, icon, wIcon: this.weather }
   413→   413→  }
   414→   414→
   415→   415→  update(dt: number, camPos: THREE.Vector3) {
   416→   416→    this.time += dt
   417→   417→    this.camPos.copy(camPos)
   418→   418→
   419→   419→    // ---- clock ----
   420→   420→    if (this.timeMode === 'cycle') {
   421→   421→      this.timeOfDay = (this.timeOfDay + (dt / this.dayLength) * 24) % 24
   422→   422→    }
   423→   423→
   424→   424→    // ---- weather state machine ----
   425→   425→    if (this.weatherMode === 'auto') {
   426→   426→      this.wTimer -= dt
   427→   427→      if (this.wTimer <= 0) {
   428→   428→        this.wTimer = 50 + Math.random() * 75
   429→   429→        const opts = NEXT_WEATHER[this.targetWeather]
   430→   430→        let roll = Math.random()
   431→   431→        for (const [kind, p] of opts) {
   432→   432→          roll -= p
   433→   433→          if (roll <= 0) { this.targetWeather = kind; break }
   434→   434→        }
   435→   435→      }
   436→   436→    }
   437→   437→    this.weather = this.targetWeather
   438→   438→    const tgt = WEATHER[this.weather]
   439→   439→    const k = Math.min(1, dt * 0.22) // ~4.5s transition constant
   440→   440→    this.cur.cloudOp += (tgt.cloudOp - this.cur.cloudOp) * k
   441→   441→    this.cur.cloudDark += (tgt.cloudDark - this.cur.cloudDark) * k
   442→   442→    this.cur.rain += (tgt.rain - this.cur.rain) * k
   443→   443→    this.cur.fogMul += (tgt.fogMul - this.cur.fogMul) * k
   444→   444→    this.cur.sunMult += (tgt.sunMult - this.cur.sunMult) * k
   445→   445→    this.cur.wind += (tgt.wind - this.cur.wind) * k
   446→   446→    this.cur.lightning += (tgt.lightning - this.cur.lightning) * k
   447→   447→
   448→   448→    // ---- sun / moon geometry ----
   449→   449→    const dayAngle = ((this.timeOfDay - 6) / 24) * Math.PI * 2 // 6:00 sunrise on the horizon
   450→   450→    const sunDir = this.tmpDir.set(Math.cos(dayAngle), Math.sin(dayAngle) * 0.96 + 0.04, 0.38).normalize()
   451→   451→    const moonDir = sunDir.clone().multiplyScalar(-1); moonDir.z = Math.abs(moonDir.z) * 0.6 + 0.2; moonDir.normalize()
   452→   452→    const elev = sunDir.y
   453→   453→
   454→   454→    // ---- palette interpolation by sun elevation ----
   455→   455→    let i = 0
   456→   456→    while (i < P.length - 2 && elev > P[i + 1].e) i++
   457→   457→    const a = P[i], b = P[i + 1]
   458→   458→    const f = Math.max(0, Math.min(1, (elev - a.e) / (b.e - a.e)))
   459→   459→    this.lerpColor(this.cTop, a.top, b.top, f)
   460→   460→    this.lerpColor(this.cHor, a.hor, b.hor, f)
   461→   461→    this.lerpColor(this.cFog, a.fog, b.fog, f)
   462→   462→    this.lerpColor(this.cSun, a.sun, b.sun, f)
   463→   463→    this.lerpColor(this.cHemiS, a.hemiS, b.hemiS, f)
   464→   464→    this.lerpColor(this.cHemiG, a.hemiG, b.hemiG, f)
   465→   465→    const amb = a.amb + (b.amb - a.amb) * f
   466→   466→    const exp = a.exp + (b.exp - a.exp) * f
   467→   467→    const star = a.star + (b.star - a.star) * f
   468→   468→    const glow = a.glow + (b.glow - a.glow) * f
   469→   469→    const haze = a.haze + (b.haze - a.haze) * f
   470→   470→    this.lerpColor(this.cloudLit, a.cloudLit, b.cloudLit, f)
   471→   471→
   472→   472→    // ---- weather modifiers ----
   473→   473→    const dark = this.cur.cloudDark
   474→   474→    const grey = this.greyDay
   475→   475→    const nightGrey = this.greyNight
   476→   476→    this.cA.copy(this.cloudLit).lerp(elev > 0 ? grey : nightGrey, Math.min(1, dark * 1.1))
   477→   477→    // fog: weather tints (overcast grey, fog pale, rain blue-grey)
   478→   478→    const fogTint = dark * 0.62 + (this.weather === 'fog' ? 0.42 : 0) + this.cur.rain * 0.24
   479→   479→    this.cFog.lerp(this.fogTintCol, Math.min(0.75, fogTint))
   480→   480→
   481→   481→    // ---- sky dome uniforms ----
   482→   482→    const u = this.domeMat.uniforms
   483→   483→    ;(u.uTop.value as THREE.Color).copy(this.cTop).lerp(grey, dark * 0.62)
   484→   484→    ;(u.uHor.value as THREE.Color).copy(this.cHor).lerp(grey, dark * 0.68)
   485→   485→    ;(u.uBottom.value as THREE.Color).copy(this.cFog).multiplyScalar(0.85)
   486→   486→    ;(u.uSunDir.value as THREE.Vector3).copy(sunDir)
   487→   487→    ;(u.uSunColor.value as THREE.Color).copy(this.cSun).lerp(grey, dark * 0.6)
   488→   488→    u.uGlow.value = glow * this.cur.sunMult
   489→   489→    u.uHaze.value = haze * (0.4 + this.cur.sunMult * 0.6)
   490→   490→
   491→   491→    // ---- stars / moon ----
   492→   492→    this.starMat.opacity = star * Math.max(0, 1 - dark * 1.35)
   493→   493→    this.stars.rotation.y = this.time * 0.004 // slow celestial drift
   494→   494→    const moonUp = Math.max(0, moonDir.y)
   495→   495→    this.moonMat.opacity = moonUp * Math.max(0.15, 1 - dark) * Math.min(1, star + 0.15) * 0.95
   496→   496→    this.moon.position.copy(moonDir).multiplyScalar(348)
   497→   497→    this.moon.visible = this.moonMat.opacity > 0.02
   498→   498→
   499→   499→    // ---- clouds (positions are dome-LOCAL: the parent group already follows the camera) ----
   500→   500→    const cloudDim = Math.max(0.1, this.cur.sunMult * 0.55 + 0.45)
   501→   501→    for (const cl of this.clouds) {
   502→   502→      cl.angle += dt * 0.006 * cl.drift * (0.4 + this.cur.wind)
   503→   503→      const cx = Math.cos(cl.angle) * cl.radius
   504→   504→      const cz = Math.sin(cl.angle) * cl.radius * 0.8
   505→   505→      const cy = cl.height + Math.sin(this.time * 0.11 + cl.radius) * 3.5
   506→   506→      const op = cl.baseOp * this.cur.cloudOp * 0.92
   507→   507→      for (let s = 0; s < cl.sprites.length; s++) {
   508→   508→        const sp = cl.sprites[s]
   509→   509→        const off = cl.offsets[s]
   510→   510→        sp.position.set(cx + off.x, cy + off.y, cz + off.z)
   511→   511→        sp.material.opacity = op
   512→   512→        sp.material.color.copy(this.cA)
   513→   513→        sp.material.color.multiplyScalar(cloudDim)
   514→   514→        sp.visible = op > 0.03
   515→   515→      }
   516→   516→    }
   517→   517→
   518→   518→    // ---- rain ----
   519→   519→    const rainOp = this.cur.rain * 0.68
   520→   520→    this.rain.visible = rainOp > 0.02
   521→   521→    if (this.rain.visible) {
   522→   522→      this.rainMat.uniforms.uTime.value = this.time
   523→   523→      ;(this.rainMat.uniforms.uCam.value as THREE.Vector3).set(this.camPos.x, Math.max(1.6, this.camPos.y - 2.5), this.camPos.z)
   524→   524→      this.rainMat.uniforms.uWind.value = this.cur.wind
   525→   525→      this.rainMat.uniforms.uOpacity.value = rainOp
   526→   526→      ;(this.rainMat.uniforms.uColor.value as THREE.Color).setHex(elev > 0.05 ? 0xb6c6da : 0x7a84a0)
   527→   527→    }
   528→   528→
   529→   529→    // ---- lightning ----
   530→   530→    if (this.cur.lightning > 0.003) {
   531→   531→      this.nextFlashIn -= dt
   532→   532→      if (this.nextFlashIn <= 0 && this.flashT <= 0) {
   533→   533→        // interval scales with storm strength: 4–20s at full storm, minutes at a drizzle's odd rumble
   534→   534→        const strength = Math.max(0.25, this.cur.lightning / WEATHER.storm.lightning)
   535→   535→        this.nextFlashIn = (4 + Math.random() * 16) / strength
   536→   536→        this.flashT = 0.42
   537→   537→        this.onLightning?.()
   538→   538→      }
   539→   539→    }
   540→   540→    if (this.flashT > 0) {
   541→   541→      this.flashT -= dt
   542→   542→      const w = Math.max(0, this.flashT / 0.42)
   543→   543→      this.flashBoost = w * w * (Math.random() > 0.4 ? 1 : 0.45) // flickery decay
   544→   544→    } else this.flashBoost = 0
   545→   545→
   546→   546→    // ---- lights + fog + exposure ----
   547→   547→    const sunUp = this.smooth(-0.02, 0.16, elev)
   548→   548→    const moonUp2 = this.smooth(0.05, 0.3, moonDir.y)
   549→   549→    const useMoon = sunUp < 0.03 && moonUp2 > 0.05
   550→   550→    if (useMoon) {
   551→   551→      this.lightDir.copy(moonDir)
   552→   552→      this.sun.color.setHex(0xb4c6f0)
   553→   553→      // brighter moonlight so night reads as 'moonlit' not 'power cut'
   554→   554→      this.sun.intensity = 1.15 * moonUp2 * Math.max(0.3, 1 - dark * 0.75)
   555→   555→    } else {
   556→   556→      this.lightDir.copy(sunDir)
   557→   557→      this.sun.color.copy(this.cSun)
   558→   558→      this.sun.intensity = 2.15 * sunUp * this.cur.sunMult + this.flashBoost * 2.6
   559→   559→    }
   560→   560→    const shadowDir = this.lightDir.clone()
   561→   561→    shadowDir.y = Math.max(shadowDir.y, 0.24)
   562→   562→    shadowDir.normalize()
   563→   563→    this.lightDir.copy(shadowDir) // engine.trackSun uses this for the shadow frustum
   564→   564→
   565→   565→    this.hemi.color.copy(this.cHemiS)
   566→   566→    this.hemi.groundColor.copy(this.cHemiG)
   567→   567→    this.hemi.intensity = amb * (1 - dark * 0.5) + this.flashBoost * 2.2
   568→   568→
   569→   569→    // at night, pull the fog wall slightly closer so distant blackness hides behind
   570→   570→    // a friendlier blue haze instead of a void
   571→   571→    const fog = this.scene.fog as THREE.Fog | null
   572→   572→    if (fog) {
   573→   573→      fog.color.copy(this.cFog)
   574→   574→      const baseFar = this.quality === 'low' ? 150 : 210
   575→   575→      fog.far = baseFar * this.cur.fogMul * (1 + this._nightF * 0.12)
   576→   576→      fog.near = Math.min(60 * Math.pow(this.cur.fogMul, 0.7), fog.far * 0.4)
   577→   577→    }
   578→   578→
   579→   579→    this.exposureBase = exp
   580→   580→    this._nightF = this.smooth(0.1, -0.24, elev) // 0 by day → 1 at deep night
   581→   581→
   582→   582→    // ---- follow the camera ----
   583→   583→    this.group.position.copy(camPos)
   584→   584→  }
   585→   585→
   586→   586→  /** renderer.toneMappingExposure should be driven from this (game reads it each frame) */
   587→   587→  get exposure() {
   588→   588→    return this.exposureBase * (1 - this.cur.cloudDark * 0.3) + this.flashBoost * 0.5
   589→   589→  }

   591→   591→  get rainLevel() {
   592→   592→    return this.cur.rain
   593→   593→  }

   595→   595→  /** 0 = full day, 1 = deep night — the world renderer drives street lights from this */
   596→   596→  get nightFactor() {
   597→   597→    return this._nightF
   598→   598→  }
   599→   599→  private _nightF = 0

   601→   601→  private sunElevation(): number {
   602→   602→    const dayAngle = ((this.timeOfDay - 6) / 24) * Math.PI * 2
   603→   603→    return Math.sin(dayAngle) * 0.96 + 0.04
   604→   604→  }

   606→   606→  private lerpColor(out: THREE.Color, a: number, b: number, f: number) {
   607→   607→    this.cA.setHex(a)
   608→   608→    this.cB.setHex(b)
   609→   609→    out.copy(this.cA).lerp(this.cB, f)
   610→   610→  }
   611→   611→
   612→   612→  private smooth(a: number, b: number, x: number) {
   613→   613→    const t = Math.max(0, Math.min(1, (x - a) / (b - a)))
   614→   614→    return t * t * (3 - 2 * t)
   615→   615→  }

   621→   621→    this.dome.geometry.dispose()
   622→   622→    this.domeMat.dispose()
   623→   623→    this.stars.geometry.dispose()

   626→   626→    this.rainMat.dispose()
   627→   627→    for (const cl of this.clouds) for (const m of cl.mats) m.dispose()
   628→   628→    this.moonMat.dispose()
   629→   629→  }
