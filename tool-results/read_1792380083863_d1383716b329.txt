    12→function mulberry32(seed: number) {
    13→  let a = seed >>> 0
    14→  return () => {
    15→    a |= 0; a = (a + 0x6d2b79f5) | 0
    16→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
    17→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    18→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    19→  }
    20→}
