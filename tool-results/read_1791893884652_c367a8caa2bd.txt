   309→   309→  }

   316→   316→        [2500, 'Tip: follow the compass up top — gold markers are stolen vote tokens', 'info'],
   317→   317→        [9500, 'Tip: press E near tokens & items to grab them · Q throws a distraction', 'squad'],

   319→   319→        [23500, 'Tip: 12 tokens unlock the boss gate in the Government District. Good luck!', 'boss'],
   320→   320→      ]
   321→   321→      for (const [delay, text, kind] of tips) {
   322→   322→        this.onboardingTimers.push(setTimeout(() => {
   323→   323→          if (this.running) useGameStore.getState().notify(text, kind)

   326→   326→      localStorage.setItem('votechor_onboarded', '1')
   327→   327→    } catch { /* private browsing — skip */ }

   350→   350→  private onJoinAck(ack: JoinAck) {
   351→   351→    if (!ack.ok) {
   352→   352→      useGameStore.getState().setJoining(false, ack.error ?? 'Join failed')
   353→   353→      useGameStore.getState().notify(ack.error ?? 'Join failed', 'warning')
   354→   354→      return
   355→   355→    }
   356→   356→    this.offline = false
   357→   357→    this.net.playerId = ack.playerId
   358→   358→    this.serverMatchStart = ack.matchStartedAt
   359→   359→    useGameStore.getState().setRoomCode(ack.roomCode)
   360→   360→    useGameStore.getState().patchHud({ playersOnline: ack.players.length })
   361→   361→    this.tokens = ack.tokensCollected
   362→   362→    this.tokensNeeded = ack.tokensNeeded
   363→   363→    for (const p of ack.players) {
   364→   364→      if (p.id !== ack.playerId) this.addRemote(p)
   365→   365→    }
   366→   366→    // sync world state
   367→   367→    if (ack.checkpoints) ack.checkpoints.forEach(cp => this.activatedCheckpoints.add(cp))
   368→   368→    if (ack.boss && ack.boss.active) this.bossActiveLocal = true
   369→   369→    useGameStore.getState().notify(`Joined room ${ack.roomCode}`, 'squad')
   370→   370→  }

   372→   372→  private onServerPlayers(players: import('./shared/types').PlayerNetState[]) {
   373→   373→    if (!this.player) return
   374→   374→    const seen = new Set<string>()
   375→   375→    let online = 0
   376→   376→    for (const p of players) {
   377→   377→      online++
   378→   378→      seen.add(p.id)
   379→   379→      if (p.id === this.net.playerId) continue
   380→   380→      let r = this.remotePlayers.get(p.id)
   381→   381→      if (!r) { r = this.addRemote(p) }
   382→   382→      r.target = { x: p.x, y: p.y, z: p.z, ry: p.ry }
   383→   383→      r.anim = p.anim
   384→   384→      r.health = p.health
   385→   385→      r.detained = p.detained
   386→   386→      r.rig.group.visible = true
   387→   387→    }
   388→   388→    for (const [id, r] of this.remotePlayers) {
   389→   389→      if (!seen.has(id)) { this.removeRemote(id) }
   390→   390→    }
   391→   391→    useGameStore.getState().patchHud({ playersOnline: online })
   392→   392→  }

   394→   394→  private addRemote(p: import('./shared/types').PlayerNetState) {
   395→   395→    const rig = new CharacterRig({ archetype: p.archetype, name: p.name })
