    66→export default function CharacterSelect({ onBack }: { onBack: () => void }) {
    67→  const profile = useGameStore((s) => s.profile)
    68→  const setProfile = useGameStore((s) => s.setProfile)
    69→  const notify = useGameStore((s) => s.notify)
    70→
    71→  const [archetype, setArchetype] = useState(profile?.archetype ?? 'student')
    72→  const [hat, setHat] = useState(profile?.cosmetics?.hat ?? 'none')
    73→  const [hair, setHair] = useState(profile?.cosmetics?.hair ?? 'default')
    74→  const [gender, setGender] = useState<'male' | 'female'>(profile?.cosmetics?.gender === 'female' ? 'female' : 'male')
    75→  const [policeOutfit, setPoliceOutfit] = useState(profile?.cosmetics?.policeOutfit ?? 'khaki')
    76→  const [policeHat, setPoliceHat] = useState(profile?.cosmetics?.policeHat ?? 'peak')
    77→  const [face, setFace] = useState(profile?.cosmetics?.face ?? 'clean')
    78→  const [build, setBuild] = useState(profile?.cosmetics?.build ?? 'regular')
    79→  const [outfit, setOutfit] = useState(profile?.cosmetics?.outfit ?? 'classic')
    80→  const [footwear, setFootwear] = useState(profile?.cosmetics?.footwear ?? 'shoes')
    81→  const [gear, setGear] = useState(profile?.cosmetics?.gear ?? 'none')
    82→  const [held, setHeld] = useState(profile?.cosmetics?.held ?? 'none')
    83→  const [glasses, setGlasses] = useState(profile?.cosmetics?.glasses ?? false)
    84→  const [skinTone, setSkinTone] = useState(profile?.cosmetics?.skinTone ?? '')
    85→  const [signText, setSignText] = useState(profile?.cosmetics?.signText ?? '')
    86→  const [role, setRole] = useState<'protector' | 'police'>(profile?.role === 'police' ? 'police' : 'protector')
    87→  const [saving, setSaving] = useState(false)
    88→  const [unlockedCodes, setUnlockedCodes] = useState<Set<string>>(new Set())
    89→  const [tab, setTab] = useState<CosmeticKind>('hat')
    90→  const [hovered, setHovered] = useState<string | null>(null)
    91→
    92→  // fetch granted achievements → cosmetic unlocks (server remains authoritative on save)
    93→  // re-fetches on mount, on window focus, and every 15s so mid-session unlocks appear live
    94→  useEffect(() => {
    95→    let cancelled = false
    96→    const load = async () => {
    97→      try {
    98→        const raw = localStorage.getItem('votechor_profile')
    99→        const token = raw ? (JSON.parse(raw) as { token?: string }).token : undefined
   100→        if (!token) return
   101→        const res = await fetch('/api/achievements', { headers: { 'x-vc-token': token } })
   102→        if (!res.ok) return
   103→        const data = (await res.json()) as { unlocked?: string[] }
   104→        if (!cancelled && Array.isArray(data.unlocked)) setUnlockedCodes(new Set(data.unlocked))
   105→      } catch { /* offline — defaults stay unlocked */ }
   106→    }
   107→    void load()
   108→    window.addEventListener('focus', load)
   109→    const iv = setInterval(load, 15000)
   110→    return () => {
   111→      cancelled = true
   112→      window.removeEventListener('focus', load)
   113→      clearInterval(iv)
   114→    }
   115→  }, [])
   116→
   117→  const level = profile?.level ?? 1
   118→  const achName = (code: string) => ACHIEVEMENTS.find((a) => a.id === code)?.name ?? code.replaceAll('_', ' ')
   119→
   120→  const isUnlocked = (item: CosmeticItem) => {
   121→    if (item.unlock.type === 'free') return true
   122→    if (item.unlock.type === 'level') return level >= item.unlock.level
   123→    return unlockedCodes.has(item.unlock.code)
   124→  }
   125→
   126→  const token = () => {
   127→    try {
   128→      const raw = localStorage.getItem('votechor_profile')
   129→      return raw ? (JSON.parse(raw) as { token?: string }).token ?? null : null
   130→    } catch {
   131→      return null
   132→    }
   133→  }
   134→
   135→  // NOTE: the API REPLACES the stored cosmetics object — always persist the FULL look.
   136→  const currentLook = () => ({
   137→    hat, hair, gender, face, build, outfit, footwear, gear, held, glasses,
   138→    policeOutfit, policeHat,
   139→    ...(skinTone ? { skinTone } : {}),
   140→    signText: signText.slice(0, 24),
   141→  })
   142→
   143→  /** `overrides` carry the just-clicked values — React state lags one tick behind, so
   144→   *  posting plain currentLook() would silently save the PREVIOUS look (r23 bugfix). */
   145→  const persist = async (overrides?: Record<string, string | boolean>) => {
   146→    setSaving(true)
   147→    try {
   148→      const res = await fetch('/api/profile', {
   149→        method: 'POST',
   150→        headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   151→        body: JSON.stringify({ cosmetics: { ...currentLook(), ...overrides } }),
   152→      })
   153→      if (!res.ok) {
   154→        const data = (await res.json().catch(() => ({}))) as { error?: string }
   155→        notify(data.error ? `Wardrobe rejected: ${data.error}` : 'Wardrobe department did not answer. Changes kept locally.', 'warning')
   156→      }
   157→    } catch {
   158→      notify('Saved locally. The server will hear about this later.', 'warning')
   159→    } finally {
   160→      setSaving(false)
   161→    }
   162→  }
   163→
   164→  const setSlot = (kind: CosmeticKind, id: string, item?: CosmeticItem) => {
   165→    if (item && !isUnlocked(item)) {
   166→      notify(`Locked: ${item.label} — ${unlockLabel(item)}`, 'warning')
   167→      return
   168→    }
   169→    switch (kind) {
   170→      case 'hat': setHat(id); break
   171→      case 'hair': setHair(id); break
   172→      case 'face': setFace(id); break
   173→      case 'build': setBuild(id); break
   174→      case 'outfit': setOutfit(id); break
   175→      case 'footwear': setFootwear(id); break
   176→      case 'gear': setGear(id); break
   177→      case 'held': setHeld(id); break
   178→      case 'policeOutfit': setPoliceOutfit(id); break
   179→      case 'policeHat': setPoliceHat(id); break
   180→    }
   181→    const patch: Partial<Profile['cosmetics']> = {}
   182→    ;(patch as Record<string, string | boolean>)[kind] = id
   183→    uiActions().updateCosmetics(patch)
   184→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, [kind]: id } })
   185→    void persist({ [kind]: id })
   186→  }
   187→
   188→  const toggleGlasses = () => {
   189→    const next = !glasses
   190→    setGlasses(next)
   191→    uiActions().updateCosmetics({ glasses: next })
   192→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, glasses: next } })
   193→    void persist({ glasses: next })
   194→  }
   195→
   196→  const chooseSkin = (hex: string) => {
   197→    const next = skinTone === hex ? '' : hex
   198→    setSkinTone(next)
   199→    uiActions().updateCosmetics({ skinTone: next })
   200→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, skinTone: next } })
   201→    void persist({ skinTone: next })
   202→  }
   203→
   204→  /** Male / female build — also nudges hair to a fitting style while untouched. */
   205→  const chooseGender = (next: 'male' | 'female') => {
   206→    if (next === gender) return
   207→    setGender(next)
   208→    let nextHair = hair
   209→    if (hair === 'default' || hair === 'buzz' || hair === 'bald') {
   210→      // untouched masculine styles flip to the signature feminine ponytail (and back)
   211→      nextHair = next === 'female' ? 'ponytail' : 'default'
   212→      setHair(nextHair)
   213→    }
   214→    uiActions().updateCosmetics({ gender: next, hair: nextHair })
   215→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, gender: next, hair: nextHair } })
   216→    void persist({ gender: next, hair: nextHair })
   217→  }
   218→
   219→  const chooseArchetype = (id: string) => {
   220→    setArchetype(id)
   221→    uiActions().selectArchetype(id)
   222→    setProfile(profile ? { ...profile, archetype: id } : profile)
   223→    // archetype is a separate API field
   224→    ;(async () => {
   225→      setSaving(true)
   226→      try {
   227→        await fetch('/api/profile', {
   228→          method: 'POST',
   229→          headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   230→          body: JSON.stringify({ archetype: id, cosmetics: currentLook() }),
   231→        })
   232→      } catch { /* local-only */ } finally { setSaving(false) }
   233→    })()
   234→  }
   235→
   236→  /** Pick a side — persisted on the profile (server) + localStorage (offline fallback). */
   237→  const chooseRole = (next: 'protector' | 'police') => {
   238→    if (next === role) return
   239→    setRole(next)
   240→    setProfile(profile ? { ...profile, role: next } : profile)
   241→    try { localStorage.setItem('votechor_role', next) } catch { /* private browsing */ }
   242→    notify(next === 'police'
   243→      ? 'Badge accepted — you report for duty next match. Keep the lathi polite.'
   244→      : 'Welcome back to the people’s side — the protest missed you.', 'info')
   245→    ;(async () => {
   246→      setSaving(true)
   247→      try {
   248→        await fetch('/api/profile', {
   249→          method: 'POST',
   250→          headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   251→          body: JSON.stringify({ role: next }),
   252→        })
   253→      } catch { /* local-only */ } finally { setSaving(false) }
   254→    })()
   255→  }
   256→
   257→  const saveSign = () => {
   258→    const text = signText.slice(0, 24)
   259→    setSignText(text)
   260→    uiActions().updateCosmetics({ signText: text })
   261→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, signText: text } })
   262→    void persist({ signText: text })
   263→  }
   264→
   265→  const randomize = () => {
   266→    const pickFree = (catalog: CosmeticItem[]) => {
   267→      const pool = catalog.filter(isUnlocked)
   268→      return pool[Math.floor(Math.random() * pool.length)]?.id ?? catalog[0].id
   269→    }
   270→    const rg = gender === 'female' ? 'male' : 'female'
   271→    const female = rg === 'female'
   272→    const rh = pickFree(HAT_CATALOG); setHat(rh)
   273→    const rhair = female
   274→      ? (['bun', 'ponytail', 'braid', 'long', 'curly', 'default'] as const)[Math.floor(Math.random() * 6)]
   275→      : (['default', 'buzz', 'curly', 'bald', 'ponytail'] as const)[Math.floor(Math.random() * 5)]
   276→    setHair(rhair)
   277→    const rf = pickFree(FACE_CATALOG); setFace(rf)
   278→    const rb = pickFree(BUILD_CATALOG); setBuild(rb)
   279→    const ro = pickFree(OUTFIT_CATALOG); setOutfit(ro)
   280→    const rw = pickFree(FOOTWEAR_CATALOG); setFootwear(rw)
   281→    const rgr = pickFree(GEAR_CATALOG); setGear(rgr)
   282→    const rd = pickFree(HELD_CATALOG); setHeld(rd)
   283→    const rpo = pickFree(POLICE_OUTFIT_CATALOG); setPoliceOutfit(rpo)
   284→    const rph = pickFree(POLICE_HAT_CATALOG); setPoliceHat(rph)
   285→    const rs = Math.random() < 0.4
   286→    setGlasses(rs)
   287→    const tone = SKIN_TONES[Math.floor(Math.random() * SKIN_TONES.length)].hex
   288→    setSkinTone(tone)
   289→    setGender(rg)
   290→    uiActions().updateCosmetics({ hat: rh, hair: rhair, gender: rg, face: rf, build: rb, outfit: ro, footwear: rw, gear: rgr, held: rd, policeOutfit: rpo, policeHat: rph, glasses: rs, skinTone: tone })
   291→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, hat: rh, hair: rhair, gender: rg, face: rf, build: rb, outfit: ro, footwear: rw, gear: rgr, held: rd, policeOutfit: rpo, policeHat: rph, glasses: rs, skinTone: tone } })
   292→    void persist({ hat: rh, hair: rhair, gender: rg, face: rf, build: rb, outfit: ro, footwear: rw, gear: rgr, held: rd, policeOutfit: rpo, policeHat: rph, glasses: rs, skinTone: tone })
   293→    notify('Stylist rolled a fresh fit — democracy never looked better', 'squad')
   294→  }
   295→
   296→  const police = role === 'police'
   297→  const look: PreviewLook = useMemo(() => ({
   298→    archetype, hat, glasses, face, build, outfit, footwear, gear, held, signText: signText.trim(),
   299→    skinTone: skinTone || undefined,
   300→    gender, hair,
   301→    police,
   302→    policeOutfit, policeHat,
   303→  }), [archetype, hat, glasses, face, build, outfit, footwear, gear, held, signText, skinTone, gender, hair, police, policeOutfit, policeHat])
   304→
   305→  const sel = ARCHETYPES.find((a) => a.id === archetype) ?? ARCHETYPES[0]
   306→  const visibleSlots = police ? [...SLOTS, ...POLICE_SLOTS] : SLOTS
   307→  const activeTab = visibleSlots.find((s) => s.kind === tab) ?? visibleSlots[0]
   308→  const slotValues: Record<CosmeticKind, string> = { hat, hair, face, build, outfit, footwear, gear, held, policeOutfit, policeHat }
   309→  const equippedCount = visibleSlots.filter((s) => slotValues[s.kind] !== DEFAULTS[s.kind]).length + (glasses ? 1 : 0) + (skinTone ? 1 : 0)
   310→
   311→  return (
   312→    <div className="vc-paper pointer-events-auto fixed inset-0 z-30 flex flex-col overflow-hidden">
   313→      <div className="pointer-events-none absolute inset-0 opacity-[0.05] [background:repeating-linear-gradient(-35deg,#2e6e4e_0_18px,transparent_18px_36px)]" />
   314→
   315→      <header className="relative flex shrink-0 items-center gap-3 border-b-2 border-[#1c1a17]/80 bg-[#f2a541]/60 px-4 py-3">
   316→        <button
   317→          onClick={onBack}
   318→          className="flex items-center gap-1.5 rounded-xl border-2 border-[#1c1a17]/80 bg-[#f5efdc] px-3 py-1.5 text-xs font-black uppercase tracking-tight text-[#1c1a17] shadow-[3px_3px_0_#1c1a17] transition-all hover:-translate-y-0.5 hover:shadow-[4px_5px_0_#1c1a17] active:translate-x-[1px] active:translate-y-[1px] active:shadow-[1px_1px_0_#1c1a17]"
   319→        >
   320→          <ArrowLeft className="h-4 w-4" /> Back
   321→        </button>
   322→        <h2 className="text-xl font-black uppercase tracking-tight text-[#1c1a17] sm:text-2xl">Protester Wardrobe</h2>
   323→        <span className="hidden rounded-full border-2 border-[#1c1a17]/60 bg-[#f5efdc]/80 px-2.5 py-0.5 text-[10px] font-black uppercase tracking-wider text-[#1c1a17]/70 sm:inline">
   324→          {equippedCount} equipped
   325→        </span>
   326→        <div className="ml-auto flex items-center gap-2">
   327→          <button
   328→            onClick={randomize}
   329→            className="flex items-center gap-1.5 rounded-xl border-2 border-[#1c1a17]/80 bg-[#2e6e4e] px-3 py-1.5 text-xs font-black uppercase tracking-tight text-[#f5efdc] shadow-[3px_3px_0_#1c1a17] transition-all hover:-translate-y-0.5 hover:shadow-[4px_5px_0_#1c1a17] active:translate-x-[1px] active:translate-y-[1px]"
   330→          >
   331→            <Dices className="h-4 w-4" /> Surprise Fit
   332→          </button>
   333→          {saving && <Save className="h-4 w-4 animate-pulse text-[#8c1f28]" />}
   334→        </div>
   335→      </header>
   336→
   337→      <div className="vc-scroll min-h-0 flex-1 overflow-y-auto">
   338→        <div className="mx-auto max-w-6xl space-y-5 px-4 py-5">
   339→          {/* Archetype grid */}
   340→          <section aria-label="Characters">
   341→            <div className="mb-2 flex items-baseline gap-2">
   342→              <h3 className="text-sm font-black uppercase tracking-widest text-[#1c1a17]">1 · Choose Your Character</h3>
   343→              <span className="text-[10px] font-bold uppercase text-[#1c1a17]/50">{ARCHETYPES.length} rebels with quirks</span>
   344→            </div>
   345→            <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   346→              {ARCHETYPES.map((a, i) => {
   347→                const active = a.id === archetype
   348→                return (
   349→                  <motion.button
   350→                    key={a.id}
   351→                    initial={{ opacity: 0, y: 14 }}
   352→                    animate={{ opacity: 1, y: 0 }}
   353→                    transition={{ delay: i * 0.04 }}
   354→                    onClick={() => chooseArchetype(a.id)}
   355→                    className={`relative rounded-2xl border-2 p-3 text-left transition-all duration-150 ${
   356→                      active
   357→                        ? 'border-[#8c1f28] bg-[#f2c14e]/40 shadow-[5px_5px_0_#8c1f28]'
   358→                        : 'border-[#1c1a17]/70 bg-[#f5efdc] shadow-[4px_4px_0_#1c1a17] hover:-translate-y-0.5 hover:shadow-[5px_6px_0_#1c1a17]'
   359→                    }`}
   360→                  >
   361→                    {active && (
   362→                      <span className="absolute -right-2 -top-2 flex h-6 w-6 items-center justify-center rounded-full border-2 border-[#1c1a17] bg-[#2e6e4e] text-[#f5efdc]">
   363→                        <Check className="h-3.5 w-3.5" strokeWidth={3} />
   364→                      </span>
   365→                    )}
   366→                    {cssMiniPreview(a)}
   367→                    <div className="mt-2 text-sm font-black uppercase tracking-tight text-[#1c1a17]">{a.name}</div>
   368→                    <div className="mt-0.5 text-[11px] font-semibold leading-snug text-[#1c1a17]/70">{a.desc}</div>
   369→                    <div className="mt-2 flex items-start gap-1 rounded-lg bg-[#2e6e4e]/15 px-2 py-1 text-[10px] font-bold text-[#2e6e4e]">
   370→                      <Gem className="mt-0.5 h-3 w-3 shrink-0" /> {a.perk}
   371→                    </div>
   372→                  </motion.button>
   373→                )
   374→              })}
   375→            </div>
   376→          </section>
   377→
   378→          {/* Choose your side: Protector vs Police */}
   379→          <section aria-label="Choose your side">
   380→            <div className="mb-2 flex items-baseline gap-2">
   381→              <h3 className="text-sm font-black uppercase tracking-widest text-[#1c1a17]">2 · Choose Your Side</h3>
   382→              <span className="text-[10px] font-bold uppercase text-[#1c1a17]/50">two badges, one city — saves to your profile</span>
   383→            </div>
   384→            <div className="grid gap-3 sm:grid-cols-2">
   385→              <motion.button
   386→                initial={{ opacity: 0, y: 14 }}
   387→                animate={{ opacity: 1, y: 0 }}
   388→                transition={{ delay: 0.05 }}
   389→                onClick={() => chooseRole('protector')}
