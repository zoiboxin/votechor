    56→    56→export default function CharacterSelect({ onBack }: { onBack: () => void }) {

    59→    59→  const notify = useGameStore((s) => s.notify)
    60→    60→
    61→    61→  const [archetype, setArchetype] = useState(profile?.archetype ?? 'student')
    62→    62→  const [hat, setHat] = useState(profile?.cosmetics?.hat ?? 'none')
    63→    63→  const [face, setFace] = useState(profile?.cosmetics?.face ?? 'clean')
    64→    64→  const [outfit, setOutfit] = useState(profile?.cosmetics?.outfit ?? 'classic')
    65→    65→  const [footwear, setFootwear] = useState(profile?.cosmetics?.footwear ?? 'shoes')

    68→    68→  const [glasses, setGlasses] = useState(profile?.cosmetics?.glasses ?? false)

    76→    76→

    79→    79→  useEffect(() => {
    80→    80→    let cancelled = false
    81→    81→    const load = async () => {
    82→    82→      try {
    83→    83→        const raw = localStorage.getItem('votechor_profile')
    84→    84→        const token = raw ? (JSON.parse(raw) as { token?: string }).token : undefined
    85→    85→        if (!token) return
    86→    86→        const res = await fetch('/api/achievements', { headers: { 'x-vc-token': token } })

    90→    90→      } catch { /* offline — defaults stay unlocked */ }
    91→    91→    }
    92→    92→    void load()
    93→    93→    window.addEventListener('focus', load)
    94→    94→    const iv = setInterval(load, 15000)
    95→    95→    return () => {
    96→    96→      cancelled = true
    97→    97→      window.removeEventListener('focus', load)
    98→    98→      clearInterval(iv)
    99→    99→    }
   100→   100→  }, [])
   101→   101→
   102→   102→  const level = profile?.level ?? 1
   103→   103→  const achName = (code: string) => ACHIEVEMENTS.find((a) => a.id === code)?.name ?? code.replaceAll('_', ' ')
   104→   104→
   105→   105→  const isUnlocked = (item: CosmeticItem) => {
   106→   106→    if (item.unlock.type === 'free') return true
   107→   107→    if (item.unlock.type === 'level') return level >= item.unlock.level
   108→   108→    return unlockedCodes.has(item.unlock.code)
   109→   109→  }
   110→   110→
   111→   111→  const token = () => {
   112→   112→    try {
   113→   113→      const raw = localStorage.getItem('votechor_profile')
   114→   114→      return raw ? (JSON.parse(raw) as { token?: string }).token ?? null : null
   115→   115→    } catch {
   116→   116→      return null
   117→   117→    }
   118→   118→  }
   119→   119→
   120→   120→  // NOTE: the API REPLACES the stored cosmetics object — always persist the FULL look.

   122→   122→    hat, face, outfit, footwear, gear, held, glasses,
   123→   123→    ...(skinTone ? { skinTone } : {}),
   124→   124→    signText: signText.slice(0, 24),
   125→   125→  })

   127→   127→  const persist = async () => {
   128→   128→    setSaving(true)

   130→   130→      const res = await fetch('/api/profile', {
   131→   131→        method: 'POST',
   132→   132→        headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   133→   133→        body: JSON.stringify({ cosmetics: currentLook() }),
   134→   134→      })

   136→   136→        const data = (await res.json().catch(() => ({}))) as { error?: string }

   138→   138→      }
   139→   139→    } catch {
   140→   140→      notify('Saved locally. The server will hear about this later.', 'warning')
   141→   141→    } finally {
   142→   142→      setSaving(false)
   143→   143→    }
   144→   144→  }

   164→   164→  const toggleGlasses = () => {

   169→   169→    void persist()
   170→   170→  }
   171→   171→
   172→   172→  const chooseSkin = (hex: string) => {
   173→   173→    const next = skinTone === hex ? '' : hex

   175→   175→    uiActions().updateCosmetics({ skinTone: next })

   178→   178→  }

   186→   186→      setSaving(true)
   187→   187→      try {
   188→   188→        await fetch('/api/profile', {
   189→   189→          method: 'POST',
   190→   190→          headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   191→   191→          body: JSON.stringify({ archetype: id, cosmetics: currentLook() }),
   192→   192→        })
   193→   193→      } catch { /* local-only */ } finally { setSaving(false) }
   194→   194→    })()
   195→   195→  }
   196→   196→

   209→   209→        await fetch('/api/profile', {
   210→   210→          method: 'POST',
   211→   211→          headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   212→   212→          body: JSON.stringify({ role: next }),
   213→   213→        })
   214→   214→      } catch { /* local-only */ } finally { setSaving(false) }

   232→   232→    const rf = pickFree(FACE_CATALOG); setFace(rf)
   233→   233→    const ro = pickFree(OUTFIT_CATALOG); setOutfit(ro)
   234→   234→    const rw = pickFree(FOOTWEAR_CATALOG); setFootwear(rw)
   235→   235→    const rg = pickFree(GEAR_CATALOG); setGear(rg)
   236→   236→    const rd = pickFree(HELD_CATALOG); setHeld(rd)
   237→   237→    const rs = Math.random() < 0.4
   238→   238→    setGlasses(rs)
   239→   239→    const tone = SKIN_TONES[Math.floor(Math.random() * SKIN_TONES.length)].hex
   240→   240→    setSkinTone(tone)
   241→   241→    uiActions().updateCosmetics({ hat: rh, face: rf, outfit: ro, footwear: rw, gear: rg, held: rd, glasses: rs, skinTone: tone })
   242→   242→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, hat: rh, face: rf, outfit: ro, footwear: rw, gear: rg, held: rd, glasses: rs, skinTone: tone } })
   243→   243→    void persist()
   244→   244→    notify('Stylist rolled a fresh fit — democracy never looked better', 'squad')
   245→   245→  }
   246→   246→
   247→   247→  const look: PreviewLook = useMemo(() => ({
   248→   248→    archetype, hat, glasses, face, outfit, footwear, gear, held, signText: signText.trim(),
   249→   249→    skinTone: skinTone || undefined,
   250→   250→  }), [archetype, hat, glasses, face, outfit, footwear, gear, held, signText, skinTone])
   251→   251→
   252→   252→  const sel = ARCHETYPES.find((a) => a.id === archetype) ?? ARCHETYPES[0]
   253→   253→  const activeTab = SLOTS.find((s) => s.kind === tab) ?? SLOTS[0]
   254→   254→  const equippedCount = WARDROBE_KINDS.filter((k) => {
   255→   255→    const v = { hat, face, outfit, footwear, gear, held }[k]
   256→   256→    return v !== DEFAULTS[k]
   257→   257→  }).length + (glasses ? 1 : 0) + (skinTone ? 1 : 0)
   258→   258→
   259→   259→  return (
   260→   260→    <div className="vc-paper pointer-events-auto fixed inset-0 z-30 flex flex-col overflow-hidden">
   261→   261→      <div className="pointer-events-none absolute inset-0 opacity-[0.05] [background:repeating-linear-gradient(-35deg,#2e6e4e_0_18px,transparent_18px_36px)]" />

   267→   267→        >
   268→   268→          <ArrowLeft className="h-4 w-4" /> Back

   274→   274→        <div className="ml-auto flex items-center gap-2">

   277→   277→            className="flex items-center gap-1.5 rounded-xl border-2 border-[#1c1a17]/80 bg-[#2e6e4e] px-3 py-1.5 text-xs font-black uppercase tracking-tight text-[#f5efdc] shadow-[3px_3px_0_#1c1a17] transition-all hover:-translate-y-0.5 hover:shadow-[4px_5px_0_#1c1a17] active:translate-x-[1px] active:translate-y-[1px]"
   278→   278→          >
   279→   279→            <Dices className="h-4 w-4" /> Surprise Fit
   280→   280→          </button>
   281→   281→          {saving && <Save className="h-4 w-4 animate-pulse text-[#8c1f28]" />}
   282→   282→        </div>
   283→   283→      </header>
   284→   284→
   285→   285→      <div className="vc-scroll min-h-0 flex-1 overflow-y-auto">
   286→   286→        <div className="mx-auto max-w-6xl space-y-5 px-4 py-5">
   287→   287→          {/* Archetype grid */}
   288→   288→          <section aria-label="Characters">

   290→   290→              <h3 className="text-sm font-black uppercase tracking-widest text-[#1c1a17]">1 · Choose Your Character</h3>

   326→   326→          {/* Choose your side: Protector vs Police */}

   331→   331→            </div>
   332→   332→            <div className="grid gap-3 sm:grid-cols-2">
