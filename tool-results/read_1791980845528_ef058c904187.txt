     8→function canvasTex(w: number, h: number, draw: (ctx: CanvasRenderingContext2D) => void, repeatX = 1, repeatY = 1): THREE.CanvasTexture {
     9→  const c = document.createElement('canvas')
    10→  c.width = w; c.height = h
    11→  draw(c.getContext('2d')!)
    12→  const t = new THREE.CanvasTexture(c)
    13→  t.colorSpace = THREE.SRGBColorSpace
    14→  t.wrapS = t.wrapT = THREE.RepeatWrapping
    15→  t.repeat.set(repeatX, repeatY)
    16→  return t
    17→}

    19→function windowTexture(color: string, variant: number): THREE.CanvasTexture {
    20→  return canvasTex(128, 128, (ctx) => {
    21→    ctx.fillStyle = color
    22→    ctx.fillRect(0, 0, 128, 128)
    23→    // subtle grime
    24→    ctx.fillStyle = 'rgba(0,0,0,0.06)'
    25→    for (let i = 0; i < 20; i++) ctx.fillRect(Math.random() * 128, Math.random() * 128, 8, 2)
    26→    const cols = variant === 2 ? 3 : 4
    27→    const rows = 4
    28→    for (let r = 0; r < rows; r++) {
    29→      for (let c = 0; c < cols; c++) {
    30→        const x = 10 + c * (108 / cols), y = 12 + r * 28
    31→        ctx.fillStyle = Math.random() < 0.25 ? '#3a3630' : '#2c3947'
    32→        ctx.fillRect(x, y, 108 / cols - 12, 18)
    33→        ctx.strokeStyle = 'rgba(255,255,255,0.25)'
    34→        ctx.lineWidth = 2
    35→        ctx.strokeRect(x, y, 108 / cols - 12, 18)
    36→        // AC units / clotheslines on some
    37→        if (Math.random() < 0.2) { ctx.fillStyle = '#d8d3c5'; ctx.fillRect(x + 4, y + 20, 12, 6) }
    38→      }
    39→    }
    40→    ctx.fillStyle = 'rgba(0,0,0,0.15)'
    41→    ctx.fillRect(0, 120, 128, 8)
    42→  })
    43→}

    45→// matching emissive map: same window grid, but only the rooms that are awake at night
    46→// glow warm/cool. Driven by worldRenderer.setNight() — black by day, city lights at dusk.
    47→function windowEmissiveTexture(variant: number): THREE.CanvasTexture {
    48→  return canvasTex(128, 128, (ctx) => {
    49→    ctx.fillStyle = '#000'
    50→    ctx.fillRect(0, 0, 128, 128)
    51→    const cols = variant === 2 ? 3 : 4
    52→    const rows = 4
    53→    for (let r = 0; r < rows; r++) {
    54→      for (let c = 0; c < cols; c++) {
    55→        if (Math.random() < 0.42) continue // dark room — nobody home
    56→        const x = 10 + c * (108 / cols), y = 12 + r * 28
    57→        const roll = Math.random()
    58→        ctx.fillStyle = roll < 0.62 ? '#ffbe6e' : roll < 0.86 ? '#ffd696' : '#a8ccf0'
    59→        ctx.fillRect(x, y, 108 / cols - 12, 18)
    60→        // half-drawn curtains on a few
    61→        if (Math.random() < 0.18) { ctx.fillStyle = 'rgba(0,0,0,0.55)'; ctx.fillRect(x, y, 108 / cols - 12, 8) }
    62→      }
    63→    }
    64→  })
    65→}

    67→// soft radial blob used for lamp head halos + ground light pools
    68→function glowTexture(): THREE.CanvasTexture {
    69→  const c = document.createElement('canvas')
    70→  c.width = 128; c.height = 128
    71→  const ctx = c.getContext('2d')!
    72→  const g = ctx.createRadialGradient(64, 64, 4, 64, 64, 62)
    73→  g.addColorStop(0, 'rgba(255,214,140,0.9)')
    74→  g.addColorStop(0.35, 'rgba(255,190,110,0.38)')
    75→  g.addColorStop(1, 'rgba(255,180,90,0)')
    76→  ctx.fillStyle = g
    77→  ctx.fillRect(0, 0, 128, 128)
    78→  const t = new THREE.CanvasTexture(c)
    79→  t.colorSpace = THREE.SRGBColorSpace
    80→  return t
    81→}

    83→function signTexture(text: string, bg = '#f5efdc', fg = '#8c1f28'): THREE.CanvasTexture {
    84→  return canvasTex(256, 64, (ctx) => {
    85→    ctx.fillStyle = bg
    86→    ctx.fillRect(0, 0, 256, 64)
    87→    ctx.strokeStyle = '#5a4a2f'; ctx.lineWidth = 4
    88→    ctx.strokeRect(2, 2, 252, 60)
    89→    ctx.fillStyle = fg
    90→    ctx.font = 'bold 22px Arial'
    91→    ctx.textAlign = 'center'
    92→    ctx.fillText(text.slice(0, 26), 128, 40)
    93→  })
    94→}

   550→  private buildExtraStreetLamps() {

   552→    // march down the road, alternating curbs, arm always pointing toward the road
   553→    let z = 2

   560→      if (!clash) spots.push({ x, z, ry })
   561→      z -= 23
   562→      side *= -1
   563→    }

   571→    const poolG = new THREE.InstancedMesh(new THREE.PlaneGeometry(5.2, 5.2), this.poolMat, n)
   572→    const m = new THREE.Matrix4()
   573→    const q = new THREE.Quaternion()
   574→    const e = new THREE.Euler()

   580→      m.compose(new THREE.Vector3(s.x + Math.cos(s.ry) * 0.35, 5, s.z - Math.sin(s.ry) * 0.35), q, one); arm.setMatrixAt(i, m)

   796→  private buildDust() {
   797→    const N = 140
   798→    const pos = new Float32Array(N * 3)
   799→    for (let i = 0; i < N; i++) {
   800→      pos[i * 3] = (Math.random() - 0.5) * 90
   801→      pos[i * 3 + 1] = 0.5 + Math.random() * 8
   802→      pos[i * 3 + 2] = Math.random() * -580
   803→    }
   804→    const geo = new THREE.BufferGeometry()
   805→    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3))
   806→    const dust = new THREE.Points(geo, new THREE.PointsMaterial({ color: '#e8d5a8', size: 0.18, transparent: true, opacity: 0.35, depthWrite: false }))
   807→    this.group.add(dust)
   808→    this.animated.push((_t, dt) => {
   809→      const arr = geo.attributes.position.array as Float32Array
   810→      for (let i = 0; i < N; i++) {
   811→        arr[i * 3] += Math.sin(i + performance.now() * 0.0002) * dt * 0.4
   812→        arr[i * 3 + 1] += dt * 0.12
   813→        if (arr[i * 3 + 1] > 9) arr[i * 3 + 1] = 0.5
   814→      }
   815→      geo.attributes.position.needsUpdate = true
   816→    })
   817→  }

   823→  update(dt: number, t: number, searchlightRys?: Record<string, number>) {
   824→    for (const fn of this.animated) fn(t, dt)
   825→    // flag wave (cheap vertex-less: skew via rotation)
   826→    for (const f of this.flags) f.rotation.y = Math.sin(t * 2.2 + f.position.x) * 0.14
   827→    // searchlight rotation from server sim
   828→    if (searchlightRys) {
   829→      for (let i = 0; i < this.searchlightGroups.length; i++) {
   830→        const sl = this.world.searchlights[i]
   831→        const ry = searchlightRys[sl.id]
   832→        if (ry !== undefined) {
   833→          this.searchlightGroups[i].rotation.y = ry
   834→        }
   835→      }
   836→    }
   837→    // crowd subtle sway
   838→    if (this.crowdBodies) this.crowdBodies.rotation.y = Math.sin(t * 0.5) * 0.004
   839→
   840→    // ---- street light pool: re-park real lights onto the nearest lamps at ~3Hz ----
   841→    if (this.lightPool.length && this.lightPositions.length) {
   842→      this.poolTimer -= dt
   843→      if (this.poolTimer <= 0) {
   844→        this.poolTimer = 0.35
   845→        const cam = this.camPos
   846→        // partial selection: K nearest bulbs (N is ~80 — a simple scan is fine)
   847→        const k = this.lightPool.length
   848→        const idx = this.lightPositions
   849→          .map((p, i) => ({ i, d: (p.x - cam.x) * (p.x - cam.x) + (p.z - cam.z) * (p.z - cam.z) }))
   850→          .sort((a, b) => a.d - b.d)
   851→          .slice(0, k)
   852→        for (let j = 0; j < k; j++) {
   853→          const l = this.lightPool[j]
   854→          const src = this.lightPositions[idx[j].i]
   855→          l.position.set(src.x, src.y, src.z)
   856→          l.userData.phase = idx[j].i * 1.7
   857→        }
   858→      }
   859→      const base = 30 * this.nightF
   860→      for (let j = 0; j < this.lightPool.length; j++) {
   861→        const l = this.lightPool[j]
   862→        // faint sodium-vapor shimmer so the glow feels alive
   863→        l.intensity = base * (1 + 0.055 * Math.sin(t * 43 + (l.userData.phase ?? j * 7.3)))
   864→      }
   865→    }
   866→  }
