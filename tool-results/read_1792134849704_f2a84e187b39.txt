    11→    11→function mulberry32(seed: number) {
    12→    12→  let a = seed >>> 0
    13→    13→  return () => {
    14→    14→    a |= 0; a = (a + 0x6d2b79f5) | 0
    15→    15→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
    16→    16→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    17→    17→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    18→    18→  }
    19→    19→}
