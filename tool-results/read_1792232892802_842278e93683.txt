    66→export default function CharacterSelect({ onBack }: { onBack: () => void }) {
    67→  const profile = useGameStore((s) => s.profile)
    68→  const setProfile = useGameStore((s) => s.setProfile)
    69→  const notify = useGameStore((s) => s.notify)
    70→  const hiFiSetting = useGameStore((s) => s.settings.hiFiModel) // preview must mirror the in-game model (v52)
    71→
    72→  const [archetype, setArchetype] = useState(profile?.archetype ?? 'student')
    73→  const [hat, setHat] = useState(profile?.cosmetics?.hat ?? 'none')
    74→  const [hair, setHair] = useState(profile?.cosmetics?.hair ?? 'default')
    75→  const [gender, setGender] = useState<'male' | 'female'>(profile?.cosmetics?.gender === 'female' ? 'female' : 'male')
    76→  const [policeOutfit, setPoliceOutfit] = useState(profile?.cosmetics?.policeOutfit ?? 'khaki')
    77→  const [policeHat, setPoliceHat] = useState(profile?.cosmetics?.policeHat ?? 'peak')
    78→  const [face, setFace] = useState(profile?.cosmetics?.face ?? 'clean')
    79→  const [build, setBuild] = useState(profile?.cosmetics?.build ?? 'regular')
    80→  const [outfit, setOutfit] = useState(profile?.cosmetics?.outfit ?? 'classic')
    81→  const [footwear, setFootwear] = useState(profile?.cosmetics?.footwear ?? 'shoes')
    82→  const [gear, setGear] = useState(profile?.cosmetics?.gear ?? 'none')
    83→  const [held, setHeld] = useState(profile?.cosmetics?.held ?? 'none')
    84→  const [glasses, setGlasses] = useState(profile?.cosmetics?.glasses ?? false)
    85→  const [skinTone, setSkinTone] = useState(profile?.cosmetics?.skinTone ?? '')
    86→  const [signText, setSignText] = useState(profile?.cosmetics?.signText ?? '')
    87→  const [role, setRole] = useState<'protector' | 'police'>(profile?.role === 'police' ? 'police' : 'protector')
    88→  const [saving, setSaving] = useState(false)
    89→  const [unlockedCodes, setUnlockedCodes] = useState<Set<string>>(new Set())
    90→  const [tab, setTab] = useState<CosmeticKind>('hat')
    91→  const [hovered, setHovered] = useState<string | null>(null)
    92→
    93→  // fetch granted achievements → cosmetic unlocks (server remains authoritative on save)
    94→  // re-fetches on mount, on window focus, and every 15s so mid-session unlocks appear live
    95→  useEffect(() => {
    96→    let cancelled = false
    97→    const load = async () => {
    98→      try {
    99→        const raw = localStorage.getItem('votechor_profile')
   100→        const token = raw ? (JSON.parse(raw) as { token?: string }).token : undefined
   101→        if (!token) return
   102→        const res = await fetch('/api/achievements', { headers: { 'x-vc-token': token } })
   103→        if (!res.ok) return
   104→        const data = (await res.json()) as { unlocked?: string[] }
   105→        if (!cancelled && Array.isArray(data.unlocked)) setUnlockedCodes(new Set(data.unlocked))
   106→      } catch { /* offline — defaults stay unlocked */ }
   107→    }
   108→    void load()
   109→    window.addEventListener('focus', load)
   110→    const iv = setInterval(load, 15000)
   111→    return () => {
   112→      cancelled = true
   113→      window.removeEventListener('focus', load)
   114→      clearInterval(iv)
   115→    }
   116→  }, [])
   117→
   118→  const level = profile?.level ?? 1
   119→  const achName = (code: string) => ACHIEVEMENTS.find((a) => a.id === code)?.name ?? code.replaceAll('_', ' ')
   120→
   121→  const isUnlocked = (item: CosmeticItem) => {
   122→    if (item.unlock.type === 'free') return true
   123→    if (item.unlock.type === 'level') return level >= item.unlock.level
   124→    return unlockedCodes.has(item.unlock.code)
   125→  }
   126→
   127→  const token = () => {
   128→    try {
   129→      const raw = localStorage.getItem('votechor_profile')
   130→      return raw ? (JSON.parse(raw) as { token?: string }).token ?? null : null
   131→    } catch {
   132→      return null
   133→    }
   134→  }
   135→
   136→  // NOTE: the API REPLACES the stored cosmetics object — always persist the FULL look.
   137→  const currentLook = () => ({
   138→    hat, hair, gender, face, build, outfit, footwear, gear, held, glasses,
   139→    policeOutfit, policeHat,
   140→    ...(skinTone ? { skinTone } : {}),
   141→    signText: signText.slice(0, 24),
   142→  })
   143→
   144→  /** `overrides` carry the just-clicked values — React state lags one tick behind, so
   145→   *  posting plain currentLook() would silently save the PREVIOUS look (r23 bugfix). */
   146→  const persist = async (overrides?: Record<string, string | boolean>) => {
   147→    setSaving(true)
   148→    try {
   149→      const res = await fetch('/api/profile', {
   150→        method: 'POST',
   151→        headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   152→        body: JSON.stringify({ cosmetics: { ...currentLook(), ...overrides } }),
   153→      })
   154→      if (!res.ok) {
   155→        const data = (await res.json().catch(() => ({}))) as { error?: string }
   156→        notify(data.error ? `Wardrobe rejected: ${data.error}` : 'Wardrobe department did not answer. Changes kept locally.', 'warning')
   157→      }
   158→    } catch {
   159→      notify('Saved locally. The server will hear about this later.', 'warning')
   160→    } finally {
   161→      setSaving(false)
   162→    }
   163→  }
   164→
   165→  const setSlot = (kind: CosmeticKind, id: string, item?: CosmeticItem) => {
   166→    if (item && !isUnlocked(item)) {
   167→      notify(`Locked: ${item.label} — ${unlockLabel(item)}`, 'warning')
   168→      return
   169→    }
   170→    switch (kind) {
   171→      case 'hat': setHat(id); break
   172→      case 'hair': setHair(id); break
   173→      case 'face': setFace(id); break
   174→      case 'build': setBuild(id); break
   175→      case 'outfit': setOutfit(id); break
   176→      case 'footwear': setFootwear(id); break
   177→      case 'gear': setGear(id); break
   178→      case 'held': setHeld(id); break
   179→      case 'policeOutfit': setPoliceOutfit(id); break
   180→      case 'policeHat': setPoliceHat(id); break
   181→    }
   182→    const patch: Partial<Profile['cosmetics']> = {}
   183→    ;(patch as Record<string, string | boolean>)[kind] = id
   184→    uiActions().updateCosmetics(patch)
   185→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, [kind]: id } })
   186→    void persist({ [kind]: id })
   187→  }
   188→
   189→  const toggleGlasses = () => {
   190→    const next = !glasses
   191→    setGlasses(next)
   192→    uiActions().updateCosmetics({ glasses: next })
   193→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, glasses: next } })
   194→    void persist({ glasses: next })
   195→  }
   196→
   197→  const chooseSkin = (hex: string) => {
   198→    const next = skinTone === hex ? '' : hex
   199→    setSkinTone(next)
   200→    uiActions().updateCosmetics({ skinTone: next })
   201→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, skinTone: next } })
   202→    void persist({ skinTone: next })
   203→  }
   204→
   205→  /** Male / female build — also nudges hair to a fitting style while untouched. */
   206→  const chooseGender = (next: 'male' | 'female') => {
   207→    if (next === gender) return
   208→    setGender(next)
   209→    let nextHair = hair
   210→    if (hair === 'default' || hair === 'buzz' || hair === 'bald') {
   211→      // untouched masculine styles flip to the signature feminine ponytail (and back)
   212→      nextHair = next === 'female' ? 'ponytail' : 'default'
   213→      setHair(nextHair)
   214→    }
   215→    uiActions().updateCosmetics({ gender: next, hair: nextHair })
   216→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, gender: next, hair: nextHair } })
   217→    void persist({ gender: next, hair: nextHair })
   218→  }
   219→
   220→  const chooseArchetype = (id: string) => {
   221→    setArchetype(id)
   222→    uiActions().selectArchetype(id)
   223→    setProfile(profile ? { ...profile, archetype: id } : profile)
   224→    // archetype is a separate API field
   225→    ;(async () => {
   226→      setSaving(true)
   227→      try {
   228→        await fetch('/api/profile', {
   229→          method: 'POST',
   230→          headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   231→          body: JSON.stringify({ archetype: id, cosmetics: currentLook() }),
   232→        })
   233→      } catch { /* local-only */ } finally { setSaving(false) }
   234→    })()
   235→  }
   236→
   237→  /** Pick a side — persisted on the profile (server) + localStorage (offline fallback). */
   238→  const chooseRole = (next: 'protector' | 'police') => {
   239→    if (next === role) return
   240→    setRole(next)
   241→    setProfile(profile ? { ...profile, role: next } : profile)
   242→    try { localStorage.setItem('votechor_role', next) } catch { /* private browsing */ }
   243→    notify(next === 'police'
   244→      ? 'Badge accepted — you report for duty next match. Keep the lathi polite.'
   245→      : 'Welcome back to the people’s side — the protest missed you.', 'info')
   246→    ;(async () => {
   247→      setSaving(true)
   248→      try {
   249→        await fetch('/api/profile', {
   250→          method: 'POST',
   251→          headers: { 'Content-Type': 'application/json', 'x-vc-token': token() ?? '' },
   252→          body: JSON.stringify({ role: next }),
   253→        })
   254→      } catch { /* local-only */ } finally { setSaving(false) }
   255→    })()
   256→  }
   257→
   258→  const saveSign = () => {
   259→    const text = signText.slice(0, 24)
   260→    setSignText(text)
   261→    uiActions().updateCosmetics({ signText: text })
   262→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, signText: text } })
   263→    void persist({ signText: text })
   264→  }
   265→
   266→  const randomize = () => {
   267→    const pickFree = (catalog: CosmeticItem[]) => {
   268→      const pool = catalog.filter(isUnlocked)
   269→      return pool[Math.floor(Math.random() * pool.length)]?.id ?? catalog[0].id
   270→    }
   271→    const rg = gender === 'female' ? 'male' : 'female'
   272→    const female = rg === 'female'
   273→    const rh = pickFree(HAT_CATALOG); setHat(rh)
   274→    const rhair = female
   275→      ? (['bun', 'ponytail', 'braid', 'long', 'curly', 'default'] as const)[Math.floor(Math.random() * 6)]
   276→      : (['default', 'buzz', 'curly', 'bald', 'ponytail'] as const)[Math.floor(Math.random() * 5)]
   277→    setHair(rhair)
   278→    const rf = pickFree(FACE_CATALOG); setFace(rf)
   279→    const rb = pickFree(BUILD_CATALOG); setBuild(rb)
   280→    const ro = pickFree(OUTFIT_CATALOG); setOutfit(ro)
   281→    const rw = pickFree(FOOTWEAR_CATALOG); setFootwear(rw)
   282→    const rgr = pickFree(GEAR_CATALOG); setGear(rgr)
   283→    const rd = pickFree(HELD_CATALOG); setHeld(rd)
   284→    const rpo = pickFree(POLICE_OUTFIT_CATALOG); setPoliceOutfit(rpo)
   285→    const rph = pickFree(POLICE_HAT_CATALOG); setPoliceHat(rph)
   286→    const rs = Math.random() < 0.4
   287→    setGlasses(rs)
   288→    const tone = SKIN_TONES[Math.floor(Math.random() * SKIN_TONES.length)].hex
   289→    setSkinTone(tone)
   290→    setGender(rg)
   291→    uiActions().updateCosmetics({ hat: rh, hair: rhair, gender: rg, face: rf, build: rb, outfit: ro, footwear: rw, gear: rgr, held: rd, policeOutfit: rpo, policeHat: rph, glasses: rs, skinTone: tone })
   292→    if (profile) setProfile({ ...profile, cosmetics: { ...profile.cosmetics, hat: rh, hair: rhair, gender: rg, face: rf, build: rb, outfit: ro, footwear: rw, gear: rgr, held: rd, policeOutfit: rpo, policeHat: rph, glasses: rs, skinTone: tone } })
   293→    void persist({ hat: rh, hair: rhair, gender: rg, face: rf, build: rb, outfit: ro, footwear: rw, gear: rgr, held: rd, policeOutfit: rpo, policeHat: rph, glasses: rs, skinTone: tone })
   294→    notify('Stylist rolled a fresh fit — democracy never looked better', 'squad')
   295→  }
   296→
   297→  const police = role === 'police'
   298→  const look: PreviewLook = useMemo(() => ({
   299→    archetype, hat, glasses, face, build, outfit, footwear, gear, held, signText: signText.trim(),
   300→    skinTone: skinTone || undefined,
   301→    gender, hair,
   302→    police,
   303→    policeOutfit, policeHat,
   304→  }), [archetype, hat, glasses, face, build, outfit, footwear, gear, held, signText, skinTone, gender, hair, police, policeOutfit, policeHat])
   305→
   306→  const sel = ARCHETYPES.find((a) => a.id === archetype) ?? ARCHETYPES[0]
   307→  const visibleSlots = police ? [...SLOTS, ...POLICE_SLOTS] : SLOTS
   308→  const activeTab = visibleSlots.find((s) => s.kind === tab) ?? visibleSlots[0]
   309→  const slotValues: Record<CosmeticKind, string> = { hat, hair, face, build, outfit, footwear, gear, held, policeOutfit, policeHat }
   310→  const equippedCount = visibleSlots.filter((s) => slotValues[s.kind] !== DEFAULTS[s.kind]).length + (glasses ? 1 : 0) + (skinTone ? 1 : 0)
   311→
   312→  return (
   313→    <div className="vc-paper pointer-events-auto fixed inset-0 z-30 flex flex-col overflow-hidden">
   314→      <div className="pointer-events-none absolute inset-0 opacity-[0.05] [background:repeating-linear-gradient(-35deg,#2e6e4e_0_18px,transparent_18px_36px)]" />
   315→
   316→      <header className="relative flex shrink-0 items-center gap-3 border-b-2 border-[#1c1a17]/80 bg-[#f2a541]/60 px-4 py-3">
   317→        <button
   318→          onClick={onBack}
   319→          className="flex items-center gap-1.5 rounded-xl border-2 border-[#1c1a17]/80 bg-[#f5efdc] px-3 py-1.5 text-xs font-black uppercase tracking-tight text-[#1c1a17] shadow-[3px_3px_0_#1c1a17] transition-all hover:-translate-y-0.5 hover:shadow-[4px_5px_0_#1c1a17] active:translate-x-[1px] active:translate-y-[1px] active:shadow-[1px_1px_0_#1c1a17]"
   320→        >
   321→          <ArrowLeft className="h-4 w-4" /> Back
   322→        </button>
   323→        <h2 className="text-xl font-black uppercase tracking-tight text-[#1c1a17] sm:text-2xl">Protester Wardrobe</h2>
   324→        <span className="hidden rounded-full border-2 border-[#1c1a17]/60 bg-[#f5efdc]/80 px-2.5 py-0.5 text-[10px] font-black uppercase tracking-wider text-[#1c1a17]/70 sm:inline">
   325→          {equippedCount} equipped
   326→        </span>
   327→        <div className="ml-auto flex items-center gap-2">
   328→          <button
   329→            onClick={randomize}
   330→            className="flex items-center gap-1.5 rounded-xl border-2 border-[#1c1a17]/80 bg-[#2e6e4e] px-3 py-1.5 text-xs font-black uppercase tracking-tight text-[#f5efdc] shadow-[3px_3px_0_#1c1a17] transition-all hover:-translate-y-0.5 hover:shadow-[4px_5px_0_#1c1a17] active:translate-x-[1px] active:translate-y-[1px]"
   331→          >
   332→            <Dices className="h-4 w-4" /> Surprise Fit
   333→          </button>
   334→          {saving && <Save className="h-4 w-4 animate-pulse text-[#8c1f28]" />}
   335→        </div>
   336→      </header>
   337→
   338→      <div className="vc-scroll min-h-0 flex-1 overflow-y-auto">
   339→        <div className="mx-auto max-w-6xl space-y-5 px-4 py-5">
   340→          {/* Archetype grid */}
   341→          <section aria-label="Characters">
   342→            <div className="mb-2 flex items-baseline gap-2">
   343→              <h3 className="text-sm font-black uppercase tracking-widest text-[#1c1a17]">1 · Choose Your Character</h3>
   344→              <span className="text-[10px] font-bold uppercase text-[#1c1a17]/50">{ARCHETYPES.length} rebels with quirks</span>
   345→            </div>
   346→            <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   347→              {ARCHETYPES.map((a, i) => {
   348→                const active = a.id === archetype
   349→                return (
   350→                  <motion.button
   351→                    key={a.id}
   352→                    initial={{ opacity: 0, y: 14 }}
   353→                    animate={{ opacity: 1, y: 0 }}
   354→                    transition={{ delay: i * 0.04 }}
   355→                    onClick={() => chooseArchetype(a.id)}
   356→                    className={`relative rounded-2xl border-2 p-3 text-left transition-all duration-150 ${
   357→                      active
   358→                        ? 'border-[#8c1f28] bg-[#f2c14e]/40 shadow-[5px_5px_0_#8c1f28]'
   359→                        : 'border-[#1c1a17]/70 bg-[#f5efdc] shadow-[4px_4px_0_#1c1a17] hover:-translate-y-0.5 hover:shadow-[5px_6px_0_#1c1a17]'
   360→                    }`}
   361→                  >
   362→                    {active && (
   363→                      <span className="absolute -right-2 -top-2 flex h-6 w-6 items-center justify-center rounded-full border-2 border-[#1c1a17] bg-[#2e6e4e] text-[#f5efdc]">
   364→                        <Check className="h-3.5 w-3.5" strokeWidth={3} />
   365→                      </span>
   366→                    )}
   367→                    {cssMiniPreview(a)}
   368→                    <div className="mt-2 text-sm font-black uppercase tracking-tight text-[#1c1a17]">{a.name}</div>
   369→                    <div className="mt-0.5 text-[11px] font-semibold leading-snug text-[#1c1a17]/70">{a.desc}</div>
   370→                    <div className="mt-2 flex items-start gap-1 rounded-lg bg-[#2e6e4e]/15 px-2 py-1 text-[10px] font-bold text-[#2e6e4e]">
   371→                      <Gem className="mt-0.5 h-3 w-3 shrink-0" /> {a.perk}
   372→                    </div>
   373→                  </motion.button>
   374→                )
   375→              })}
   376→            </div>
   377→          </section>
   378→
   379→          {/* Choose your side: Protector vs Police */}
   380→          <section aria-label="Choose your side">
   381→            <div className="mb-2 flex items-baseline gap-2">
   382→              <h3 className="text-sm font-black uppercase tracking-widest text-[#1c1a17]">2 · Choose Your Side</h3>
   383→              <span className="text-[10px] font-bold uppercase text-[#1c1a17]/50">two badges, one city — saves to your profile</span>
   384→            </div>
   385→            <div className="grid gap-3 sm:grid-cols-2">
   386→              <motion.button
   387→                initial={{ opacity: 0, y: 14 }}
   388→                animate={{ opacity: 1, y: 0 }}
