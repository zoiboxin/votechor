     8→function canvasTex(w: number, h: number, draw: (ctx: CanvasRenderingContext2D) => void, repeatX = 1, repeatY = 1): THREE.CanvasTexture {
     9→  const c = document.createElement('canvas')
    10→  c.width = w; c.height = h
    11→  draw(c.getContext('2d')!)
    12→  const t = new THREE.CanvasTexture(c)
    13→  t.colorSpace = THREE.SRGBColorSpace
    14→  t.wrapS = t.wrapT = THREE.RepeatWrapping
    15→  t.repeat.set(repeatX, repeatY)
    16→  return t
    17→}

    19→function windowTexture(color: string, variant: number): THREE.CanvasTexture {
    20→  return canvasTex(128, 128, (ctx) => {
    21→    ctx.fillStyle = color
    22→    ctx.fillRect(0, 0, 128, 128)
    23→    // subtle grime
    24→    ctx.fillStyle = 'rgba(0,0,0,0.06)'
    25→    for (let i = 0; i < 20; i++) ctx.fillRect(Math.random() * 128, Math.random() * 128, 8, 2)
    26→    const cols = variant === 2 ? 3 : 4
    27→    const rows = 4
    28→    for (let r = 0; r < rows; r++) {
    29→      for (let c = 0; c < cols; c++) {
    30→        const x = 10 + c * (108 / cols), y = 12 + r * 28
    31→        ctx.fillStyle = Math.random() < 0.25 ? '#3a3630' : '#2c3947'
    32→        ctx.fillRect(x, y, 108 / cols - 12, 18)
    33→        ctx.strokeStyle = 'rgba(255,255,255,0.25)'
    34→        ctx.lineWidth = 2
    35→        ctx.strokeRect(x, y, 108 / cols - 12, 18)
    36→        // AC units / clotheslines on some
    37→        if (Math.random() < 0.2) { ctx.fillStyle = '#d8d3c5'; ctx.fillRect(x + 4, y + 20, 12, 6) }
    38→      }
    39→    }
    40→    ctx.fillStyle = 'rgba(0,0,0,0.15)'
    41→    ctx.fillRect(0, 120, 128, 8)
    42→  })
    43→}

    45→function signTexture(text: string, bg = '#f5efdc', fg = '#8c1f28'): THREE.CanvasTexture {
    46→  return canvasTex(256, 64, (ctx) => {
    47→    ctx.fillStyle = bg
    48→    ctx.fillRect(0, 0, 256, 64)
    49→    ctx.strokeStyle = '#5a4a2f'; ctx.lineWidth = 4
    50→    ctx.strokeRect(2, 2, 252, 60)
    51→    ctx.fillStyle = fg
    52→    ctx.font = 'bold 22px Arial'
    53→    ctx.textAlign = 'center'
    54→    ctx.fillText(text.slice(0, 26), 128, 40)
    55→  })
    56→}

   468→  private buildCrowd(world: WorldDef, quality: 'low' | 'medium' | 'high') {
   469→    const crowd = world.decorations.filter(d => d.kind === 'crowd')
   470→    if (!crowd.length) return
   471→    const count = quality === 'low' ? Math.floor(crowd.length * 0.5) : crowd.length
   472→    const bodies = new THREE.InstancedMesh(
   473→      new THREE.CylinderGeometry(0.22, 0.3, 1.05, 6),
   474→      new THREE.MeshStandardMaterial({ flatShading: true }),
   475→      count,
   476→    )
   477→    const heads = new THREE.InstancedMesh(
   478→      new THREE.SphereGeometry(0.15, 6, 5),
   479→      new THREE.MeshStandardMaterial({ color: '#c68a5c', flatShading: true }),
   480→      count,
   481→    )
   482→    const m = new THREE.Matrix4()
   483→    const color = new THREE.Color()
   484→    const shirtColors = ['#2f9e6e', '#d9822b', '#7ea8d8', '#b54a32', '#e6e0d2', '#8c4a5e']
   485→    let placed = 0
   486→    for (let i = 0; i < crowd.length && placed < count; i++) {
   487→      const c = crowd[i]
   488→      const bob = Math.random() * 0.05
   489→      m.makeTranslation(c.x, 0.62 + bob, c.z)
   490→      bodies.setMatrixAt(placed, m)
   491→      color.set(shirtColors[(c.variant ?? i) % shirtColors.length])
   492→      bodies.setColorAt(placed, color)
   493→      m.makeTranslation(c.x, 1.28 + bob, c.z)
   494→      heads.setMatrixAt(placed, m)
   495→      placed++
   496→      if (c.sign) {
   497→        // small sign group (max ~15 textured signs)
   498→        if (this.crowdSigns.length < 16) {
   499→          const sg = new THREE.Group()
   500→          const board = new THREE.Mesh(new THREE.PlaneGeometry(0.75, 0.45), new THREE.MeshBasicMaterial({ map: signTexture(c.sign, '#f5efdc', '#8c1f28'), side: THREE.DoubleSide }))
   501→          board.position.set(0, 1.55, 0)
   502→          sg.add(board)
   503→          sg.position.set(c.x, 0, c.z)
   504→          sg.rotation.y = (c.ry ?? 0) + Math.PI
   505→          this.crowdSigns.push(sg)
   506→          this.group.add(sg)
   507→        }
   508→      }
   509→    }
   510→    bodies.count = placed
   511→    heads.count = placed
   512→    bodies.castShadow = quality === 'high'
   513→    this.group.add(bodies, heads)
   514→    this.crowdBodies = bodies
   515→    this.crowdHeads = heads
   516→  }
