   141→   141→/** Lighten/darken a hex color by ±0-255 per channel (string in, string out). */
   142→   142→function shade(hex: string, amt: number): string {
   143→   143→  const c = new THREE.Color(hex)
   144→   144→  const f = (v: number) => Math.max(0, Math.min(1, v + amt / 255))
   145→   145→  return `#${new THREE.Color(f(c.r), f(c.g), f(c.b)).getHexString()}`

   149→   149→// Eyes + brows + nose + mouth are merged into ONE vertex-colored geometry per
   150→   150→// unique (skin, hair, gender) triple — a single extra draw call per rig.
   151→   151→function tintedClone(geo: THREE.BufferGeometry, color: string): THREE.BufferGeometry {
   152→   152→  const c = new THREE.Color(color)
   153→   153→  const src = geo.clone()
   154→   154→  const count = src.attributes.position.count
   155→   155→  const arr = new Float32Array(count * 3)
   156→   156→  for (let i = 0; i < count; i++) {
   157→   157→    arr[i * 3] = c.r; arr[i * 3 + 1] = c.g; arr[i * 3 + 2] = c.b
   158→   158→  }
   159→   159→  src.setAttribute('color', new THREE.BufferAttribute(arr, 3))
   160→   160→  return src
   161→   161→}

   178→   178→function faceGeometryFor(skinHex: string, hairHex: string, female: boolean): THREE.BufferGeometry {
   179→   179→  const key = `${skinHex}|${hairHex}|${female ? 'f' : 'm'}`
   180→   180→  const hit = faceGeoCache.get(key)
   181→   181→  if (hit) return hit
   182→   182→  const parts: THREE.BufferGeometry[] = []
   183→   183→  const place = (
   184→   184→    geo: THREE.BufferGeometry, color: string,
   185→   185→    px: number, py: number, pz: number,
   186→   186→    sx = 1, sy = 1, sz = 1, rz = 0,
   187→   187→  ) => {
   188→   188→    const g = tintedClone(geo, color)
   189→   189→    const m = new THREE.Matrix4()
   190→   190→    m.compose(
   191→   191→      new THREE.Vector3(px, py, pz),
   192→   192→      new THREE.Quaternion().setFromEuler(new THREE.Euler(0, 0, rz)),
   193→   193→      new THREE.Vector3(sx, sy, sz),
   194→   194→    )
   195→   195→    g.applyMatrix4(m)
   196→   196→    parts.push(g)
   197→   197→  }
   198→   198→  const dark = '#241a12'
   199→   199→  // ears — subtle skin-shaded shells on the sides (all genders)
   200→   200→  for (const side of [1, -1]) {
   201→   201→    place(G.ear, shade(skinHex, -22), side * 0.15, 0.1, -0.008, 0.55, 0.82, 0.62)
   202→   202→  }
   203→   203→  // eyes — almond white + brown iris + dark pupil + tiny catchlight dot.
   204→   204→  // Pushed proud of the (now smooth) head sphere so they read at any quality.
   205→   205→  for (const side of [1, -1]) {

   210→   210→    // lash line — thin dark arc hugging the top of the eye

   216→   216→    place(G.brow, shade(hairHex, 4), side * 0.076, 0.161, 0.138, 0.82, female ? 0.8 : 0.95, 1, side * 0.3)
   217→   217→  }
   218→   218→  // female-only: soft cheek blush baked as a skin/rose blend (vertex colors only)
   219→   219→  if (female) {
   220→   220→    const blushCol = new THREE.Color(skinHex).lerp(new THREE.Color('#d46a6a'), 0.38).getStyle()
   221→   221→    for (const side of [1, -1]) {

   223→   223→    }
   224→   224→  }

   230→   230→  }
   231→   231→  // mouth — fuller two-tone lips for female, thinner shaded line for male
   232→   232→  if (female) {

   238→   238→  const merged = mergeGeometries(parts, false) ?? new THREE.BufferGeometry()
   239→   239→  for (const p of parts) p.dispose()
   240→   240→  sharedGeo.add(merged)

   246→   246→// The original body was plain boxes with legs that stopped 0.335 units short of

   251→   251→function xg(
   252→   252→  geo: THREE.BufferGeometry,
   253→   253→  px: number, py: number, pz: number,

   259→   259→    new THREE.Quaternion().setFromEuler(new THREE.Euler(rx, ry, rz)),
   260→   260→    new THREE.Vector3(sx, sy, sz),
   261→   261→  )
   262→   262→  g.applyMatrix4(m)

   284→   284→function bodyProfile(female: boolean, build: BodyBuild): BodyProfile {
   285→   285→  const p: BodyProfile = female
   286→   286→    ? // hourglass: narrower shoulders/chest, pinched waist, wider hips
   287→   287→      { shoulder: 0.19, chest: 0.163, waist: 0.143, hipTop: 0.148, hipBottom: 0.182, armW: 0.86, legW: 0.9 }
   288→   288→    : // classic V: broad shoulders tapering to the waist
   289→   289→      { shoulder: 0.225, chest: 0.185, waist: 0.163, hipTop: 0.166, hipBottom: 0.156, armW: 1, legW: 1 }
   290→   290→  if (build === 'slim') {
   291→   291→    p.shoulder *= 0.92; p.chest *= 0.92; p.waist *= 0.9
   292→   292→    p.hipTop *= 0.92; p.hipBottom *= 0.92; p.armW *= 0.94; p.legW *= 0.92
   293→   293→  } else if (build === 'sturdy') {
   294→   294→    p.shoulder = Math.max(p.shoulder * 1.08, 0.235)
   295→   295→    p.chest *= 1.14; p.waist *= 1.2
   296→   296→    p.hipTop *= 1.1; p.hipBottom *= 1.08; p.armW *= 1.12; p.legW *= 1.12
   297→   297→  }
   298→   298→  return p
   299→   299→}

   358→   358→  ])

   362→   362→ * Full shoe — ONE vertex-colored merged geometry (leg-pivot space).
   363→   363→ * The sole bottom sits at y -0.9 == world ground (leg pivot is at 0.9).
   364→   364→ * Sneaker anatomy: white midsole + rounded toe, colored upper + toe cap + heel
   365→   365→ * collar, lace rows, side stripes, heel pull tab. The upper takes the ACCENT
   366→   366→ * color so shoes stay customizable through the existing palette system.
