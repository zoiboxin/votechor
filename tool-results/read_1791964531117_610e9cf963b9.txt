     1→     1→// VoteChor Seasons — persistent 100-protester worlds for the solo queue.
     2→     2→//
     3→     3→// Design (user request):
     4→     4→//  • A solo player joins ONE season. Every season holds up to 100 occupants.
     5→     5→//  • Every free slot is filled by a server-simulated bot protester, so a fresh
     6→     6→//    season immediately feels like a living city (you + 99 bots).
     7→     7→//  • When a NEW human joins a season, one bot steps aside (removed + announced)
     8→     8→//    so the human takes its place — population stays at 100.
     9→     9→//  • When ANY occupant leaves, a fresh bot walks in from the plaza to backfill.
    10→    10→//  • When a season is "full" of humans (SEASON_MAX_HUMANS), matchmaking creates
    11→    11→//    the next season instead.
    12→    12→//  • Bots are realistic: they wander the streets on a nav graph, idle, chat
    13→    13→//    (quick messages), emote, socialize with each other AND with humans, flee
    14→    14→//    police, join protest marches to checkpoints, and pick up supplies.
    15→    15→//    Police will still chase them (they are real NPC targets), the boss ignores
    16→    16→//    them, and they get detained / released like everyone else.
    17→    17→//
    18→    18→// Network: bots are regular PlayerState entries inside the Room, so snapshots,
    19→    19→// NPC brains, vehicles, checkpoints, detention and the client renderer all work
    20→    20→// on them with zero special-casing. A per-recipient AOI broadcaster keeps the
    21→    21→// 12 Hz player snapshot cheap (near bots at full rate, far bots on a sweep).

    67→    67→function angleLerp(from: number, to: number, t: number): number {
    68→    68→  let d = to - from
    69→    69→  while (d > Math.PI) d -= Math.PI * 2
    70→    70→  while (d < -Math.PI) d += Math.PI * 2
    71→    71→  return from + d * Math.min(1, t)
    72→    72→}

    74→    74→// ---------- nav graph (built once from the shared deterministic world) ----------
    75→    75→interface NavNode { x: number; z: number; links: number[] }
    76→    76→
    77→    77→const NAV_GRID = 10
    78→    78→const NAV_PAD = 1.7 // keep this far from building walls
    79→    79→const NAV_Z_MAX = -474 // bots never wander into the Government District / boss arena
    80→    80→
    81→    81→function pointBlocked(x: number, z: number, pad: number): boolean {
    82→    82→  for (const c of WORLD.colliders) {
    83→    83→    if (c.walkable) continue // low walkable platforms (stage) are fine
    84→    84→    if (x >= c.minX - pad && x <= c.maxX + pad && z >= c.minZ - pad && z <= c.maxZ + pad) return true
    85→    85→  }
    86→    86→  return false
    87→    87→}
    88→    88→
    89→    89→function clearPath(ax: number, az: number, bx: number, bz: number): boolean {
    90→    90→  const d = dist2d(ax, az, bx, bz)
    91→    91→  const steps = Math.max(1, Math.ceil(d / 2))
    92→    92→  for (let i = 0; i <= steps; i++) {
    93→    93→    const t = i / steps
    94→    94→    if (pointBlocked(ax + (bx - ax) * t, az + (bz - az) * t, 0.8)) return false
    95→    95→  }
    96→    96→  return true
    97→    97→}

    99→    99→function buildNavGraph(): NavNode[] {
   100→   100→  const nodes: NavNode[] = []
   101→   101→  const index = new Map<string, number>()
   102→   102→  for (let x = -45; x <= 45; x += NAV_GRID) {
   103→   103→    for (let z = NAV_Z_MAX; z <= 6; z += NAV_GRID) {
   104→   104→      if (pointBlocked(x, z, NAV_PAD)) continue
   105→   105→      index.set(`${x}|${z}`, nodes.length)
   106→   106→      nodes.push({ x, z, links: [] })
   107→   107→    }
   108→   108→  }
   109→   109→  // 8-neighborhood edges with line-of-sight (no cutting through buildings)
   110→   110→  for (let i = 0; i < nodes.length; i++) {
   111→   111→    const n = nodes[i]
   112→   112→    for (const [dx, dz] of [[NAV_GRID, 0], [0, NAV_GRID], [NAV_GRID, NAV_GRID], [NAV_GRID, -NAV_GRID]] as const) {
   113→   113→      const j = index.get(`${n.x + dx}|${n.z + dz}`)
   114→   114→      if (j === undefined) continue
   115→   115→      const m = nodes[j]
   116→   116→      if (clearPath(n.x, n.z, m.x, m.z)) { n.links.push(j); m.links.push(i) }
   117→   117→    }
   118→   118→  }
   119→   119→  return nodes
   120→   120→}

   124→   124→function nearestNode(x: number, z: number): number {
   125→   125→  let best = 0
   126→   126→  let bestD = Infinity
   127→   127→  for (let i = 0; i < NAV.length; i++) {
   128→   128→    const d = (NAV[i].x - x) ** 2 + (NAV[i].z - z) ** 2
   129→   129→    if (d < bestD) { bestD = d; best = i }
   130→   130→  }
   131→   131→  return best
   132→   132→}

   344→   344→  /** A bot to sacrifice for an incoming human — prefer one NOT standing next to a human. */
   345→   345→  private pickRemovableBot(): Bot | null {
   346→   346→    const all = [...this.bots.values()]
   347→   347→    if (all.length === 0) return null
   348→   348→    const humans = this.room.humans()
   349→   349→    const away = all.filter((b) => !humans.some((h) => dist2d(h.x, h.z, b.p.x, b.p.z) < 9))
   350→   350→    return pick(away.length > 0 ? away : all)

   355→   355→  private tick() {
   356→   356→    if (this.disposed) return
   357→   357→    const now = Date.now()
   358→   358→    const dt = TICK_MS / 1000
   359→   359→
   360→   360→    for (const bot of this.bots.values()) this.stepBot(bot, dt, now)
   361→   361→
   362→   362→    // churn — the world breathes: someone leaves, someone new arrives
   363→   363→    if (now >= this.churnAt && this.room.humansCount() > 0) {
   364→   364→      const victim = this.pickRemovableBot()
   365→   365→      if (victim) {
   366→   366→        this.removeBot(victim, true)
   367→   367→        const t = setTimeout(() => {
   368→   368→          if (this.disposed) return
   369→   369→          if (this.room.humansCount() + this.bots.size < SEASON_CAP) this.createBot('arrive', true)
   370→   370→          this.emitSeason()
   371→   371→        }, rand(2500, 7000))
   372→   372→        t.unref?.()
   373→   373→        this.churnTimers.push(t)
   374→   374→      }
   375→   375→      this.churnAt = now + rand(CHURN_MIN_MS, CHURN_MAX_MS)
   376→   376→    }
   377→   377→
   378→   378→    // protest marches to the next checkpoint
   379→   379→    if (now >= this.marchAt && this.room.humansCount() > 0) {
   380→   380→      this.startMarch()
   381→   381→      this.marchAt = now + rand(MARCH_MIN_MS, MARCH_MAX_MS)
   382→   382→    }
   383→   383→  }
   384→   384→
   385→   385→  private startMarch() {
   386→   386→    const inactive = WORLD.checkpoints.filter((c) => !this.room.checkpoints.has(c.id))
   387→   387→    if (inactive.length === 0) return
   388→   388→    const cp = pick(inactive)
   389→   389→    const crowd = [...this.bots.values()].filter((b) => b.state === 'wander' && !b.p.detained && !b.p.dazed)
   390→   390→    if (crowd.length < 3) return
   391→   391→    const n = Math.min(crowd.length, randInt(3, 8))
   392→   392→    for (let i = 0; i < n; i++) {
   393→   393→      const b = crowd.splice(randInt(0, crowd.length - 1), 1)[0]
   394→   394→      b.state = 'march'
   395→   395→      b.marchCp = cp.id
   396→   396→      b.goal = { x: cp.x + rand(-5, 5), z: cp.z + rand(-5, 5) }
   397→   397→      b.wp = null
   398→   398→      b.speed = rand(3.8, 4.4)
   399→   399→    }
   400→   400→    // a march leader shouts the plan
   401→   401→    const leader = this.bots.get(crowd[0]?.p.id ?? '')
   402→   402→    if (leader) this.io.to(this.room.id).emit('game:quickmsg', leader.p.id, randInt(5, 8)) // "To the flyover!" / "Form a squad!" / "The gate needs votes!"
   403→   403→  }
   404→   404→
   405→   405→  private stepBot(bot: Bot, dt: number, now: number) {
   406→   406→    const p = bot.p
   407→   407→    if (p.detained || p.dazed) return // capture/daze systems own the bot
   408→   408→
   409→   409→    // --- threat scan: police chasing me, or an officer breathing down my neck ---
   410→   410→    if (bot.state !== 'flee') {
   411→   411→      for (const n of this.room.npcs) {
   412→   412→        if (n.type === 'searchlight') continue
   413→   413→        const d = dist2d(n.x, n.z, p.x, p.z)
   414→   414→        if ((n.state === 'chase' && n.targetId === p.id && d < 20) || d < 5.5) {
   415→   415→          bot.state = 'flee'
   416→   416→          bot.fleeFrom = { x: n.x, z: n.z }
   417→   417→          bot.stateUntil = now + rand(2200, 3600)
   418→   418→          bot.partner = null
   419→   419→          bot.marchCp = null
   420→   420→          break
   421→   421→        }
   422→   422→      }
   423→   423→    }
   424→   424→
   425→   425→    // --- emote chatter (bots stand still and perform) ---
   426→   426→    if (now < bot.emoteUntil) {
   427→   427→      p.anim = bot.emoteAnim
   428→   428→      p.lastStateTime = now
   429→   429→      return
   430→   430→    }
   431→   431→    if (now >= bot.nextEmoteAt && bot.state !== 'flee' && bot.state !== 'arrive') {
   432→   432→      const emote = pick(BOT_EMOTES)
   433→   433→      bot.emoteUntil = now + rand(1500, 2200)
   434→   434→      bot.emoteAnim = EMOTE_ANIM[emote] ?? 'celebrate'
   435→   435→      p.anim = bot.emoteAnim
   436→   436→      bot.nextEmoteAt = now + rand(26_000, 80_000)
   437→   437→      this.io.to(this.room.id).emit('game:emote', p.id, emote)
   438→   438→      return
   439→   439→    }
   440→   440→
   441→   441→    // --- one-liners ---
   442→   442→    if (now >= bot.nextQuipAt && bot.state !== 'flee') {
   443→   443→      bot.nextQuipAt = now + rand(50_000, 170_000)
   444→   444→      this.io.to(this.room.id).emit('game:quickmsg', p.id, randInt(0, 9))
   445→   445→    }
   446→   446→
   447→   447→    // --- supply runs: grab a nearby non-token item ---
   448→   448→    if (now >= bot.nextPickupAt && bot.state !== 'flee') {
   449→   449→      bot.nextPickupAt = now + rand(30_000, 80_000)
   450→   450→      for (const def of WORLD.items) {
   451→   451→        if (def.type === 'token') continue
   452→   452→        if (dist2d(def.x, def.z, p.x, p.z) > 2.5) continue
   453→   453→        const st = this.room.itemStates.get(def.id)
   454→   454→        if (st && !st.taken && this.room.pickup(p.id, def.id)) break
   455→   455→      }
   456→   456→    }
   457→   457→
   458→   458→    // --- state machine ---
   459→   459→    switch (bot.state) {
   460→   460→      case 'arrive': {
   461→   461→        // walk from the spawn gate into the plaza, then blend into the crowd
   462→   462→        if (this.moveToward(bot, bot.goal!.x, bot.goal!.z, dt) || dist2d(p.x, p.z, bot.goal!.x, bot.goal!.z) < 4) {
   463→   463→          bot.state = 'wander'
   464→   464→          bot.goal = randomGoal()
   465→   465→          bot.wp = null
   466→   466→        }
   467→   467→        break
   468→   468→      }
   469→   469→      case 'wander': {
   470→   470→        if (!bot.wp) { this.pickWaypoint(bot); break }
   471→   471→        if (this.moveToward(bot, bot.wp.x, bot.wp.z, dt)) {
   472→   472→          bot.wp = null
   473→   473→          // arrived at the wider goal?
   474→   474→          if (bot.goal && dist2d(p.x, p.z, bot.goal.x, bot.goal.z) < 7) {
   475→   475→            bot.goal = randomGoal()
   476→   476→            if (chance(0.5)) { this.enterIdle(bot, now); break }
   477→   477→            if (chance(0.45)) { this.trySocialize(bot, now); break }
   478→   478→          }
   479→   479→        }
   480→   480→        break
   481→   481→      }
   482→   482→      case 'idle': {
   483→   483→        p.anim = 'idle'
   484→   484→        p.ry += Math.sin(now / 900 + p.x) * dt * 0.6 // look around
   485→   485→        if (now >= bot.stateUntil) {
   486→   486→          if (chance(0.35)) this.trySocialize(bot, now)
   487→   487→          else { bot.state = 'wander'; bot.goal = randomGoal(); bot.wp = null }
   488→   488→        }
   489→   489→        break
   490→   490→      }
   491→   491→      case 'social': {
   492→   492→        const partner = partnerPos(this, bot)
   493→   493→        if (!partner) { bot.state = 'wander'; bot.wp = null; bot.goal = randomGoal(); break }
   494→   494→        const d = dist2d(p.x, p.z, partner.x, partner.z)
   495→   495→        if (d > 4.5) {
   496→   496→          this.moveToward(bot, partner.x, partner.z, dt)
   497→   497→        } else {
   498→   498→          p.anim = 'idle'
   499→   499→          p.ry = angleLerp(p.ry, Math.atan2(partner.x - p.x, partner.z - p.z), dt * 6)
   500→   500→        }
   501→   501→        if (now >= bot.stateUntil) { bot.state = 'wander'; bot.partner = null; bot.wp = null; bot.goal = randomGoal() }
   502→   502→        break
   503→   503→      }
   504→   504→      case 'flee': {
   505→   505→        if (bot.fleeFrom) this.moveAway(bot, bot.fleeFrom.x, bot.fleeFrom.z, dt)
   506→   506→        if (now >= bot.stateUntil) {
   507→   507→          bot.state = 'wander'
   508→   508→          bot.fleeFrom = null
   509→   509→          bot.node = nearestNode(p.x, p.z)
   510→   510→          bot.goal = randomGoal()
   511→   511→          bot.wp = null
   512→   512→          bot.speed = rand(2.3, 3.4)
   513→   513→        }
   514→   514→        break
   515→   515→      }
   516→   516→      case 'march': {
   517→   517→        const cp = WORLD.checkpoints.find((c) => c.id === bot.marchCp)
   518→   518→        if (!cp || this.room.checkpoints.has(cp.id)) {
   519→   519→          // checkpoint reached (by anyone) — celebrate then disperse
   520→   520→          if (chance(0.6)) { bot.emoteUntil = now + 2200; bot.emoteAnim = 'celebrate' }
   521→   521→          bot.state = 'wander'
   522→   522→          bot.marchCp = null
   523→   523→          bot.goal = randomGoal()
   524→   524→          bot.wp = null
   525→   525→          break
   526→   526→        }
   527→   527→        if (this.moveToward(bot, bot.goal!.x, bot.goal!.z, dt)) {
   528→   528→          if (chance(0.5)) { bot.emoteUntil = now + 2000; bot.emoteAnim = 'celebrate' }
   529→   529→          bot.state = 'wander'
   530→   530→          bot.marchCp = null
   531→   531→          bot.goal = randomGoal()
   532→   532→          bot.wp = null
   533→   533→        }
   534→   534→        break
   535→   535→      }
   536→   536→    }
   537→   537→
   538→   538→    p.lastX = p.x
   539→   539→    p.lastZ = p.z
   540→   540→    p.lastStateTime = now
   541→   541→  }

   543→   543→  private enterIdle(bot: Bot, now: number) {
   544→   544→    bot.state = 'idle'
   545→   545→    bot.stateUntil = now + rand(2000, 6000)
   546→   546→    bot.wp = null
   547→   547→  }
   548→   548→
   549→   549→  private trySocialize(bot: Bot, now: number) {
   550→   550→    const p = bot.p
   551→   551→    // prefer a nearby human (bots gravitate toward real protesters), else another bot
   552→   552→    let target: { id: string; x: number; z: number } | null = null
   553→   553→    for (const h of this.room.humans()) {
   554→   554→      if (h.detained || h.id === p.id) continue
   555→   555→      const d = dist2d(h.x, h.z, p.x, p.z)
   556→   556→      if (d < 10 && d > 1 && chance(0.5)) { target = { id: h.id, x: h.x, z: h.z }; break }
   557→   557→    }
   558→   558→    if (!target) {
   559→   559→      for (const b of this.bots.values()) {
   560→   560→        if (b.p.id === p.id || b.p.detained || b.p.dazed || b.state === 'march' || b.state === 'flee') continue
   561→   561→        const d = dist2d(b.p.x, b.p.z, p.x, p.z)
   562→   562→        if (d < 9 && d > 1.2) { target = { id: b.p.id, x: b.p.x, z: b.p.z }; break }
   563→   563→      }
   564→   564→    }
   565→   565→    if (!target) { this.enterIdle(bot, now); return }
   566→   566→    bot.state = 'social'
   567→   567→    bot.partner = target.id
   568→   568→    bot.stateUntil = now + rand(4500, 10_000)
   569→   569→    bot.wp = null
   570→   570→    if (chance(0.5)) this.io.to(this.room.id).emit('game:quickmsg', p.id, randInt(0, 9))
   571→   571→    // mutual: if the partner is a bot, it turns to face us too
   572→   572→    const other = this.bots.get(target.id)
   573→   573→    if (other && other.state !== 'flee' && other.state !== 'march') {
   574→   574→      other.state = 'social'
   575→   575→      other.partner = p.id
   576→   576→      other.stateUntil = bot.stateUntil
   577→   577→      if (chance(0.5)) this.io.to(this.room.id).emit('game:quickmsg', other.p.id, randInt(0, 9))
   578→   578→    }
   579→   579→  }

   581→   581→  /** Pick the next nav-graph waypoint, biased toward the current goal. */
   582→   582→  private pickWaypoint(bot: Bot) {
   583→   583→    const node = NAV[bot.node]
   584→   584→    if (!node || node.links.length === 0) { bot.node = nearestNode(bot.p.x, bot.p.z); return }
   585→   585→    let next: number
   586→   586→    if (bot.goal && chance(0.68)) {
   587→   587→      let best = node.links[0]
   588→   588→      let bestD = Infinity
   589→   589→      for (const l of node.links) {
   590→   590→        const d = (NAV[l].x - bot.goal.x) ** 2 + (NAV[l].z - bot.goal.z) ** 2
   591→   591→        if (d < bestD) { bestD = d; best = l }
   592→   592→      }
   593→   593→      next = best
   594→   594→    } else {
   595→   595→      next = pick(node.links)
   596→   596→    }
   597→   597→    bot.node = next
   598→   598→    const n = NAV[next]
   599→   599→    // small lateral jitter so bots don't walk in perfect single file
   600→   600→    const jx = n.x + rand(-2.2, 2.2)
   601→   601→    const jz = n.z + rand(-2.2, 2.2)
   602→   602→    bot.wp = pointBlocked(jx, jz, 1.0) ? { x: n.x, z: n.z } : { x: jx, z: jz }
   603→   603→  }
   604→   604→
   605→   605→  /** Move toward a point; returns true on arrival. Handles anim + facing. */
   606→   606→  private moveToward(bot: Bot, tx: number, tz: number, dt: number): boolean {
   607→   607→    const p = bot.p
   608→   608→    const dx = tx - p.x
   609→   609→    const dz = tz - p.z
   610→   610→    const d = Math.hypot(dx, dz)
   611→   611→    if (d < 0.6) { p.anim = 'idle'; return true }
   612→   612→    const step = bot.speed * dt
   613→   613→    if (d <= step) { p.x = tx; p.z = tz; p.anim = 'idle'; return true }
   614→   614→    p.x += (dx / d) * step
   615→   615→    p.z += (dz / d) * step
   616→   616→    p.ry = angleLerp(p.ry, Math.atan2(dx, dz), dt * 8)
   617→   617→    p.anim = bot.speed > 4.6 ? 'run' : 'walk'
   618→   618→    return false
   619→   619→  }
   620→   620→
   621→   621→  /** Run away from a threat, steering around walls. */
   622→   622→  private moveAway(bot: Bot, fx: number, fz: number, dt: number) {
   623→   623→    const p = bot.p
   624→   624→    const speed = 6.6
   625→   625→    let dx = p.x - fx
   626→   626→    let dz = p.z - fz
   627→   627→    const d = Math.hypot(dx, dz) || 1
   628→   628→    dx /= d; dz /= d
   629→   629→    const step = speed * dt
   630→   630→    // try the direct route, then swing up to ±100° if a wall is in the way
   631→   631→    for (const ang of [0, 0.5, -0.5, 1.0, -1.0, 1.6, -1.6]) {
   632→   632→      const ca = Math.cos(ang), sa = Math.sin(ang)
   633→   633→      const nx = p.x + (dx * ca - dz * sa) * step
   634→   634→      const nz = p.z + (dx * sa + dz * ca) * step
   635→   635→      if (nx < -45 || nx > 45 || nz > 7 || nz < NAV_Z_MAX + 2) continue
   636→   636→      if (!pointBlocked(nx, nz, 1.1)) {
   637→   637→        p.x = nx
   638→   638→        p.z = nz
   639→   639→        p.ry = angleLerp(p.ry, Math.atan2(dx * ca - dz * sa, dx * sa + dz * ca), dt * 10)
   640→   640→        p.anim = 'run'
   641→   641→        return
   642→   642→      }
   643→   643→    }
   644→   644→    p.anim = 'run' // cornered — keep running on the spot
   645→   645→  }

   647→   647→  // ---------- AOI broadcast (replaces Room's flat game:players blast) ----------
   648→   648→
   649→   649→  private broadcastPlayers() {
   650→   650→    if (this.disposed) return
   651→   651→    const humans: PlayerState[] = this.room.humans()
   652→   652→    if (humans.length === 0) return
   653→   653→    const bots: PlayerState[] = []
   654→   654→    for (const b of this.bots.values()) bots.push(b.p)
   655→   655→    const n = bots.length
   656→   656→    // rotating far-window: every far bot gets a refresh ≈ every (n / FAR_WINDOW) ticks
   657→   657→    const windowIds = new Set<string>()
   658→   658→    if (n > 0) {
   659→   659→      for (let i = 0; i < Math.min(BOT_FAR_WINDOW, n); i++) windowIds.add(bots[(this.farCursor + i) % n].id)
   660→   660→      this.farCursor = (this.farCursor + BOT_FAR_WINDOW) % n
   661→   661→    }
   662→   662→    const humanPayload = humans.map(playerNet)
   663→   663→    for (const h of humans) {
   664→   664→      const payload: PlayerNetState[] = [...humanPayload]
   665→   665→      for (const b of bots) {
   666→   666→        const dx = b.x - h.x
   667→   667→        const dz = b.z - h.z
   668→   668→        if (dx * dx + dz * dz <= BOT_AOI * BOT_AOI || windowIds.has(b.id)) payload.push(playerNet(b))
   669→   669→      }
   670→   670→      this.io.to(h.id).emit('game:players', payload)
   671→   671→    }
   672→   672→  }
   673→   673→}

   675→   675→/** Partner position for a socializing bot (human or bot). */
   676→   676→function partnerPos(season: Season, bot: Bot): { x: number; z: number } | null {
   677→   677→  if (!bot.partner) return null
   678→   678→  const other = season.room.getPlayer(bot.partner)
   679→   679→  if (!other || other.detained || other.dazed) return null
   680→   680→  return { x: other.x, z: other.z }
   681→   681→}

   683→   683→// ---------- matchmaking ----------
   684→   684→
   685→   685→export class SeasonManager {
   686→   686→  private seasons = new Map<string, Season>()
   687→   687→  private counter = 0
   688→   688→
   689→   689→  constructor(private io: SocketIOServer, private rooms: Map<string, Room>) {}
   690→   690→
   691→   691→  /**
   692→   692→   * Find the liveliest season with room for one more human, or create the next one.
   693→   693→   * "Full" = SEASON_MAX_HUMANS real players — bots always backfill the rest.
   694→   694→   */
   695→   695→  assign(): Season {
   696→   696→    let best: Season | null = null
   697→   697→    for (const s of this.seasons.values()) {
   698→   698→      if (s.room.humansCount() >= SEASON_MAX_HUMANS) continue
   699→   699→      if (!best) { best = s; continue }
   700→   700→      const bh = best.room.humansCount()
   701→   701→      const sh = s.room.humansCount()
   702→   702→      if (sh > bh || (sh === bh && s.createdAt > best.createdAt)) best = s
   703→   703→    }
   704→   704→    if (best) return best
   705→   705→    this.counter++
   706→   706→    const season = new Season(this.io, `sea_${this.counter}`, `SEASON #${this.counter}`, (id) => {
   707→   707→      const s = this.seasons.get(id)
   708→   708→      if (s) { s.dispose(); this.seasons.delete(id); this.rooms.delete(id) }
   709→   709→    })
   710→   710→    this.seasons.set(season.id, season)
   711→   711→    this.rooms.set(season.id, season.room)
   712→   712→    return season
   713→   713→  }
   714→   714→
   715→   715→  get(roomId: string): Season | undefined {
   716→   716→    return this.seasons.get(roomId)
   717→   717→  }
   718→   718→
   719→   719→  /** Real (socket-backed) players across every season — for the global join cap. */
   720→   720→  totalHumans(): number {
   721→   721→    let n = 0
   722→   722→    for (const s of this.seasons.values()) n += s.room.humansCount()
   723→   723→    return n
   724→   724→  }
   725→   725→}
